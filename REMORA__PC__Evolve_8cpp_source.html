34/**
35 * Midpoint-method advection in terrain-following coordinates. Horizontally the particle
36 * moves with u and v at a fixed fractional w-level index (see REMORA_PC.H), so it follows
37 * the s-levels over sloping bathymetry; vertically it moves across the levels at
38 * omega / Hz. The index is reflected off the bed and the free surface and then mapped
39 * back to a height, so a particle can never leave the water column.
40 *
41 * Particle positions are in the uniform coordinates of the AMReX geometry, whose cell size
42 * is (prob_hi - prob_lo) / n_cell, while u and v are physical velocities along the grid
43 * directions. They are converted with the local metric, dx/dt = u * pm * dx_geom and
44 * dy/dt = v * pn * dy_geom, so particles cover the right physical distance on stretched or
45 * curvilinear grids; on a uniform grid the factors are 1.
46 *
47 * The index is recovered from the stored height on the current grid, so a particle stays
48 * at a fixed height, not a fixed index, while the free surface moves between steps.
49 *
50 * @param a_lev level at which to evolve particles
51 * @param a_dt time step
52 * @param a_umac u, v and omega, the velocity across the w-levels (z-nodal, m/s)
53 * @param a_z_height z nodal positions of grid
54 * @param a_pm 1/dx, the inverse physical cell width in x
55 * @param a_pn 1/dy, the inverse physical cell width in y
56 */

103 v_ptr[2] = soa.GetRealData(REMORAParticlesRealIdxSoA::vz).data();

117 const ParticleReal x = p.pos(0);

119 int kfound;

124

126 plo, dxi, uarr);

130 // puts s on the next level; only then are the level heights recomputed
131 const int ks = level_index_cell(s, nz);
132 const Real dz = (ks == kfound) ? dz_found
133 : level_height(x, y, ks+1, plo, dxi, zheight)

137 // Physical velocity to the rate of change of the geometry coordinate:
138 // pm * dx_geom is the number of geometry metres per physical metre

140 const Real ydot = v * interpolate_horizontal(x, y, 0, false, plo, dxi, pnarr) / dxi[1];

143 ParticleReal z_old = zero;
144 if (ipass == 0) {

148 p.pos(0) += static_cast<ParticleReal>(myhalf*a_dt*xdot);
149 p.pos(1) += static_cast<ParticleReal>(myhalf*a_dt*ydot);
150 s_new = s + myhalf*a_dt*sdot;
151 } else {
152 const auto s_old = static_cast<Real>(v_ptr[2][i]);
153 z_old = level_index_to_height(v_ptr[0][i], v_ptr[1][i], s_old, nz,
154 plo, dxi, zheight);
155 p.pos(0) = v_ptr[0][i] + static_cast<ParticleReal>(a_dt*xdot);
156 p.pos(1) = v_ptr[1][i] + static_cast<ParticleReal>(a_dt*ydot);
157 s_new = s_old + a_dt*sdot;
158 }
159
160 // omega vanishes at the bed and the surface, so only the time
161 // discretization can carry a particle past them: reflect it back
162 if (s_new < zero) { s_new = -s_new; }
163 if (s_new > static_cast<Real>(nz)) { s_new = Real(2*nz) - s_new; }
164 s_new = amrex::max(zero, amrex::min(static_cast<Real>(nz), s_new));
165
166 p.pos(2) = level_index_to_height(p.pos(0), p.pos(1), s_new, nz,
167 plo, dxi, zheight);
168 p.idata(REMORAParticlesIntIdxAoS::k) = level_index_cell(s_new, nz);
169
170 if (ipass == 1) {
171 v_ptr[0][i] = u;
172 v_ptr[1][i] = v;
173 v_ptr[2][i] = (p.pos(2) - z_old) / a_dt;
174 }
175 });
176 }
177 }
