111/**
112 * Make a periodic quantity (a longitude, period 360, or an angle, period 2 pi) continuous
113 * over a FAB that holds a whole domain, by bringing every value within half a period of
114 * the value in the FAB's first cell. A regional grid that crosses the branch cut then
115 * averages across it correctly: 179.5 and -179.5 average to 180, not to 0. The result may
116 * lie outside [-180, 180); that is the same longitude, and users of it normalize.
117 *
118 * @param[inout] local the field as read on this rank: the whole domain in one piece,
119 * or nothing
120 * @param[in ] period the period
121 */
122void
123unwrap_to_first_cell (NCLocalFab& local, Real period)
124{
125 AMREX_ALWAYS_ASSERT(local.pieces.size() <= 1);
126 if (local.pieces.empty()) { return; }
127 FArrayBox& fab = local.pieces[0];
128 Real ref = Real(0.0);
129 Gpu::copy(Gpu::deviceToHost, fab.dataPtr(), fab.dataPtr() + 1, &ref);
130 auto arr = fab.array();
131 ParallelFor(fab.box(), [=] AMREX_GPU_DEVICE (int i, int j, int k)
132 {
133 arr(i,j,k) -= period * std::round((arr(i,j,k) - ref) / period);
134 });
135 Gpu::streamSynchronize();
136}

197
198 // Each rank reads the part of the file its own FABs cover.
199 const NCRegion region = nc_local_region({cons_new[lev], xvel_new[lev], yvel_new[lev]});
200
201 for (int idx = 0; idx < num_boxes_at_level[lev]; idx++)
202 {
203 read_data_from_netcdf(lev, boxes_at_level[lev][idx], region, nc_init_file[lev][idx],
204 NC_temp_fab[idx], NC_salt_fab[idx],
205 NC_xvel_fab[idx], NC_yvel_fab[idx]);
206 }
207
208
209 MultiFab mf_temp(*cons_new[lev], make_alias, Temp_comp, 1);
210 MultiFab mf_salt(*cons_new[lev], make_alias, Salt_comp, 1);
211
212#ifdef _OPENMP
213#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
214#endif
215 {
216 // Don't tile this since we are operating on full FABs in this routine
217 for ( MFIter mfi(*cons_new[lev], false); mfi.isValid(); ++mfi )
218 {
219 // Define fabs for holding the initial data
220 FArrayBox &temp_fab = mf_temp[mfi];
221 FArrayBox &salt_fab = mf_salt[mfi];
222 FArrayBox &xvel_fab = (*xvel_new[lev])[mfi];
223 FArrayBox &yvel_fab = (*yvel_new[lev])[mfi];
224
225 init_state_from_netcdf(lev, temp_fab, salt_fab,

227 NC_temp_fab, NC_salt_fab,
228 NC_xvel_fab, NC_yvel_fab);
229 } // mf
230 } // omp

232 // Zero every tracer past salinity. This is the starting point for the passive
233 // scalars, which the read below overwrites for each dye the file carries; the
234 // biology block is overwritten by init_biology_ic immediately after, so zeroing it

238 // Passive scalars come from the same file, and from the same remora.ic_type that
239 // brought us here, so they are read here rather than through a source flag of
240 // their own the way biology is.
241 init_scalars_from_netcdf(lev);

584void
585REMORA::init_grid_vars_from_netcdf (int lev)
586{
587 BL_PROFILE("REMORA::init_grid_vars_from_netcdf()");
588 // *** FArrayBox's at this level for holding the INITIAL data
589 Vector<NCLocalFab> NC_pm_fab ; NC_pm_fab.resize(num_boxes_at_level[lev]);
590 Vector<NCLocalFab> NC_pn_fab ; NC_pn_fab.resize(num_boxes_at_level[lev]);
591
592 Vector<NCLocalFab> NC_xr_fab ; NC_xr_fab.resize(num_boxes_at_level[lev]);
593 Vector<NCLocalFab> NC_yr_fab ; NC_yr_fab.resize(num_boxes_at_level[lev]);
594 Vector<NCLocalFab> NC_xu_fab ; NC_xu_fab.resize(num_boxes_at_level[lev]);
595 Vector<NCLocalFab> NC_yu_fab ; NC_yu_fab.resize(num_boxes_at_level[lev]);
596 Vector<NCLocalFab> NC_xv_fab ; NC_xv_fab.resize(num_boxes_at_level[lev]);
597 Vector<NCLocalFab> NC_yv_fab ; NC_yv_fab.resize(num_boxes_at_level[lev]);
598 Vector<NCLocalFab> NC_xp_fab ; NC_xp_fab.resize(num_boxes_at_level[lev]);
599 Vector<NCLocalFab> NC_yp_fab ; NC_yp_fab.resize(num_boxes_at_level[lev]);
600
601 // Optional spherical psi coordinates (see read_spherical_grid_vars_from_netcdf)
602 Vector<NCLocalFab> NC_lonp_fab ; NC_lonp_fab.resize(num_boxes_at_level[lev]);
603 Vector<NCLocalFab> NC_latp_fab ; NC_latp_fab.resize(num_boxes_at_level[lev]);
604 bool have_spherical_psi = true;
605
606 // Optional spherical rho coordinates and grid angle (see read_optional_grid_vars_from_netcdf)
607 Vector<NCLocalFab> NC_lonr_fab ; NC_lonr_fab.resize(num_boxes_at_level[lev]);
608 Vector<NCLocalFab> NC_latr_fab ; NC_latr_fab.resize(num_boxes_at_level[lev]);
609 Vector<NCLocalFab> NC_angler_fab; NC_angler_fab.resize(num_boxes_at_level[lev]);
610 bool have_spherical_rho = true;
611 bool have_angle = true;
612
613 // lon/lat and angle are laid out like x_rho and x_psi, so these cover them too.
614 const NCRegion region = nc_local_region({vec_pm[lev].get(), vec_xr[lev].get(), vec_xu[lev].get(),
615 vec_xv[lev].get(), vec_xp[lev].get()});
616
617 for (int idx = 0; idx < num_boxes_at_level[lev]; idx++)
618 {
619 read_grid_vars_from_netcdf(lev, boxes_at_level[lev][idx], region, nc_grid_file[lev][idx],
620 NC_pm_fab[idx], NC_pn_fab[idx],
621 NC_xr_fab[idx], NC_yr_fab[idx],
622 NC_xu_fab[idx], NC_yu_fab[idx],
623 NC_xv_fab[idx], NC_yv_fab[idx],
624 NC_xp_fab[idx], NC_yp_fab[idx]);
625
626 // All boxes at this level must agree: a partial set would leave holes in
627 // the corner mesh that a conservative remap cannot detect.
628 have_spherical_psi = read_optional_grid_vars_from_netcdf(
629 boxes_at_level[lev][idx], region, nc_grid_file[lev][idx],
630 {"lon_psi", "lat_psi"}, {&NC_lonp_fab[idx], &NC_latp_fab[idx]},
631 "spherical psi coordinates") && have_spherical_psi;
632
633 // Mirror vec_xp exactly: same nodal psi BoxArray, DistributionMap and ngrow.
634 if (have_spherical_psi && vec_lonp[lev] == nullptr) {
635 vec_lonp[lev].reset(new MultiFab(vec_xp[lev]->boxArray(), vec_xp[lev]->DistributionMap(),
636 1, vec_xp[lev]->nGrowVect()));
637 vec_latp[lev].reset(new MultiFab(vec_xp[lev]->boxArray(), vec_xp[lev]->DistributionMap(),
638 1, vec_xp[lev]->nGrowVect()));
639 }
640
641 have_spherical_rho = read_optional_grid_vars_from_netcdf(
642 boxes_at_level[lev][idx], region, nc_grid_file[lev][idx],
643 {"lon_rho", "lat_rho"}, {&NC_lonr_fab[idx], &NC_latr_fab[idx]},
644 "spherical rho coordinates") && have_spherical_rho;
645 have_angle = read_optional_grid_vars_from_netcdf(
646 boxes_at_level[lev][idx], region, nc_grid_file[lev][idx],
647 {"angle"}, {&NC_angler_fab[idx]},
648 "grid rotation angle") && have_angle;
649
650 // Mirror vec_xr exactly: same rho BoxArray, DistributionMap and ngrow.
651 if (have_spherical_rho && vec_lonr[lev] == nullptr) {
652 vec_lonr[lev].reset(new MultiFab(vec_xr[lev]->boxArray(), vec_xr[lev]->DistributionMap(),
653 1, vec_xr[lev]->nGrowVect()));
654 vec_latr[lev].reset(new MultiFab(vec_xr[lev]->boxArray(), vec_xr[lev]->DistributionMap(),
655 1, vec_xr[lev]->nGrowVect()));
656 }
657 if (have_angle && vec_angler[lev] == nullptr) {
658 vec_angler[lev].reset(new MultiFab(vec_xr[lev]->boxArray(), vec_xr[lev]->DistributionMap(),
659 1, vec_xr[lev]->nGrowVect()));
660 }
661 // These are made after the level's other fields were zeroed in dropped boxes
662 // (zero_level0_dry_ghost_cells), so zero theirs now
663 if (lev == 0) {
664 for (auto* v : {&vec_lonr, &vec_latr, &vec_angler}) {
665 if ((*v)[lev]) { set_dry_ghost_cells(*(*v)[lev], zero); }
666 }
667 }
668
669#ifdef _OPENMP
670#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
671#endif
672 {
673 // Don't tile this since we are operating on full FABs in this routine
674 for ( MFIter mfi(*cons_new[lev], false); mfi.isValid(); ++mfi )
675 {
676 FArrayBox &pm_fab = (*vec_pm[lev])[mfi];
677 FArrayBox &pn_fab = (*vec_pn[lev])[mfi];
678 FArrayBox &xr_fab = (*vec_xr[lev])[mfi];
679 FArrayBox &yr_fab = (*vec_yr[lev])[mfi];
680 FArrayBox &xu_fab = (*vec_xu[lev])[mfi];
681 FArrayBox &yu_fab = (*vec_yu[lev])[mfi];
682 FArrayBox &xv_fab = (*vec_xv[lev])[mfi];
683 FArrayBox &yv_fab = (*vec_yv[lev])[mfi];
684 FArrayBox &xp_fab = (*vec_xp[lev])[mfi];
685 FArrayBox &yp_fab = (*vec_yp[lev])[mfi];
686
687 //
688 // copy_to copies on intersection from the pieces read for this rank, which cover its FABs
689 //
690
691 NC_pm_fab[idx].copy_to(pm_fab);
692 NC_pn_fab[idx].copy_to(pn_fab);
693
694 NC_xr_fab[idx].copy_to(xr_fab);
695 NC_yr_fab[idx].copy_to(yr_fab);
696 NC_xu_fab[idx].copy_to(xu_fab);
697 NC_yu_fab[idx].copy_to(yu_fab);
698 NC_xv_fab[idx].copy_to(xv_fab);
699 NC_yv_fab[idx].copy_to(yv_fab);
700 NC_xp_fab[idx].copy_to(xp_fab);
701 NC_yp_fab[idx].copy_to(yp_fab);
702
703 if (have_spherical_psi) {
704 NC_lonp_fab[idx].copy_to((*vec_lonp[lev])[mfi]);
705 NC_latp_fab[idx].copy_to((*vec_latp[lev])[mfi]);
706 }
707 if (have_spherical_rho) {
708 NC_lonr_fab[idx].copy_to((*vec_lonr[lev])[mfi]);
709 NC_latr_fab[idx].copy_to((*vec_latr[lev])[mfi]);
710 }
711 if (have_angle) {
712 NC_angler_fab[idx].copy_to((*vec_angler[lev])[mfi]);
713 }
714 } // mf
715 } // omp
716 } // idx
717
718 // Drop any partial allocation so the getter's null contract stays honest.
719 if (!have_spherical_psi) {
720 vec_lonp[lev].reset();
721 vec_latp[lev].reset();
722 }
723 if (!have_spherical_rho) {
724 vec_lonr[lev].reset();
725 vec_latr[lev].reset();
726 }
727 if (!have_angle) {
728 vec_angler[lev].reset();
729 }
730
731 Real dummy_time = zero;
732 if (lev > 0) {
733 FillPatch(lev,dummy_time,*vec_pm[lev],GetVecOfPtrs(vec_pm),
734 foextrap_periodic_bc(),
735 BdyVars::null,0,false);
736 FillPatch(lev,dummy_time,*vec_pn[lev],GetVecOfPtrs(vec_pn),
737 foextrap_periodic_bc(),
738 BdyVars::null,0,false);
739 }
740
741
742 //
743 // We need values of pm and pn outside the domain so we fill
744 // them here with foextrap
745 //
746 // We first fill interior ghost cells because we will need to extrapolate
747 // from ghost cells inside the domain to ghost cells outside the domain
748 //
749 vec_pm[lev]->FillBoundary(geom[lev].periodicity());
750 vec_pn[lev]->FillBoundary(geom[lev].periodicity());
751
752 vec_xr[lev]->FillBoundary(geom[lev].periodicity());
753 vec_yr[lev]->FillBoundary(geom[lev].periodicity());
754 vec_xu[lev]->FillBoundary(geom[lev].periodicity());
755 vec_yu[lev]->FillBoundary(geom[lev].periodicity());
756 vec_xv[lev]->FillBoundary(geom[lev].periodicity());
757 vec_yv[lev]->FillBoundary(geom[lev].periodicity());
758 vec_xp[lev]->FillBoundary(geom[lev].periodicity());
759 vec_yp[lev]->FillBoundary(geom[lev].periodicity());
760 if (vec_lonp[lev] != nullptr) {
761 vec_lonp[lev]->FillBoundary(geom[lev].periodicity());
762 vec_latp[lev]->FillBoundary(geom[lev].periodicity());
763 }
764
765 extrapolate_metric_to_physical_boundaries(*vec_pm[lev], geom[lev]);
766 extrapolate_metric_to_physical_boundaries(*vec_pn[lev], geom[lev]);
767
768 // The file supplies the one-cell ring around the domain. The ghost rows beyond it
769 // continue the coordinates linearly, so anything interpolated onto those cells lands
770 // where they are rather than on a copy of the edge; longitude steps are taken modulo
771 // 360 so a ring that crosses the dateline continues past it instead of jumping. The
772 // angle is copied outward like a metric.
773 if (vec_lonr[lev] != nullptr) {
774 vec_lonr[lev]->FillBoundary(geom[lev].periodicity());
775 vec_latr[lev]->FillBoundary(geom[lev].periodicity());
776 extrapolate_metric_to_physical_boundaries(*vec_lonr[lev], geom[lev], true, Real(360.0));
777 extrapolate_metric_to_physical_boundaries(*vec_latr[lev], geom[lev], true);
778 }
779 if (vec_angler[lev] != nullptr) {
780 vec_angler[lev]->FillBoundary(geom[lev].periodicity());
781 extrapolate_metric_to_physical_boundaries(*vec_angler[lev], geom[lev]);
782 }
783}

923 * @param lev Integer specifying the current level
924 */
925void
926REMORA::init_bdry_from_netcdf (int lev)
927{
928 BL_PROFILE("REMORA::init_bdry_from_netcdf()");
929 if (nc_bdry_file.empty() || nc_bdry_file[0].empty()) {
930 amrex::Error("NetCDF boundary file name must be provided via input");
931 }
932
933 // One boundary series per BdyVars slot. Every cell-centered tracer gets one, named
934 // for the variable itself ("temp", "salt", "NO3", ...), so the file is expected to
935 // hold e.g. NO3_west following the same convention ROMS uses. A series whose
936 // variable needs no boundary data reads nothing -- NCTimeSeriesBoundary only asks
937 // for the sides flagged in phys_bc_need_data -- so tracers left on a local BC do
938 // not require the file to contain anything for them.
939 amrex::Vector<std::string> field_name (num_bdy_vars());
940 amrex::Vector<IntVect > index_types(num_bdy_vars());
941 std::vector<bool > is_2d (num_bdy_vars());
942
943 field_name[BdyVars::u] = "u"; index_types[BdyVars::u] = IntVect(1,0,0); is_2d[BdyVars::u] = false;
944 field_name[BdyVars::v] = "v"; index_types[BdyVars::v] = IntVect(0,1,0); is_2d[BdyVars::v] = false;
945 for (int icomp = 0; icomp < ncons; ++icomp) {
946 const int ibdy = BdyVars::cons(icomp);
947 field_name[ibdy] = cons_names[icomp];
948 index_types[ibdy] = IntVect(0,0,0);
949 is_2d[ibdy] = false;
950 }
951 field_name[bdy_ubar()] = "ubar"; index_types[bdy_ubar()] = IntVect(1,0,0); is_2d[bdy_ubar()] = true;
952 field_name[bdy_vbar()] = "vbar"; index_types[bdy_vbar()] = IntVect(0,1,0); is_2d[bdy_vbar()] = true;
953 field_name[bdy_zeta()] = "zeta"; index_types[bdy_zeta()] = IntVect(0,0,0); is_2d[bdy_zeta()] = true;
954
955 amrex::Print() << "DOING INIT AT LEVEL " << lev << std::endl;
956 // A level removed by a regrid and later made again comes back through here;
957 // start from an empty vector so the slots below stay at ivar.
958 boundary_series[lev].clear();
959 int rx = 1; int ry = 1;
960 if (lev > 0) {
961 for (int k = lev-1; k >= 0; k--) {
962 rx *= ref_ratio[k][0];
963 ry *= ref_ratio[k][1];
964 }
965 }
966 for (int ivar = 0; ivar < num_bdy_vars(); ivar++) {
967 boundary_series[lev].push_back(std::unique_ptr<NCTimeSeriesBoundary>(new NCTimeSeriesBoundary(lev, geom, nc_bdry_file, field_name[ivar],
968 bdry_time_name_byvar[ivar],
969 index_types[ivar],
970 phys_bc_need_data[ivar], is_2d[ivar], rx, ry,
971 solverChoice.time_ref)));
972 boundary_series[lev][ivar]->Initialize();
973 }
