17/** Print a NetCDF shape as "(5848, 69, 81)" */
18std::string shape_to_string (const std::vector<MPI_Offset>& shape)
19{
20 std::string s = "(";
21 for (std::size_t i = 0; i < shape.size(); ++i) {
22 if (i > 0) { s += ", "; }
23 s += std::to_string(static_cast<long long>(shape[i]));
24 }
25 return s + ")";
26}

28/** The spatial shape, ([s,] eta, xi), a native file has for a field on this grid.
29 *
30 * The reader (fill_fab_from_arrays) lays the file out on the grid index for
31 * index: a rho field's first column lands on the ghost cell at -1 and a face
32 * field's on face 0, so a native ROMS file carries the boundary ring and has
33 * rho (ny+2, nx+2), u-face (ny+2, nx+1), v-face (ny+1, nx+2),
34 * with the s_rho count in front for a 3D field.
35 */
36std::vector<MPI_Offset> native_spatial_shape (const amrex::Box& domain,
37 const amrex::IndexType ixtype, bool is2d)
38{
39 std::vector<MPI_Offset> shape;
40 if (!is2d) {
41 shape.push_back(domain.length(2) + (ixtype.nodeCentered(2) ? 1 : 0));
42 }
43 shape.push_back(domain.length(1) + (ixtype.nodeCentered(1) ? 1 : 2));
44 shape.push_back(domain.length(0) + (ixtype.nodeCentered(0) ? 1 : 2));
45 return shape;
46}

48/** Whether the trailing dimensions of a (time, [s,] eta, xi) variable are the native shape */
49bool matches_native_shape (const std::vector<MPI_Offset>& file_shape,
50 const std::vector<MPI_Offset>& native_shape)
51{
52 const std::size_t n = native_shape.size();
53 if (file_shape.size() < n) { return false; }
54 return std::equal(native_shape.begin(), native_shape.end(), file_shape.end() - n);
55}

57/** Why a field of this shape is refused.
58 *
59 * Any shape other than the native one is a field on another grid, and copying it
60 * index for index would put it in the wrong place without a word. Such a field is
61 * interpolated when it can be (see LatLonRegrid); `reason` says why this one cannot.
62 */
63std::string native_shape_mismatch_message (const std::string& field_name,
64 const std::string& file_name,
65 const std::vector<MPI_Offset>& file_shape,
66 const amrex::Box& domain,
67 const amrex::IndexType ixtype, bool is2d,
68 const std::string& reason)
69{
70 const bool nodal_x = ixtype.nodeCentered(0);
71 const bool nodal_y = ixtype.nodeCentered(1);
72 const std::string point = nodal_x ? (nodal_y ? "psi" : "u") : (nodal_y ? "v" : "rho");
73 const std::string file_dims = is2d ? "(time, eta, xi)" : "(time, s, eta, xi)";
74 const std::string grid_dims = is2d ? "(eta, xi)" : "(s, eta, xi)";
75 const std::vector<MPI_Offset> n_cell = is2d
76 ? std::vector<MPI_Offset>{domain.length(1), domain.length(0)}
77 : std::vector<MPI_Offset>{domain.length(2), domain.length(1), domain.length(0)};
78
79 return "NetCDF time series variable '" + field_name + "' in " + file_name
80 + " has shape " + file_dims + " = " + shape_to_string(file_shape)
81 + ", but the level 0 grid needs " + grid_dims + " = "
82 + shape_to_string(native_spatial_shape(domain, ixtype, is2d)) + " for a " + point
83 + "-point field: remora.n_cell is " + grid_dims + " = " + shape_to_string(n_cell)
84 + ", a cell-centered axis has n_cell + 2 points (the ROMS boundary ring) and a face"
85 + " axis n_cell + 1. A field of another shape is taken to be on another grid, and it"
86 + " cannot be interpolated here because " + reason + ".";
87}

89/** Why a field on another grid cannot be interpolated, or empty when it can */
90std::string regrid_refusal (const LatLonRegridOptions* options,
91 const amrex::MultiFab* target_lon, const amrex::MultiFab* target_lat,
92 const amrex::IndexType ixtype, bool is2d)
93{
94 if (options == nullptr) {
95 return "only surface forcing is interpolated from another grid; supply this field on"
96 " the model grid";
97 }
98 if (!options->enabled) {
99 return "remora.frc_regrid = off; supply the field on the model grid, or set it to auto";
100 }
101 if (!is2d || ixtype != amrex::IndexType::TheCellType()) {
102 return "only 2D rho-point forcing is interpolated from another grid, and this field is"
103 " on u or v points or 3D";
104 }
105 if (target_lon == nullptr || target_lat == nullptr) {
106 return "the model grid has no lon_rho and lat_rho to interpolate to; they come from"
107 " remora.nc_grid_file_0 (or remora.nc_grid_file_hires)";
108 }
109 return "";
110}

145NCTimeSeries::SourceGridKind
146NCTimeSeries::source_grid (const amrex::Vector<std::string>& a_file_names,
147 const std::string& a_field_name, const amrex::Box& a_domain,
148 amrex::IndexType ixtype, bool a_is2d)
149{
150 // Per file: 0 model grid (or missing, for Initialize to report), 1 uniform, 2 lat-lon.
151 // The same rule Initialize applies: a shape other than the model grid's, or CF
152 // longitude and latitude axes on the last two dimensions, mark a lat-lon grid.
153 bool any_latlon = false, all_uniform = true;
154 for (const auto& file_name : a_file_names) {
155 int kind = 0;
156 {
157 auto ncf = ncutils::NCFile::open(file_name, NC_NOCLOBBER);
158 ncmpi_begin_indep_data(ncf.ncid);
159 if (amrex::ParallelDescriptor::IOProcessor() && ncf.has_var(a_field_name)) {
160 const auto var = ncf.var(a_field_name);
161 if (var.ndim() == 1) {
162 kind = 1;
163 } else if (var.ndim() == (a_is2d ? 3 : 4) &&
164 !matches_native_shape(var.shape(), native_spatial_shape(a_domain, ixtype, a_is2d))) {
165 kind = 2;
166 }
167 }
168 ncf.close();
169 amrex::ParallelDescriptor::Bcast(&kind, 1, amrex::ParallelDescriptor::IOProcessorNumber());
170 }
171 if (kind == 0 && LatLonRegrid::has_latlon_axes(file_name, a_field_name)) { kind = 2; }
172 any_latlon = any_latlon || kind == 2;
173 all_uniform = all_uniform && kind == 1;
174 }
175 if (any_latlon) { return SourceGridKind::latlon; }
176 return (all_uniform && !a_file_names.empty()) ? SourceGridKind::uniform : SourceGridKind::model;
177}

187void NCTimeSeries::Initialize() {
188 // open file
189 amrex::Print() << "Loading " << field_name << " from NetCDF file(s)" << std::endl;
190
191 // The time field can have any number of names, depending on the field.
192 // If not specified in input file (time_name.empty()) then set it by default
193 if (time_name.empty())
194 {
195 if (field_name.find("wind") != std::string::npos) {
196 time_name = "wind_time";
197 } else if ((field_name.find("str") != std::string::npos) and (field_name[0] == 's')) {
198 time_name = "sms_time";
199 } else if ((field_name.find("str") != std::string::npos) and (field_name[0] == 'b')) {
200 time_name = "bms_time";
201 } else {
202 time_name = "ocean_time";
203 }
204 }
205
206 amrex::Vector<int> file_is_cycle;
207 amrex::Vector<double> file_cycle_length;
208 file_is_spatially_uniform.clear();
209 file_is_regridded.clear();
210 regrid.reset();
211 for (int ifile = 0; ifile < file_names.size(); ++ifile) {
212 const std::string& file_name = file_names[ifile];
213
214 // The time stamps must count days. Their reference date, if the units
215 // name one, is honored: this is the shift that puts them on the model clock.
216 const double offset_days = ReadNetCDFTimeOffsetDays(file_name, time_name, time_ref,
217 true, true);
218
219 // `cycle_length` is normally numeric, so use the NCVar utility directly.
220 // By ROMS convention it is in the same units as the time variable.
221 // REMORA currently requires time units in days, so convert to seconds below.
222 bool l_is_cycle = false;
223 double l_cycle_length = 0.0;
224 int l_is_spatially_uniform = 0;
225 // Shape check verdict, decided on the IO rank and broadcast so that every rank
226 // leaves the collective file calls below before any of them aborts.
227 int l_shape_mismatch = 0;
228 std::vector<long long> l_file_shape(4, 0);
229
230 auto ncf = ncutils::NCFile::open(file_name, NC_NOCLOBBER);
231 ncmpi_begin_indep_data(ncf.ncid);
232
233 if (amrex::ParallelDescriptor::IOProcessor())
234 {
235 auto time_var = ncf.var(time_name);
236
237 l_is_cycle = time_var.has_attr("cycle_length");
238
239 if (l_is_cycle) {
240 std::vector<double> cycle_attr;
241 time_var.get_attr("cycle_length", cycle_attr);
242
243 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
244 cycle_attr.size() == 1,
245 "NetCDF time variable cycle_length attribute must be scalar");
246
247 l_cycle_length = cycle_attr[0] * 60.0 * 60.0 * 24.0;
248 }
249
250 if (!ncf.has_var(field_name)) {
251 amrex::Abort("NetCDF time series variable " + field_name +
252 " not found in " + file_name);
253 }
254
255 const auto field_var = ncf.var(field_name);
256 const int field_rank = field_var.ndim();
257 const int expected_spatial_rank = is2d ? 3 : 4;
258 l_is_spatially_uniform = (field_rank == 1) ? 1 : 0;
259
260 if (field_rank != 1 && field_rank != expected_spatial_rank) {
261 amrex::Abort("Unsupported NetCDF time series rank for variable " +
262 field_name + " in " + file_name + ": rank " +
263 std::to_string(field_rank) + ". Expected rank 1 for " +
264 "uniform time-only data or rank " +
265 std::to_string(expected_spatial_rank) +
266 " for spatial data.");
267 }
268
269 if (field_rank == expected_spatial_rank) {
270 const std::vector<MPI_Offset> file_shape = field_var.shape();
271 for (int i = 0; i < field_rank; ++i) {
272 l_file_shape[i] = static_cast<long long>(file_shape[i]);
273 }
274 l_shape_mismatch = matches_native_shape(
275 file_shape, native_spatial_shape(domain, mf_var->ixType(), is2d)) ? 0 : 1;
276 }
277 }
278
279 ncf.close();
280
281 const int ioproc = amrex::ParallelDescriptor::IOProcessorNumber();
282 int is_cycle_int = l_is_cycle ? 1 : 0;
283 amrex::ParallelDescriptor::Bcast(&is_cycle_int, 1, ioproc);
284 amrex::ParallelDescriptor::Bcast(&l_cycle_length, 1, ioproc);
285 amrex::ParallelDescriptor::Bcast(&l_is_spatially_uniform, 1, ioproc);
286 amrex::ParallelDescriptor::Bcast(&l_shape_mismatch, 1, ioproc);
287 amrex::ParallelDescriptor::Bcast(l_file_shape.data(), l_file_shape.size(), ioproc);
288
289 const std::string refusal = regrid_refusal(regrid_options, target_lon, target_lat,
290 mf_var->ixType(), is2d);
291 // A lat-lon grid that happens to have the model grid's shape is still a lat-lon
292 // grid; its dimensions' own longitude and latitude coordinates give it away.
293 if (!l_shape_mismatch && refusal.empty() && !l_is_spatially_uniform &&
294 LatLonRegrid::has_latlon_axes(file_name, field_name)) {
295 l_shape_mismatch = 1;
296 }
297 if (l_shape_mismatch) {
298 const int field_rank = is2d ? 3 : 4;
299 const std::vector<MPI_Offset> file_shape(l_file_shape.begin(),
300 l_file_shape.begin() + field_rank);
301 if (!refusal.empty()) {
302 amrex::Abort(native_shape_mismatch_message(field_name, file_name, file_shape,
303 domain, mf_var->ixType(), is2d, refusal));
304 }
305 // Every file of a series on the same lat-lon grid, so one set of weights serves
306 const std::vector<long long> shape3(l_file_shape.begin(), l_file_shape.begin() + 3);
307 if (regrid == nullptr) {
308 // Should the file have been on the model grid instead, say what that needs
309 const std::vector<MPI_Offset> native = native_spatial_shape(domain, mf_var->ixType(), is2d);
310 const std::string context =
311 " It was taken to be on a lat-lon grid because its shape (eta, xi) = (" +
312 std::to_string(static_cast<long long>(file_shape[1])) + ", " +
313 std::to_string(static_cast<long long>(file_shape[2])) + ") is not the model grid's " +
314 shape_to_string(native) + "; a file on the model grid has remora.n_cell + 2 rho" +
315 " points along each axis, the ROMS boundary ring included.";
316 regrid = std::make_unique<LatLonRegrid>();
317 regrid->define(file_name, field_name, shape3, *target_lon, *target_lat,
318 *mf_var, domain, *regrid_options, context);
319 } else if (!regrid->same_source_grid(file_name, field_name, shape3)) {
320 amrex::Abort("The files of the '" + field_name + "' time series are on different"
321 " lat-lon grids (" + file_names[0] + " and " + file_name + "); a"
322 " series must keep one grid throughout.");
323 }
324 }
325 file_is_regridded.push_back(l_shape_mismatch);
326
327 file_is_cycle.push_back(is_cycle_int);
328 file_cycle_length.push_back(l_cycle_length);
329 file_is_spatially_uniform.push_back(l_is_spatially_uniform);
330
331 // get times and put in array
332 amrex::Vector<NDArray<double>> array_ts(1);
333 ReadNetCDFFile(file_name, {time_name}, array_ts); // filled only on proc 0
334 if (amrex::ParallelDescriptor::IOProcessor())
335 {
336 int ntimes_io = array_ts[0].get_vshape()[0];
337 for (int nt(0); nt < ntimes_io; nt++)
338 {
339 // Onto the model clock, then from days to seconds
340 ocean_times.push_back((*(array_ts[0].get_data() + nt) + offset_days) * 60.0 * 60.0 * 24.0);
341 file_for_time.push_back(ifile);
342 file_itime_offset.push_back(nt);
343 }
344 }
345 }
346
347 // A series on the model grid in one file and a lat-lon grid in another would switch
348 // between the two at the file boundary; spatially uniform files are fine with either.
349 if (regrid != nullptr) {
350 for (int ifile = 0; ifile < file_names.size(); ++ifile) {
351 if (!file_is_regridded[ifile] && !file_is_spatially_uniform[ifile]) {
352 amrex::Abort("The files of the '" + field_name + "' time series mix the model grid ("
353 + file_names[ifile] + ") with a lat-lon grid; a series must keep one"
354 " grid throughout.");
355 }
356 }
357 }
358
359 bool all_files_cycle_equal = std::all_of(file_is_cycle.begin(), file_is_cycle.end(),
360 [&](const auto& x) { return x == file_is_cycle.front(); });
361 bool all_files_cycle_length_equal = std::all_of(file_cycle_length.begin(), file_cycle_length.end(),
362 [&](const auto& x) { return x == file_cycle_length.front(); });
363
364 if (!all_files_cycle_equal || !all_files_cycle_length_equal) {
365 amrex::Abort("If one time series file in a set has a cycle, they all must, and cycle lengths must be equal");
366 }
367
368 // Store values to class members
369 is_cycle = file_is_cycle[0];
370 cycle_length = file_cycle_length[0];
371
372 // Arrays will be padded if time series file gives a cycle
373 int ntimes = (is_cycle) ? ocean_times.size() + 2 : ocean_times.size();
374 // Only do checks on IO processor since ocean_times isn't populated on other ranks yet
375 if (amrex::ParallelDescriptor::IOProcessor()) {
376 AMREX_ASSERT(std::is_sorted(ocean_times.begin(), ocean_times.end()));
377 if (ntimes <= 1) {
378 amrex::Error("Time series data must be given at at least two times");
379 }
380 }
381 int ioproc = amrex::ParallelDescriptor::IOProcessorNumber();
382 amrex::ParallelDescriptor::Bcast(&ntimes,1,ioproc);
383 if (!(amrex::ParallelDescriptor::IOProcessor())) {
384 ocean_times.resize(ntimes);
385 file_for_time.resize(ntimes);
386 file_itime_offset.resize(ntimes);
387 } else {
388 // If we're in a cycle, hack the lists to close the loop on the IOProc rank
389 if (is_cycle) {
390 ocean_times.insert(ocean_times.begin(),ocean_times[ocean_times.size()-1]-cycle_length);
391 // "first time" is now at index 1 because we already added to the front.
392 ocean_times.insert(ocean_times.end(), ocean_times[1]+cycle_length);
393 file_for_time.insert(file_for_time.begin(),file_for_time[file_for_time.size()-1]);
394 file_for_time.insert(file_for_time.end(), file_for_time[1]);
395 file_itime_offset.insert(file_itime_offset.begin(),file_itime_offset[file_itime_offset.size()-1]);
396 file_itime_offset.insert(file_itime_offset.end(), file_itime_offset[1]);
397 }
398 }
399 amrex::ParallelDescriptor::Bcast(ocean_times.data(), ocean_times.size(), ioproc);
400 amrex::ParallelDescriptor::Bcast(file_for_time.data(), file_for_time.size(), ioproc);
401 amrex::ParallelDescriptor::Bcast(file_itime_offset.data(), file_itime_offset.size(), ioproc);
402
403 // Initialize MultiFabs
404 // NetCDF data is always read and temporally interpolated on level 0.
405 mf_before = new amrex::MultiFab(mf_var->boxArray(), mf_var->DistributionMap(), 1, mf_var->nGrowVect());
406 mf_after = new amrex::MultiFab(mf_var->boxArray(), mf_var->DistributionMap(), 1, mf_var->nGrowVect());
407 mf_interp_lev0 = new amrex::MultiFab(mf_var->boxArray(), mf_var->DistributionMap(), 1, mf_var->nGrowVect());
408
409 // dummy initialization
410 i_time_before = -100;
411}

413/**
414 * @param time time to interpolate to
415 */
416void NCTimeSeries::update_interpolated_to_time (double time, int lev,
417 amrex::MultiFab* mf_lev,
418 const amrex::Vector<amrex::Geometry>& geom,
419 const amrex::Vector<amrex::IntVect>& ref_ratio) {
420
421 // Wrap into [ocean_times[0], ocean_times[0]+cycle_length], which the padded time
422 // array always covers. The phase is taken relative to the first stored time rather
423 // than to zero because a cycling file is free to carry an absolute time axis (days
424 // since some epoch) alongside its cycle length; fmod(time,cycle_length) would then
425 // land far below every time the file stores.
426 double l_time = time;
427 if (is_cycle) {
428 const double t_lo = ocean_times[0];
429 l_time = t_lo + std::fmod(time - t_lo, cycle_length);
430 if (l_time < t_lo) l_time += cycle_length;
431 }
432 // Figure out time index:
433 int i_time_before_old = i_time_before;
434 int i_time_new = -1;
435 for (int nt=0; nt < ocean_times.size()-1; nt++) {
436 if ((ocean_times[nt] <= l_time) and (ocean_times[nt+1] >= l_time)) {
437 i_time_new = nt;
438 break;
439 }
440 }
441 // Bracketing must succeed: falling through with the dummy index would read the
442 // time series at a negative offset. This is a runtime check, not an assert,
443 // because the failure is a mismatch between the file and the run.
444 if (i_time_new < 0) {
445 amrex::Abort("Time " + std::to_string(time) + " (mapped to " + std::to_string(l_time)
446 + ") is not spanned by the time series for '" + field_name + "', which covers ["
447 + std::to_string(ocean_times[0]) + ", "
448 + std::to_string(ocean_times[ocean_times.size()-1])
449 + "]. Extend the forcing file to cover the run, or give its time variable a"
450 + " cycle_length attribute so it repeats. The file's times were placed on the"
451 + " model clock from the reference date in their units attribute (the mapping"
452 + " is printed when the series is loaded), so also check remora.time_ref and"
453 + " remora.start_time.");
454 }
455 i_time_before = i_time_new;
456 time_before = ocean_times[i_time_before];
457 time_after = ocean_times[i_time_before+1];
458 int i_time_after = i_time_before + 1;
459
460 if (i_time_before_old + 1 == i_time_before) {
461 // swap multifabs so we only have to read in one MultiFab
462 std::swap(mf_before, mf_after);
463 read_in_at_time(mf_after, i_time_after);
464 } else if (i_time_before_old != i_time_before) {
465 read_in_at_time(mf_after, i_time_after);
466 read_in_at_time(mf_before, i_time_before);
467 }
468
469 // Both differences are small, so Real holds them even when the times do not.
470 const amrex::Real dt = static_cast<amrex::Real>(time_after - time_before);
471 const amrex::Real time_since_before = static_cast<amrex::Real>(l_time - time_before);
472
473 auto nodality = mf_interp_lev0->ixType();
474
475 // Everywhere the file holds data: the domain plus the ROMS ring outside a non-nodal edge.
476 // That includes every ghost cell inside the domain, which the FillPatch after this would
477 // refill with the same values from the neighbouring box, except where no box is there
478 // (remora.drop_dry_boxes): those cells would otherwise keep whatever mf_interp_lev0 held.
479 const amrex::Box data_box = amrex::grow(amrex::convert(domain, nodality),
480 amrex::IntVect(1-nodality[0],1-nodality[1],0));
481
482#ifdef AMREX_USE_OMP
483#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
484#endif
485 for (amrex::MFIter mfi(*mf_interp_lev0,true); mfi.isValid(); ++mfi) {
486 amrex::Box bx = mfi.growntilebox() & data_box;
487
488
489 // Temporal interpolation is done once on level 0.
490 amrex::MultiFab* mf_to_fill = mf_interp_lev0;
491 amrex::Array4<amrex::Real> to_fill = mf_to_fill->array(mfi);
492 amrex::Array4<const amrex::Real> before = mf_before->const_array(mfi);
493 amrex::Array4<const amrex::Real> after = mf_after->const_array(mfi);
494 amrex::ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
495 {
496 to_fill(i,j,k) = before(i,j,k) + time_since_before * (after(i,j,k) - before(i,j,k)) / dt;
497 });
498 }
499
500 amrex::MultiFab* mf_to_fill_lev = mf_lev;
501 if (save_interpolated) {
502 if (mf_interpolated_lev.size() <= static_cast<amrex::Long>(lev)) {
503 mf_interpolated_lev.resize(lev+1);
504 }
505 if (!mf_interpolated_lev[lev] ||
506 mf_interpolated_lev[lev]->boxArray() != mf_lev->boxArray() ||
507 mf_interpolated_lev[lev]->nGrowVect() != mf_lev->nGrowVect()) {
508 mf_interpolated_lev[lev] = std::make_unique<amrex::MultiFab>(
509 mf_lev->boxArray(), mf_lev->DistributionMap(), 1, mf_lev->nGrowVect());
510 }
511 mf_to_fill_lev = mf_interpolated_lev[lev].get();
512 }
513
514 last_updated_lev = lev;
515
516 if (lev == 0) {
517 amrex::MultiFab::Copy(*mf_to_fill_lev, *mf_interp_lev0, 0, 0, 1, mf_to_fill_lev->nGrowVect());
518 return;
519 }
520
521 amrex::PhysBCFunctNoOp null_bc_for_fill;
522 amrex::Vector<amrex::BCRec> null_dom_bcs(1);
523 for (int dir = 0; dir < AMREX_SPACEDIM; ++dir) {
524 null_dom_bcs[0].setLo(dir, amrex::BCType::int_dir);
525 null_dom_bcs[0].setHi(dir, amrex::BCType::int_dir);
526 }
527
528 amrex::Interpolater* mapper = nullptr;
529 const auto& idx_type = mf_to_fill_lev->ixType();
530 if (idx_type == amrex::IndexType(amrex::IntVect(0,0,0))) {
531 mapper = &amrex::cell_cons_interp;
532 } else {
533 mapper = &amrex::face_cons_linear_interp;
534 }
535
536 // The interpolation reads coarse cells one beyond the coarse cell under each fine cell, so
537 // near the physical boundary it reads level-0 ghost cells (#759). The time interpolation
538 // above set the domain, the ring the file holds outside a non-nodal edge, and the ghost
539 // cells inside the domain. Copy the outermost of those rows outward across each
540 // non-periodic side, then fill the ghost cells across a periodic edge from the cells they
541 // image. EnforcePeriodicity reads ghost cells too, so a cell that is both beyond a
542 // periodic edge and outside a wall gets its image's (already extended) value, and every
543 // ghost cell then agrees with its periodic image. Only the copy used here is changed;
544 // level 0 itself returned above.
545 {
546 const amrex::Box& dbx = data_box;
547 const bool per_x = geom[0].isPeriodic(0);
548 const bool per_y = geom[0].isPeriodic(1);
549#ifdef AMREX_USE_OMP
550#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
551#endif
552 for (amrex::MFIter mfi(*mf_interp_lev0,true); mfi.isValid(); ++mfi) {
553 const amrex::Box bx = mfi.growntilebox();
554 amrex::Array4<amrex::Real> a = mf_interp_lev0->array(mfi);
555 const auto lo = amrex::lbound(dbx);
556 const auto hi = amrex::ubound(dbx);
557 amrex::ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
558 {
559 const int ii = per_x ? i : amrex::max(lo.x, amrex::min(hi.x, i));
560 const int jj = per_y ? j : amrex::max(lo.y, amrex::min(hi.y, j));
561 // Only cells outside the data in a non-periodic direction are written, and
562 // their sources lie inside it in every non-periodic direction
563 if (ii != i || jj != j) { a(i,j,k) = a(ii,jj,k); }
564 });
565 }
566 mf_interp_lev0->EnforcePeriodicity(geom[0].periodicity());
567 }
568
569 // Take the coarse data from those ghost cells, without periodic wrap: each coarse cell
570 // then has a single source, its own value, as every ghost cell now holds the value the
571 // wrap would give. Fill the fine ghost cells outside the domain as far as the coarse
572 // ghost cells reach, horizontally only: nothing above fills level-0 ghost cells in z.
573 const amrex::IntVect ratio = cumulative_ref_ratio(lev, ref_ratio);
574 const amrex::IntVect cng = mf_interp_lev0->nGrowVect();
575 const amrex::IntVect fng = mf_to_fill_lev->nGrowVect();
576 const auto coarsener = mapper->BoxCoarsener(ratio);
577 amrex::IntVect ng_outside(0);
578 bool fits = true;
579 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
580 bool found = false;
581 for (int n = (d < 2) ? fng[d] : 0; n >= 0 && !found; --n) {
582 amrex::IntVect nv(0);
583 nv[d] = n;
584 const amrex::Box crse = coarsener.doit(amrex::Box(-nv, amrex::IntVect(32), idx_type));
585 if (-crse.smallEnd(d) <= cng[d]) {
586 ng_outside[d] = n;
587 found = true;
588 }
589 }
590 fits = fits && found;
591 }
592
593 if (fits) {
594 const amrex::Geometry& cg = geom[0];
595 const amrex::Geometry crse_geom_nonperiodic(cg.Domain(), cg.ProbDomain(), cg.Coord(),
596 amrex::Array<int,AMREX_SPACEDIM>{AMREX_D_DECL(0,0,0)});
597 amrex::InterpFromCoarseLevel(*mf_to_fill_lev, fng, ng_outside, *mf_interp_lev0,
598 0, 0, 1,
599 crse_geom_nonperiodic, geom[lev],
600 ratio, mapper, null_dom_bcs, 0);
601 } else {
602 // Too few level-0 ghost cells for the stencil at the boundary: as before
603 amrex::InterpFromCoarseLevel(*mf_to_fill_lev, zero, *mf_interp_lev0,
604 0, 0, 1,
605 geom[0], geom[lev],
606 null_bc_for_fill, 0, null_bc_for_fill, 0,
607 ratio, mapper, null_dom_bcs, 0);
608 }
609
610 mf_to_fill_lev->FillBoundary(geom[lev].periodicity());
611}

613amrex::IntVect
614NCTimeSeries::cumulative_ref_ratio (int lev,
615 const amrex::Vector<amrex::IntVect>& ref_ratio) const {
616 amrex::IntVect rr(1,1,1);
617 for (int l = 0; l < lev; ++l) {
618 rr[0] *= ref_ratio[l][0];
619 rr[1] *= ref_ratio[l][1];
620 rr[2] *= ref_ratio[l][2];
621 }
622 return rr;
623}

663 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(

666 uniform_value = *(array_dat[0].get_data());

678 }
679
680 // This all assumes that we're on level 0 with only one boxes_at_level
681 amrex::FArrayBox NC_fab;

683 amrex::Vector<std::string> NC_names;

687 NC_names.push_back(field_name);
688

691 } else {
692 NC_dim_types.push_back(NC_Data_Dims_Type::Time_BT_SN_WE);
693 }
694
695 BuildFABsFromNetCDFFile<amrex::FArrayBox,amrex::Real>(domain, file_name, NC_names, NC_dim_types,
696 NC_fabs, true, itime_offset);

698#ifdef _OPENMP
699#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
700#endif
701 {
702 // Don't tile this since we are operating on full FABs in this routine
703 for ( amrex::MFIter mfi(*mf, false); mfi.isValid(); ++mfi )

705 amrex::FArrayBox &fab = (*mf)[mfi];
706

709 // This only works here because we have broadcast the FArrayBox of data from the netcdf file to all ranks

711 fab.template copy<amrex::RunOn::Device>(NC_fab);
712 } // mf
713 } // omp
714}
