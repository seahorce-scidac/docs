12/**
13 * Air-sea O2 and CO2 exchange shared by the Fennel and ECB kernels, whose ROMS
14 * sources (fennel.h, ecb.h) carry identical copies of these formulas.
15 *
16 * Each coefficient set belongs to one published relation, so the Schmidt
17 * polynomial and the leading rate coefficient of the transfer velocity are
18 * selected together; ROMS makes the same pairings with RW14_OXYGEN_SC,
19 * OCMIP_OXYGEN_SC and RW14_CO2_SC. Pick a set on the host and capture it by
20 * value: the kernels never read these namespace-scope objects on the device.
21 */
22struct SchmidtCoefficients {
23 amrex::Real A;
24 amrex::Real B;
25 amrex::Real C;
26 amrex::Real D;
27 amrex::Real E;
28 amrex::Real rate; //!< leading coefficient of the transfer velocity [cm/hr]
29};

54/** \brief Schmidt number at temperature \p T [degC]. */
55AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
56amrex::Real schmidt_number (SchmidtCoefficients const& sc, amrex::Real T) noexcept
57{
58 return sc.A - T * (sc.B - T * (sc.C - T * (sc.D - T * sc.E)));
59}

61/** \brief Transfer-velocity rate factor for one step of \p dtdays days. */

63amrex::Real transfer_rate (SchmidtCoefficients const& sc, amrex::Real dtdays) noexcept

68/** \brief Squared 10 m wind speed [m2/s2] at rho point (i,j).
69 *
70 * From the winds under bulk fluxes, otherwise from the surface stress
71 * averaged to the rho point, as ROMS does without BULK_FLUXES.
72 */
73AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
74amrex::Real u10_squared (bool bulk_fluxes,
75 amrex::Array4<amrex::Real const> const& uwind,
76 amrex::Array4<amrex::Real const> const& vwind,
77 amrex::Array4<amrex::Real const> const& sustr,
78 amrex::Array4<amrex::Real const> const& svstr,
79 int i, int j, amrex::Real rho0) noexcept
80{
81 using amrex::Real;
82 if (bulk_fluxes) {
83 return uwind(i,j,0) * uwind(i,j,0) + vwind(i,j,0) * vwind(i,j,0);
84 }
85 const Real cff1 = rho0 * Real(550.0);
86 return cff1 *
87 std::sqrt((Real(0.5) * (sustr(i,j,0) + sustr(i+1,j,0))) *
88 (Real(0.5) * (sustr(i,j,0) + sustr(i+1,j,0))) +
89 (Real(0.5) * (svstr(i,j,0) + svstr(i,j+1,0))) *
90 (Real(0.5) * (svstr(i,j,0) + svstr(i,j+1,0))));
91}

101/** \brief O2 saturation [mmol/m3] from temperature [degC] and salinity, by the
102 * Garcia and Gordon (1992) formula converted from ml/l. */
103AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
104amrex::Real o2_saturation (amrex::Real T, amrex::Real S) noexcept
105{
106 using amrex::Real;
107 constexpr Real OA0 = Real(2.00907);
108 constexpr Real OA1 = Real(3.22014);
109 constexpr Real OA2 = Real(4.05010);
110 constexpr Real OA3 = Real(4.94457);
111 constexpr Real OA4 = Real(-0.256847);
112 constexpr Real OA5 = Real(3.88767);
113 constexpr Real OB0 = Real(-0.00624523);
114 constexpr Real OB1 = Real(-0.00737614);
115 constexpr Real OB2 = Real(-0.0103410);
116 constexpr Real OB3 = Real(-0.00817083);
117 constexpr Real OC0 = Real(-0.000000488682);
118 constexpr Real l2mol = Real(1000.0) / Real(22.3916); // liter to mol
119
120 const Real TS = std::log((Real(298.15) - T) / (Real(273.15) + T));
121 const Real AA = OA0 + TS * (OA1 + TS * (OA2 + TS * (OA3 + TS * (OA4 + TS * OA5)))) +
122 S * (OB0 + TS * (OB1 + TS * (OB2 + TS * OB3))) +
123 OC0 * S * S;
124 return l2mol * std::exp(AA);
125}

127/** \brief CO2 solubility [mol/(kg atm)] from temperature [degC] and salinity,
128 * by the Weiss (1974) formula. */
129AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
130amrex::Real co2_solubility (amrex::Real T, amrex::Real S) noexcept
131{
132 using amrex::Real;
133 constexpr Real A1 = Real(-60.2409);
134 constexpr Real A2 = Real(93.4517);
135 constexpr Real A3 = Real(23.3585);
136 constexpr Real B1 = Real(0.023517);
137 constexpr Real B2 = Real(-0.023656);
138 constexpr Real B3 = Real(0.0047036);
139
140 const Real TempK = Real(0.01) * (T + Real(273.15));
141 return std::exp(A1 +
142 A2 / TempK +
143 A3 * std::log(TempK) +
144 S * (B1 + TempK * (B2 + B3 * TempK)));
145}
