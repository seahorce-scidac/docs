11/**
12 * \brief Tiling for the loops of this file: boxes are split only in j (and k), so every
13 * tile spans the full i extent of its box.
14 *
15 * A compiler that contracts multiply-adds (GCC with -march=native, the Cray wrappers on
16 * Perlmutter) may round the same expression differently in the vectorized body of the
17 * i loop and in its remainder. With tiles that split i, which cells fall in the
18 * remainder depends on the tile size, and z_w, z_r and Hz then changed in their last
19 * bits with remora.omp_tile_size (8x8 tiles differed from 16-row strips and from no
20 * tiling; the differences grew to 1e-8 in the velocities within five steps). With
21 * full-row tiles every cell has the same position in the i loop whatever the tiling.
22 * Untiled (GPU) iteration is unchanged.
23 */
24inline MFItInfo RowTilingIfNotGPU ()
25{
26 MFItInfo info;
27 if (TilingIfNotGPU()) {
28 IntVect ts = FabArrayBase::mfiter_tile_size;
29 ts[0] = 1024000;
30 info.EnableTiling(ts);
31 }
32 return info;
33}

35/**
36 * \brief Depth at stretched coordinate sc (stretching value Cs) in a column of depth hwater
37 * under free surface zeta, ROMS Vtransform 2. Shared by stretch_transform and
38 * stretch_transform_full_domain.
39 */
40AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
41Real stretched_depth (Real sc, Real Cs, Real hc, Real hwater, Real zeta)
42{
43 Real cff = hc*sc;
44 Real hinv = one/(hc+hwater);
45 Real cff2 = (cff+Cs*hwater)*hinv;
46 return zeta+(zeta+hwater)*cff2;
47}

56 Real ds = one / Real(N);
57
58 Box bx(IntVect(0,0,0),IntVect(0,0,N));

60 const auto local_theta_b = solverChoice.theta_b;
61
62 //amrex::Vector<Real> s_w_vec(N+1); auto s_w_dat = s_w_vec.data();
63 //amrex::Vector<Real> Cs_w_vec(N+1); auto Cs_w_dat = Cs_w_vec.data();

67 auto Cs_w_dat = Cs_w.data();
68 auto s_r_dat = s_r.data();
69 auto Cs_r_dat = Cs_r.data();
70
71 amrex::ParallelFor(bx, [=] AMREX_GPU_DEVICE (int , int , int k)
72 {
73 Real Csur,Cbot;
74
75 s_w_dat[k]=ds*(k-N);
76 if (local_theta_s > zero) {
77 Csur=(one-std::cosh(local_theta_s*s_w_dat[k]))/
78 (std::cosh(local_theta_s)-one);
79 } else {
80 Csur=-s_w_dat[k]*s_w_dat[k];
81 }
82
83 if (local_theta_b > zero) {
84 Cbot=(std::exp(local_theta_b*Csur)-one)/
85 (one-std::exp(-local_theta_b));
86 Cs_w_dat[k]=Cbot;
87 } else {
88 Cs_w_dat[k]=Csur;
89 }
90
91 if (k<N) {
92 s_r_dat[k]=ds*(k-N+Real(0.5));
93
94 if (local_theta_s > zero) {
95 Csur=(one-std::cosh(local_theta_s*s_r_dat[k]))/
96 (std::cosh(local_theta_s)-one);
97 } else {
98 Csur=-s_r_dat[k]*s_r_dat[k];
99 }
100
101 if (local_theta_b > zero) {
102 Cbot=(std::exp(local_theta_b*Csur)-one)/
103 (one-std::exp(-local_theta_b));
104 Cs_r_dat[k]=Cbot;
105 } else {
106 Cs_r_dat[k]=Csur;
107 }
108 }

115void
116REMORA::stretch_transform (int lev)
117{
118 BL_PROFILE("REMORA::stretch_transform()");
119 std::unique_ptr<MultiFab>& mf_z_w = vec_z_w[lev];
120 std::unique_ptr<MultiFab>& mf_z_r = vec_z_r[lev];
121 std::unique_ptr<MultiFab>& mf_Hz = vec_Hz[lev];
122 std::unique_ptr<MultiFab>& mf_h = vec_h[lev];
123 std::unique_ptr<MultiFab>& mf_Zt_avg1 = vec_Zt_avg1[lev];
124 std::unique_ptr<MultiFab>& mf_z_phys_nd = vec_z_phys_nd[lev];
125
126 [[maybe_unused]] auto s_w_dat = s_w.data();
127 auto Cs_w_dat = Cs_w.data();
128 [[maybe_unused]] auto s_r_dat = s_r.data();
129 auto Cs_r_dat = Cs_r.data();
130
131#ifdef AMREX_USE_OMP
132#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
133#endif
134 for ( MFIter mfi(*cons_new[lev], RowTilingIfNotGPU()); mfi.isValid(); ++mfi )
135 {
136 Array4<Real> const& z_w = (mf_z_w)->array(mfi);
137 Array4<Real> const& z_r = (mf_z_r)->array(mfi);
138 Array4<Real> const& Hz = (mf_Hz)->array(mfi);
139 Array4<Real> const& h = (mf_h)->array(mfi);
140 Array4<Real> const& Zt_avg1 = (mf_Zt_avg1)->array(mfi);
141 // Grown only where the tile meets the box boundary, so no two tiles of one box
142 // write the same cell: z_w is first set to h and then overwritten below, and a
143 // neighbouring tile forming Hz in between would read the bathymetry.
144 Box gbx2 = mfi.growntilebox(IntVect(NGROW,NGROW,0));
145 Box gbx3 = mfi.growntilebox(IntVect(NGROW+1,NGROW+1,0));
146 Box gbx2D = gbx2;
147 gbx2D.makeSlab(2,0);
148 Box gbx3D = gbx3;
149 gbx3D.makeSlab(2,0);
150 Box wgbx3 = gbx3;
151 wgbx3.surroundingNodes(2);
152
153 const auto & geomdata = Geom(lev).data();
154
155 int nz = geom[lev].Domain().length(2);
156
157 auto N = nz; // Number of vertical "levels" aka, NZ
158 //forcing tcline to be the same as probhi for now, one in DataStruct.H other in inputs
159
160 Real hc = -min(geomdata.ProbHi(2),-solverChoice.tcline); // Do we need to enforce min here?
161
162 Real ds = one / Real(N);
163
164 amrex::ParallelFor(wgbx3, [=] AMREX_GPU_DEVICE (int i, int j, int k)
165 {
166 z_w(i,j,k) = h(i,j,0);
167 });
168
169 // ROMS Transform 2
170 Gpu::streamSynchronize();
171
172 amrex::ParallelFor(wgbx3, [=] AMREX_GPU_DEVICE (int i, int j, int k)
173 {
174 if (k < N) {
175 Real sc_r=ds*(k-N+Real(0.5));
176 Real sc_w=ds*(k-N);
177 Real hwater=h(i,j,0);
178
179 if (k==0) {
180 h(i,j,0,1) = stretched_depth(sc_w, Cs_w_dat[k], hc, hwater, Zt_avg1(i,j,0));
181 z_w(i,j,0) = h(i,j,0,1);
182 } else {
183 z_w(i,j,k)=stretched_depth(sc_w, Cs_w_dat[k], hc, hwater, Zt_avg1(i,j,0));
184 }
185
186 z_r(i,j,k)=stretched_depth(sc_r, Cs_r_dat[k], hc, hwater, Zt_avg1(i,j,0));
187
188 } else { // k == N
189 // HACK: should actually be the normal expression with coeffs evaluated at k=N-1
190 z_w(i,j,N)=Zt_avg1(i,j,0);
191 }
192 });
193
194 Gpu::streamSynchronize();
195
196 amrex::ParallelFor(gbx3, [=] AMREX_GPU_DEVICE (int i, int j, int k)
197 {
198 Hz(i,j,k)=z_w(i,j,k+1)-z_w(i,j,k);
199 });
200
201 } // mfi
202
203 vec_z_w[lev]->FillBoundary(geom[lev].periodicity());
204 vec_z_r[lev]->FillBoundary(geom[lev].periodicity());
205 vec_Hz[lev]->FillBoundary(geom[lev].periodicity());
206
207 // Define nodal z as average of z on w-faces
208#ifdef AMREX_USE_OMP
209#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
210#endif
211 for ( MFIter mfi(*cons_new[lev], RowTilingIfNotGPU()); mfi.isValid(); ++mfi )
212 {
213 Array4<Real> const& z_w = (mf_z_w)->array(mfi);
214 Array4<Real> const& z_phys_nd = (mf_z_phys_nd)->array(mfi);
215
216 Box z_w_box = Box(z_w);
217 auto const lo = amrex::lbound(z_w_box);
218 auto const hi = amrex::ubound(z_w_box);
219
220 //
221 // NOTE: we assume that all boxes extend the full extent of the domain in the vertical direction
222 //
223 // NOTE: z_phys_nd(i,j,k) and z_w(i,j,k) both refer to the node on the LOW side of cell (i,j,k)
224 //
225
226 // Nodal in all three directions so that the outermost node in x and y is
227 // filled as well. We do not grow in the vertical: z_w has no node there,
228 // and nothing reads z_phys_nd outside the column.
229 Box bx = mfi.grownnodaltilebox(-1,IntVect(NGROW,NGROW,0));
230
231 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
232 {
233 // Node (i,j,k) is the low corner of cell (i,j,k), so the four water
234 // columns surrounding it are cells (i-1,j-1) through (i,j). This is
235 // the psi-point averaging ROMS uses in set_depth / metrics.
236 // Indices are clamped so that a node on the edge of the array picks up the
237 // one-sided value; for now we assume all boundaries are constant
238 // height -- we will enforce periodicity below.
239 int ii = std::min(std::max(i , lo.x), hi.x);
240 int im = std::min(std::max(i-1, lo.x), hi.x);
241 int jj = std::min(std::max(j , lo.y), hi.y);
242 int jm = std::min(std::max(j-1, lo.y), hi.y);
243
244 z_phys_nd(i,j,k)=Real(0.25)*( z_w(im,jm,k) + z_w(ii,jm,k) +
245 z_w(im,jj,k) + z_w(ii,jj,k) );
246 });
247 } // mf
248
249 // Note that we do *not* want to do a multilevel fill here -- we have
250 // already filled z_phys_nd on the grown boxes, but we enforce periodicity just in case
251 vec_z_phys_nd[lev]->FillBoundary(geom[lev].periodicity());
252}
253/**
254 * z_r and z_w for a level not yet created, on full-domain arrays, over every grow cell of
255 * mf_z_w. mf_h and mf_zeta must cover the same region.
256 *
257 * @param[in ] lev level whose vertical grid to use
258 * @param[in ] mf_h bathymetry (component 0), 2D
259 * @param[in ] mf_zeta free surface (component 0), 2D
260 * @param[ out] mf_z_r depth of rho points
261 * @param[ out] mf_z_w depth of w points, on the same layout as mf_z_r but nodal in z
262 */
263void
264REMORA::stretch_transform_full_domain (int lev, const MultiFab& mf_h, const MultiFab& mf_zeta,
265 MultiFab& mf_z_r, MultiFab& mf_z_w)
266{
267 BL_PROFILE("REMORA::stretch_transform_full_domain()");
268 auto Cs_w_dat = Cs_w.data();
269 auto Cs_r_dat = Cs_r.data();
270
271 const auto& geomdata = Geom(lev).data();
272 const int N = geom[lev].Domain().length(2);
273 const Real hc = -min(geomdata.ProbHi(2),-solverChoice.tcline);
274 const Real ds = one / Real(N);
275
276#ifdef AMREX_USE_OMP
277#pragma omp parallel if (amrex::Gpu::notInLaunchRegion())
278#endif
279 for (MFIter mfi(mf_z_w, RowTilingIfNotGPU()); mfi.isValid(); ++mfi)
280 {
281 const Box& wbx = mfi.growntilebox();
282 Array4<const Real> const& h = mf_h.const_array(mfi);
283 Array4<const Real> const& zeta = mf_zeta.const_array(mfi);
284 Array4< Real> const& z_r = mf_z_r.array(mfi);
285 Array4< Real> const& z_w = mf_z_w.array(mfi);
286
287 ParallelFor(wbx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
288 {
289 if (k < N) {
290 z_w(i,j,k) = stretched_depth(ds*(k-N), Cs_w_dat[k], hc, h(i,j,0), zeta(i,j,0));
291 z_r(i,j,k) = stretched_depth(ds*(k-N+Real(0.5)), Cs_r_dat[k], hc, h(i,j,0),
292 zeta(i,j,0));
293 } else {
294 z_w(i,j,N) = zeta(i,j,0);
295 }
296 });
297 }
298 Gpu::streamSynchronize();
299}
