1/**
2 * \file REMORA_FennelBridge.cpp
3 *
4 * Path A of the Fennel parity campaign: pack REMORA state into ROMS-shaped
5 * buffers, call the unmodified ROMS kernel through the isohelper, and unpack
6 * the result. Selected at run time by remora.use_biology_cpp_answer = 0.
7 *
8 * The pack/unpack boundary is deliberately explicit rather than aliasing
9 * FArrayBox memory: REMORA FABs carry ghost cells and index k from 0 at the
10 * bottom, while ROMS wants a contiguous 1-based k over the valid region. An
11 * explicit copy keeps both paths auditable and immune to ghost-layout
12 * surprises, at a cost that is irrelevant for validation runs.
13 *
14 * See Source/Biology/Fortran/tag_map.md for the diagnostic contract.
15 */

32// Column-major (Fortran) linear index into a packed buffer covering
33// [ilo,ihi] x [jlo,jhi] with nk levels and nc components.
34AMREX_FORCE_INLINE
35int fidx (int i, int j, int k, int n,
36 int ilo, int jlo, int ni, int nj, int nk) noexcept
37{
38 return ((n * nk + k) * nj + (j - jlo)) * ni + (i - ilo);
39}

43void
44REMORABiology::FennelBiology::advance_fortran (const BiologyAdvanceContext& context) const
45{
46#ifdef AMREX_USE_GPU
47 amrex::ignore_unused(context);
48 amrex::Abort("Fennel Fortran bridge is host-only. Serial-first acceptance "
49 "is the default validation scope; GPU parity is a separate "
50 "lane and is not opened by this campaign.");
51#else
52
53 auto const& mf_cons_old = context.old_state;
54 auto& mf_cons_new = context.new_state;
55 auto const& solverChoice = context.solver_choice;
56 const int N = context.top_layer;
57 const int Bio_comp = context.component_start;
58 const Real dt_lev = context.dt;
59 const auto parms = m_parameters;
60
61 const bool need_gas = parms.oxygen || parms.carbon;
62 const bool do_bulk_flux = solverChoice.bulk_fluxes;
63
64 if (context.hz == nullptr || context.z_w == nullptr ||
65 context.z_r == nullptr || context.mask == nullptr ||
66 context.shortwave == nullptr ||
67 (need_gas && do_bulk_flux &&
68 (context.wind_u == nullptr || context.wind_v == nullptr)) ||
69 (need_gas && !do_bulk_flux &&
70 (context.stress_u == nullptr || context.stress_v == nullptr))) {
71 amrex::Abort("Fennel Fortran bridge requires Hz, z_r, z_w, rmask, "
72 "srflx, and, when carbon or oxygen is active, either "
73 "uwind/vwind (bulk_fluxes) or sustr/svstr");
74 }
75
76 // The ROMS buffer is temperature, salinity, then the biology tracers packed
77 // contiguously: buffer slot 0 is itemp, slot 1 is isalt, slot 2 is iNO3_, and so on.
78 // The ROMS biology slots follow fennel_mod.h:527-561. List the tracer names in that
79 // order here, independently of the order FennelBiology::configure declares them in,
80 // and look each slot's REMORA component up by name, which also skips any passive
81 // (dye) scalars between salinity and the biology block.
82 std::vector<std::string> roms_names = {"NO3", "NH4", "chlorophyll", "phytoplankton",
83 "zooplankton", "LdetritusN", "SdetritusN"};
84 if (parms.river_don) { roms_names.emplace_back("RdetritusN"); }
85 if (parms.po4) { roms_names.emplace_back("PO4"); }
86 if (parms.carbon) {
87 roms_names.insert(roms_names.end(), {"LdetritusC", "SdetritusC", "TIC", "alkalinity"});
88 }
89 if (parms.carbon && parms.river_don) { roms_names.emplace_back("RdetritusC"); }
90 if (parms.oxygen) { roms_names.emplace_back("oxygen"); }
91 if (parms.odu) { roms_names.emplace_back("ODU"); }
92 const int nbt = static_cast<int>(roms_names.size());
93 const int ntrc = nbt + 2;
94 AMREX_ALWAYS_ASSERT(nbt == static_cast<int>(m_declaration.tracer_names.size()));
95
96 std::vector<int> slot_comp(ntrc);
97 for (int slot = 0; slot < ntrc; ++slot) {
98 slot_comp[slot] = (slot < Tracer_comp)
99 ? slot : m_declaration.component_index(roms_names[slot - Tracer_comp], Bio_comp);
100 AMREX_ALWAYS_ASSERT(slot_comp[slot] >= 0);
101 }
102 auto cons_comp_for_slot = [&slot_comp] (int slot) noexcept {
103 return slot_comp[slot];
104 };
105
106 // ROMS N(ng) is a level count; REMORA N is the last valid k index.
107 const int nz = N + 1;
108
109 // REMORA stores shortwave radiation in W/m2. ROMS fennel.h expects
110 // kinematic srflx and converts it back with rho0*Cp when forming PARsur.
111 const Real srflx_to_roms_kinematic =
112 Real(1.0) / (solverChoice.rho0 * Cp);
113
114 for (MFIter mfi(mf_cons_new, false); mfi.isValid(); ++mfi) {
115 const Box bx = mfi.validbox();
116
117 const int ilo = bx.smallEnd(0);
118 const int ihi = bx.bigEnd(0);
119 const int jlo = bx.smallEnd(1);
120 const int jhi = bx.bigEnd(1);
121
122 const int ni = ihi - ilo + 1;
123 const int nj = jhi - jlo + 1;
124
125 // Stress buffers are one cell wider in each direction because
126 // fennel.h averages sustr(i+1,j) and svstr(i,j+1) to cell centres.
127 const int nis = ni + 1;
128 const int njs = nj + 1;
129
130 Array4<Real const> const& state_old = mf_cons_old.const_array(mfi);
131 Array4<Real const> const& state_new = mf_cons_new.const_array(mfi);
132 Array4<Real> const& state_out = mf_cons_new.array(mfi);
133 Array4<Real const> const& Hz_a = context.hz->const_array(mfi);
134 Array4<Real const> const& z_r_a = context.z_r->const_array(mfi);
135 Array4<Real const> const& z_w_a = context.z_w->const_array(mfi);
136 Array4<Real const> const& srflx_a = context.shortwave->const_array(mfi);
137 Array4<Real const> const& mskr_a = context.mask->const_array(mfi);
138
139 Array4<Real const> sustr_a;
140 Array4<Real const> svstr_a;
141 Array4<Real const> uwind_a;
142 Array4<Real const> vwind_a;
143 if (need_gas) {
144 if (do_bulk_flux) {
145 uwind_a = context.wind_u->const_array(mfi);
146 vwind_a = context.wind_v->const_array(mfi);
147 } else {
148 sustr_a = context.stress_u->const_array(mfi);
149 svstr_a = context.stress_v->const_array(mfi);
150 }
151 }
152
153 Vector<double> b_rmask(std::size_t(ni) * nj, 0.0);
154 Vector<double> b_srflx(std::size_t(ni) * nj, 0.0);
155 Vector<double> b_pH (std::size_t(ni) * nj, 0.0);
156 Vector<double> b_Hz (std::size_t(ni) * nj * nz, 0.0);
157 Vector<double> b_z_r (std::size_t(ni) * nj * nz, 0.0);
158 Vector<double> b_z_w (std::size_t(ni) * nj * (nz + 1), 0.0);
159 Vector<double> b_sustr(std::size_t(nis) * njs, 0.0);
160 Vector<double> b_svstr(std::size_t(nis) * njs, 0.0);
161 Vector<double> b_uwind(std::size_t(ni) * nj, 0.0);
162 Vector<double> b_vwind(std::size_t(ni) * nj, 0.0);
163 Vector<double> b_told (std::size_t(ni) * nj * nz * ntrc, 0.0);
164 Vector<double> b_tnew (std::size_t(ni) * nj * nz * ntrc, 0.0);
165
166 // ---- pack -------------------------------------------------------
167 for (int j = jlo; j <= jhi; ++j) {
168 for (int i = ilo; i <= ihi; ++i) {
169 const int s = fidx(i, j, 0, 0, ilo, jlo, ni, nj, 1);
170 b_rmask[s] = mskr_a(i, j, 0);
171 b_srflx[s] = srflx_a(i, j, 0) * srflx_to_roms_kinematic;
172 }
173 }
174
175 // ROMS k is 1-based over the valid column: ROMS k = REMORA k + 1.
176 // This is the only index-base translation in the bridge.
177 for (int k = 0; k < nz; ++k) {
178 for (int j = jlo; j <= jhi; ++j) {
179 for (int i = ilo; i <= ihi; ++i) {
180 const int s = fidx(i, j, k, 0, ilo, jlo, ni, nj, nz);
181 b_Hz [s] = Hz_a (i, j, k);
182 b_z_r[s] = z_r_a(i, j, k);
183 }
184 }
185 }
186
187 // z_w has no shift: ROMS z_w(0:N(ng)) and REMORA z_w k=0..N+1 both
188 // index faces from the bottom, and both hold nz+1 values.
189 for (int k = 0; k <= nz; ++k) {
190 for (int j = jlo; j <= jhi; ++j) {
191 for (int i = ilo; i <= ihi; ++i) {
192 b_z_w[fidx(i, j, k, 0, ilo, jlo, ni, nj, nz + 1)] =
193 z_w_a(i, j, k);
194 }
195 }
196 }
197
198 if (need_gas && !do_bulk_flux) {
199 for (int j = jlo; j <= jhi + 1; ++j) {
200 for (int i = ilo; i <= ihi + 1; ++i) {
201 const int s = fidx(i, j, 0, 0, ilo, jlo, nis, njs, 1);
202 b_sustr[s] = sustr_a(i, j, 0);
203 b_svstr[s] = svstr_a(i, j, 0);
204 }
205 }
206 }
207
208 if (need_gas && do_bulk_flux) {
209 for (int j = jlo; j <= jhi; ++j) {
210 for (int i = ilo; i <= ihi; ++i) {
211 const int s = fidx(i, j, 0, 0, ilo, jlo, ni, nj, 1);
212 b_uwind[s] = uwind_a(i, j, 0);
213 b_vwind[s] = vwind_a(i, j, 0);
214 }
215 }
216 }
217
218 // Buffer slot n is ROMS tracer n+1; cons_comp_for_slot skips any dye scalars.
219 for (int n = 0; n < ntrc; ++n) {
220 const int comp = cons_comp_for_slot(n);
221 for (int k = 0; k < nz; ++k) {
222 for (int j = jlo; j <= jhi; ++j) {
223 for (int i = ilo; i <= ihi; ++i) {
224 const int s =
225 fidx(i, j, k, n, ilo, jlo, ni, nj, nz);
226 b_told[s] = state_old(i, j, k, comp);
227 b_tnew[s] = state_new(i, j, k, comp);
228 }
229 }
230 }
231 }
232
233 // ---- call -------------------------------------------------------
234 fennel_bridge_advance_c(
235 ilo, ihi, jlo, jhi, nz, nbt,
236 ilo, ihi, jlo, jhi,
237 parms.po4 ? 1 : 0,
238 parms.carbon ? 1 : 0,
239 parms.oxygen ? 1 : 0,
240 parms.odu ? 1 : 0,
241 parms.denitrification ? 1 : 0,
242 parms.bio_sediment ? 1 : 0,
243 do_bulk_flux ? 1 : 0,
244 parms.river_don ? 1 : 0,
245 parms.talk_nonconserv ? 1 : 0,
246 // Enum ordinals rather than bools: these options are three- and
247 // two-way choices, and the Fortran side compares them against the
248 // CPP set it was compiled with.
249 static_cast<int>(parms.pco2air_type),
250 static_cast<int>(parms.co2_schmidt),
251 static_cast<int>(parms.o2_schmidt),
252 parms.BioIter,
253 double(dt_lev), double(solverChoice.rho0), double(Cp),
254 double(parms.AttSW), double(parms.AttChl), double(parms.PARfrac),
255 double(parms.Vp0), double(parms.I_thNH4), double(parms.D_p5NH4),
256 double(parms.NitriR), double(parms.K_NO3), double(parms.K_NH4),
257 double(parms.K_PO4), double(parms.K_Phy), double(parms.Chl2C_m),
258 double(parms.ChlMin), double(parms.PhyCN), double(parms.R_P2N),
259 double(parms.PhyIP), double(parms.PhyIS), double(parms.PhyMin),
260 double(parms.PhyMR), double(parms.ZooAE_N), double(parms.ZooCN),
261 double(parms.ZooBM), double(parms.ZooER), double(parms.ZooGR),
262 double(parms.ZooMin), double(parms.ZooMR), double(parms.LDeRRN),
263 double(parms.LDeRRC), double(parms.CoagR), double(parms.SDeRRN),
264 double(parms.SDeRRC), double(parms.RDeRRN), double(parms.RDeRRC),
265 double(parms.wPhy), double(parms.wLDet),
266 double(parms.wSDet), double(parms.pCO2air),
267 b_rmask.data(), b_Hz.data(), b_z_r.data(), b_z_w.data(),
268 b_srflx.data(), b_sustr.data(), b_svstr.data(),
269 b_uwind.data(), b_vwind.data(), b_pH.data(),
270 b_told.data(), b_tnew.data(),
271 m_debug, m_debug_i, m_debug_j);
272
273 // ---- unpack -----------------------------------------------------
274 // biology_tile updates only the biology tracers, so only those are
275 // written back; temperature and salinity in cons_new stay untouched.
276 for (int n = Tracer_comp; n < ntrc; ++n) {
277 const int comp = cons_comp_for_slot(n);
278 for (int k = 0; k < nz; ++k) {
279 for (int j = jlo; j <= jhi; ++j) {
280 for (int i = ilo; i <= ihi; ++i) {
281 state_out(i, j, k, comp) =
282 Real(b_tnew[fidx(i, j, k, n, ilo, jlo,
283 ni, nj, nz)]);
284 }
285 }
286 }
287 }
288 }
