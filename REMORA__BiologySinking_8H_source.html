13/**
14 * Vertical sinking of one particulate tracer in column (i,j), by the ROMS
15 * piecewise-parabolic reconstruction with WENO-reconciled interfaces and
16 * monotone boundary closures, followed by a CFL-free semi-Lagrangian flux.
17 * Shared by the Fennel and ECB kernels, whose ROMS sources carry identical
18 * copies.
19 *
20 * \p nlev is the number of layers, k = 0..nlev-1. On entry \p qc holds the
21 * concentration to sink (each model restricts it as its ROMS source does).
22 * On exit \p bio holds the sunk concentration and FC(i,j,0) the flux through
23 * the bottom face, which the caller returns at the seafloor. \p sink_distance
24 * is dtdays * |w|. \p qc, \p bR, \p bL, \p WR and \p WL are cell scratch, and
25 * \p FC face scratch, over at least the column.
26 */
27AMREX_GPU_DEVICE AMREX_FORCE_INLINE
28void sink_column (int i, int j, int nlev, amrex::Real sink_distance,
29 amrex::Array4<amrex::Real> const& bio,
30 amrex::Array4<amrex::Real> const& qc,
31 amrex::Array4<amrex::Real> const& bR,
32 amrex::Array4<amrex::Real> const& bL,
33 amrex::Array4<amrex::Real> const& WR,
34 amrex::Array4<amrex::Real> const& WL,
35 amrex::Array4<amrex::Real> const& FC,
36 amrex::Array4<amrex::Real const> const& Hz,
37 amrex::Array4<amrex::Real const> const& z_w,
38 amrex::Array4<amrex::Real> const& inv_hz,
39 amrex::Array4<amrex::Real> const& inv_hz2,
40 amrex::Array4<amrex::Real> const& inv_hz3) noexcept
41{
42 using amrex::Real;
43 constexpr Real zero = Real(0.0);
44 constexpr Real one = Real(1.0);
45 constexpr Real two = Real(2.0);
46 constexpr Real cff_weno = Real(1.0e-14);
47 // The boundary closures below touch layers 0, 1, N-1 and N.
48 AMREX_ASSERT(nlev >= 2);
49 const int N = nlev - 1; // top layer
50
51 // Reconstruct vertical profiles as parabolic segments within each grid box.
52 for (int iface = 1; iface <= N; ++iface) {
53 FC(i,j,iface) = (qc(i,j,iface) - qc(i,j,iface-1)) * inv_hz2(i,j,iface-1);
54 }
55 for (int k = 1; k < N; ++k) {
56 Real dltR = Hz(i,j,k) * FC(i,j,k+1);
57 Real dltL = Hz(i,j,k) * FC(i,j,k);
58 const Real cff = Hz(i,j,k-1) + two * Hz(i,j,k) + Hz(i,j,k+1);
59 const Real cffR = cff * FC(i,j,k+1);
60 const Real cffL = cff * FC(i,j,k);
61
62 // Apply PPM monotonicity constraint to prevent oscillations within
63 // the grid box.
64 if ((dltR * dltL) <= zero) {
65 dltR = zero;
66 dltL = zero;
67 } else if (std::abs(dltR) > std::abs(cffL)) {
68 dltR = cffL;
69 } else if (std::abs(dltL) > std::abs(cffR)) {
70 dltL = cffR;
71 }
72
73 // Compute right and left side values of parabolic segments; WR and WL
74 // are measures of quadratic variations.
75 const Real curv = (dltR - dltL) * inv_hz3(i,j,k);
76 dltR -= curv * Hz(i,j,k+1);
77 dltL += curv * Hz(i,j,k-1);
78 bR(i,j,k) = qc(i,j,k) + dltR;
79 bL(i,j,k) = qc(i,j,k) - dltL;
80 WR(i,j,k) = (two * dltR - dltL) * (two * dltR - dltL);
81 WL(i,j,k) = (dltR - two * dltL) * (dltR - two * dltL);
82 }
83 // Reconcile interfacial values using a WENO procedure so the whole profile
84 // remains monotonic.
85 for (int k = 1; k < N - 1; ++k) {
86 const Real dltL = amrex::max(cff_weno, WL(i,j,k));
87 const Real dltR = amrex::max(cff_weno, WR(i,j,k+1));
88 bR(i,j,k) = (dltR * bR(i,j,k) + dltL * bL(i,j,k+1)) / (dltR + dltL);
89 bL(i,j,k+1) = bR(i,j,k);
90 }
91 bR(i,j,N) = qc(i,j,N); // default strictly monotonic conditions
92 bL(i,j,N) = qc(i,j,N);
93 bR(i,j,N - 1) = qc(i,j,N);
94 bL(i,j,1) = qc(i,j,0); // bottom grid boxes are re-assumed to be
95 bR(i,j,0) = qc(i,j,0); // piecewise constant
96 bL(i,j,0) = qc(i,j,0);
97
98 // Apply monotonicity constraint again, since reconciled interfacial values
99 // may cause non-monotonic behavior inside the grid box.
100 for (int k = 0; k <= N; ++k) {
101 Real dltR = bR(i,j,k) - qc(i,j,k);
102 Real dltL = qc(i,j,k) - bL(i,j,k);
103 const Real cffR = two * dltR;
104 const Real cffL = two * dltL;
105 if ((dltR * dltL) < zero) {
106 dltR = zero;
107 dltL = zero;
108 } else if (std::abs(dltR) > std::abs(cffL)) {
109 dltR = cffL;
110 } else if (std::abs(dltL) > std::abs(cffR)) {
111 dltL = cffR;
112 }
113 bR(i,j,k) = qc(i,j,k) + dltR;
114 bL(i,j,k) = qc(i,j,k) - dltL;
115 }
116
117 // Compute vertical advective fluxes. The algorithm is free of a CFL
118 // criterion by allowing semi-Lagrangian integration bounds to use as many
119 // upstream grid boxes as necessary. WL is the z-coordinate of the
120 // departure point for grid box interface z_w; FC is the finite-volume flux.
121 for (int k = 0; k <= N; ++k) {
122 WL(i,j,k) = z_w(i,j,k) + sink_distance;
123 WR(i,j,k) = Hz(i,j,k) * qc(i,j,k);
124 }
125 FC(i,j,N + 1) = zero; // no-flux boundary condition
126 for (int k = 0; k <= N; ++k) {
127 int ksource = k;
128 Real flux = zero;
129 for (int ks = k; ks < N; ++ks) {
130 if (WL(i,j,k) > z_w(i,j,ks+1)) {
131 ksource = ks + 1;
132 flux += WR(i,j,ks);
133 }
134 }
135 // Finalize the flux by adding the fractional part from the source box.
136 const Real cu = amrex::min(one, (WL(i,j,k) - z_w(i,j,ksource)) * inv_hz(i,j,ksource));
137 flux += Hz(i,j,ksource) * cu *
138 (bL(i,j,ksource) + cu * (Real(0.5) * (bR(i,j,ksource) - bL(i,j,ksource)) -
139 (Real(1.5) - cu) * (bR(i,j,ksource) + bL(i,j,ksource) -
140 two * qc(i,j,ksource))));
141 FC(i,j,k) = flux;
142 }
143 for (int k = 0; k <= N; ++k) {
144 bio(i,j,k) = qc(i,j,k) + (FC(i,j,k+1) - FC(i,j,k)) * inv_hz(i,j,k);
145 }
146}
