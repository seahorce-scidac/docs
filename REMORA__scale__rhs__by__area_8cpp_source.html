5/**
6 * Convert the RHS momentum terms ru, rv, ru2d or rv2d between the form the kernels keep
7 * them in and the per-area form, which is that times cff = (pm+pm')*(pn+pn'), the
8 * inverse cell area of the face up to a factor of four. Coarse and fine faces differ in
9 * area, so data moved between levels is interpolated in the per-area form. The whole
10 * array is converted, ghost cells included.
11 *
12 * mf must share its level's boxes and distribution, so that the metrics of the same box
13 * index cover it, and carry NGROW ghost cells laterally.
14 *
15 * @param[in ] lev level whose pm and pn apply
16 * @param[inout] mf x-face (dir 0) or y-face (dir 1) data, 3D or 2D
17 * @param[in ] dir face direction
18 * @param[in ] to_per_area true to multiply by cff, false to divide by it
19 */
20void
21REMORA::scale_rhs_by_area (int lev, MultiFab& mf, int dir, bool to_per_area)
22{
23 const int ioff = (dir == 0) ? 1 : 0;
24 const int joff = (dir == 1) ? 1 : 0;
25 const int ncomp = mf.nComp();
26#ifdef _OPENMP

28#endif
29 for ( MFIter mfi(mf, TilingIfNotGPU()); mfi.isValid(); ++mfi )
30 {
31 Array4<Real const> const& pm = vec_pm[lev]->const_array(mfi);
32 Array4<Real const> const& pn = vec_pn[lev]->const_array(mfi);
33 Array4<Real > const& r = mf.array(mfi);
34
35 Box bx = mfi.growntilebox(IntVect(NGROW,NGROW,0));
36

39 {
40 Real cff = (pm(i,j,0)+pm(i-ioff,j-joff,0)) * (pn(i,j,0)+pn(i-ioff,j-joff,0));
41 r(i,j,k,n) = r(i,j,k,n) * cff;
42 });
43 } else {
