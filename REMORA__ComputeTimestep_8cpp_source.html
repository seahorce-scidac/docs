56void
57REMORA::ComputeDt ()
58{
59 Vector<Real> dt_tmp(finest_level+1);
60
61 for (int lev = 0; lev <= finest_level; ++lev)
62 {
63 dt_tmp[lev] = estTimeStep(lev);
64 }
65
66 ParallelDescriptor::ReduceRealMin(&dt_tmp[0], dt_tmp.size());
67
68 Real dt_0 = dt_tmp[0];
69 int n_factor = 1;
70 for (int lev = 0; lev <= finest_level; ++lev) {
71 dt_tmp[lev] = amrex::min(dt_tmp[lev], change_max*dt[lev]);
72 n_factor *= nsubsteps[lev];
73 dt_0 = amrex::min(dt_0, n_factor*dt_tmp[lev]);
74 }
75
76 // dt_0 is identical on every rank here, so this aborts collectively. The
77 // bogus_large_value test is the load-bearing one: the sentinel used to survive to
78 // this point intact whenever stop_time was left at its default of Real::max(), and to
79 // be laundered into stop_time - t_new[0] -- a single step covering the entire run --
80 // whenever stop_time was set. The change_max cap above cannot catch either case,
81 // since dt[] is itself seeded to bogus_large_value.
82 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(dt_0 > zero && std::isfinite(dt_0) &&
83 dt_0 < bogus_large_value,
84 "REMORA::ComputeDt: computed a non-positive, "
85 "non-finite, or unusably large level-0 dt");
86
87 // Limit dt's by the value of stop_time.
88 const Real eps = Real(1.e-3)*dt_0;
89 if (t_new[0] + dt_0 > stop_time - eps) {
90 dt_0 = stop_time - t_new[0];
91 }
92
93 dt[0] = dt_0;
94 for (int lev = 1; lev <= finest_level; ++lev) {
95 dt[lev] = dt[lev-1] / nsubsteps[lev];
96 }
97}

99/**
100 * Estimate the largest stable slow (baroclinic) timestep on a level.
101 *
102 * The estimate is the smaller of an advective limit and an external gravity wave limit,
103 * each a maximum over the wet cells of the level:
104 *
105 * dt_adv = cfl / max( |u|*pm, |v|*pn, |w|/Hz )
106 * dt_grav = cfl * ndtfast / max( sqrt(g*|h|) * sqrt(pm^2 + pn^2) )
107 *
108 * The second is the Courant quantity ROMS forms as Cg_max in metrics.F. It is what keeps
109 * the estimate finite for a run started from rest, where every velocity is zero and the
110 * advective limit alone is undefined.
111 *
112 * pm and pn are the per-cell 1/dx and 1/dy metric terms rather than the geometry's
113 * uniform cell size, so stretched and curvilinear grids give the right answer.
114 *
115 * @param[in] level level of refinement
116 */
117Real
118REMORA::estTimeStep(int level) const
119{
120 BL_PROFILE("REMORA::estTimeStep()");
121
122 // The barotropic mode is substepped ndtfast times per baroclinic step
123 // (REMORA_Advance.cpp), so the slow step only has to resolve the external gravity
124 // wave to within that ratio. ReadParameters guarantees ndtfast is positive.
125
126 // g is a file-scope constexpr; hoist it into a local for the device lambda.
127 const Real grav = solverChoice.g;
128
129 MultiFab ccvel(grids[level],dmap[level],3,0);
130
131 average_face_to_cellcenter(ccvel,0,
132 Array<const MultiFab*,3>{xvel_new[level], yvel_new[level], zvel_new[level]});
133
134 // Two maxima over the same cells. amrex::ReduceMax's FabArray overloads take at most
135 // three FabArrays and this needs six, so drive ReduceOps directly. Unlike
136 // amrex::ReduceMax, ReduceOps::eval has no host fallback in a GPU build, so no
137 // Gpu::LaunchSafeGuard is wanted here.
138 ReduceOps<ReduceOpMax, ReduceOpMax> reduce_op;
139 ReduceData<Real, Real> reduce_data(reduce_op);
140 using ReduceTuple = typename decltype(reduce_data)::Type;
141
142#ifdef _OPENMP
143#pragma omp parallel if (Gpu::notInLaunchRegion())
144#endif
145 for ( MFIter mfi(ccvel, TilingIfNotGPU()); mfi.isValid(); ++mfi )
146 {
147 // ccvel carries no ghost cells, so this is exactly its valid region -- the same
148 // cells the advective estimate has always visited.
149 const Box& bx = mfi.tilebox();
150
151 Array4<Real const> const& u = ccvel.const_array(mfi);
152 Array4<Real const> const& Hz = vec_Hz[level]->const_array(mfi);
153
154 // h, pm, pn and mskr live on the z-slab BoxArray built box-for-box from
155 // grids[level] with the same DistributionMapping, so they index off this same
156 // MFIter and are read at k = 0.
157 Array4<Real const> const& h = vec_h[level]->const_array(mfi);
158 Array4<Real const> const& pm = vec_pm[level]->const_array(mfi);
159 Array4<Real const> const& pn = vec_pn[level]->const_array(mfi);
160 Array4<Real const> const& mskr = vec_mskr[level]->const_array(mfi);
161
162 reduce_op.eval(bx, reduce_data,
163 [=] AMREX_GPU_DEVICE (int i, int j, int k) -> ReduceTuple
164 {
165 // Land contributes nothing: h there is a fill depth, not a real one.
166 if (mskr(i,j,0) <= Real(0.5)) { return {zero, zero}; }
167
168 Real inv_adv = amrex::max(amrex::Math::abs(u(i,j,k,0)) * pm(i,j,0),
169 amrex::Math::abs(u(i,j,k,1)) * pn(i,j,0));
170
171 // Hz is the true thickness of this terrain-following layer; the geometry's
172 // uniform dz is the thickness of a sigma level, not of a cell.
173 if (Hz(i,j,k) > zero) {
174 inv_adv = amrex::max(inv_adv, amrex::Math::abs(u(i,j,k,2)) / Hz(i,j,k));
175 }
176
177 // Component 0 of vec_h is the bathymetry, positive down. Component 1 is not a
178 // second copy of it -- stretch_transform overwrites it with the bottom z_w.
179 const Real c = std::sqrt(grav * amrex::Math::abs(h(i,j,0,0)));
180 const Real inv_bt = c * std::sqrt(pm(i,j,0)*pm(i,j,0) + pn(i,j,0)*pn(i,j,0));
181
182 return {inv_adv, inv_bt};
183 });
184 } // mfi
185
186 ReduceTuple host_tuple = reduce_data.value(reduce_op);
187
188 // A rank owning no boxes never calls eval and leaves its local maxima at
189 // std::numeric_limits<Real>::lowest(); the MPI max repairs that, and every test below
190 // is against the reduced values, which are identical on every rank.
191 Real inv[2] = { amrex::get<0>(host_tuple), amrex::get<1>(host_tuple) };
192 ParallelDescriptor::ReduceRealMax(inv, 2);
193 const Real inv_adv = inv[0];
194 const Real inv_bt = inv[1];
195
196 // Guard both divisions rather than letting them produce inf: inv_adv is zero for any
197 // quiescent start, and inv_bt is zero for a level that is entirely land.
198 const Real estdt_adv = (inv_adv > zero) ? cfl / inv_adv : bogus_large_value;
199 const Real estdt_bt = (inv_bt > zero) ? cfl * Real(ndtfast) / inv_bt : bogus_large_value;
200
201 const Real estdt_lowM = amrex::min(estdt_adv, estdt_bt);
202
203 if (verbose) {
204 amrex::Print() << "Using cfl = " << cfl << std::endl;
205 if (inv_adv > zero) {
206 amrex::Print() << " advective limit at level " << level << ": " << estdt_adv << std::endl;
207 } else {
208 amrex::Print() << " advective limit at level " << level << ": none (velocity is zero)" << std::endl;
209 }
210 if (inv_bt > zero) {
211 amrex::Print() << " gravity wave limit at level " << level << ": " << estdt_bt
212 << " (ndtfast = " << ndtfast << ")" << std::endl;
213 } else {
214 amrex::Print() << " gravity wave limit at level " << level << ": none (no wet cell has a depth)" << std::endl;
215 }
216 if (fixed_dt > zero) {
217 if (estdt_lowM < bogus_large_value) {
218 amrex::Print() << "Slow dt at level " << level << " would be: " << estdt_lowM << std::endl;
219 } else {
220 amrex::Print() << "Slow dt at level " << level << " would be undefined " << std::endl;
221 }
222 amrex::Print() << "Fixed dt at level " << level << " is: " << fixed_dt << std::endl;
223 } else {
224 amrex::Print() << "Slow dt at level " << level << ": " << estdt_lowM << std::endl;
225 }
226 }
227
228 if (fixed_dt > zero) {
229 return fixed_dt;
230 }
231
232 // Water at rest still carries gravity waves, so estdt_bt is finite for any level with
233 // one wet cell of nonzero depth. Reaching here means there is no such cell; returning
234 // the sentinel would let ComputeDt clamp dt to stop_time - t_new[0] and run the whole
235 // simulation in a single step.
236 if (estdt_lowM >= bogus_large_value) {
237 amrex::Abort("REMORA::estTimeStep: cannot estimate a timestep at level " +
238 std::to_string(level) + ". No wet cell (mskr > 0.5) has a nonzero "
239 "depth, so neither the advective nor the gravity wave limit is "
240 "defined. Check the bathymetry and the land mask, or set "
241 "remora.fixed_dt");
242 }
243
244 return estdt_lowM;
245}
