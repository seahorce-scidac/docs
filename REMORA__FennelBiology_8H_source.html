21/**
22 * Schmidt-number formulation for the air-sea gas transfer velocity.
23 *
24 * CO2 offers the Wanninkhof (1992) and (2014) coefficient sets; O2 adds the
25 * Keeling et al. (1998) OCMIP set. The choice also selects the leading rate
26 * coefficient, 0.31 for 1992 and 0.251 for 2014, so the two must move
27 * together and are not independently settable.
28 */
29enum class CO2SchmidtType {
30 wanninkhof1992,
31 wanninkhof2014
32};

34enum class O2SchmidtType {
35 wanninkhof1992,
