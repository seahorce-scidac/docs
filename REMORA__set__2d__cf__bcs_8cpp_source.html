11/** \brief Replace each group of nr faces along the interface by their mean.
12 *
13 * ROMS gives every fine face under one parent face the same parent flux: get_persisted2d
14 * copies DU_avg2 at the donor face picked by integer division of the fine index, and
15 * put_refine2d and u2dbc_im scale it by the edge-length ratio. AMReX's face interpolator gives
16 * the flux a linear variation along the interface instead. Both make the fine fluxes sum to
17 * the parent's; they differ in how the total is shared out. That interpolation being
18 * conservative, the group mean is the parent value, so averaging turns the linear profile back
19 * into ROMS's constant one.
20 *
21 * Only groups lying wholly inside a box are touched, and only faces the set mask covers, so
22 * faces the parent never wrote keep their zero.
23 */
24void group_average_faces (MultiFab& mf, const iMultiFab& mask, int set_mask, int tdir, int nr)
25{
26 if (nr <= 1) { return; }
27#ifdef AMREX_USE_OMP
28#pragma omp parallel if (Gpu::notInLaunchRegion())
29#endif
30 for (MFIter mfi(mf); mfi.isValid(); ++mfi)
31 {
32 const Box& bx = mfi.validbox();
33 const auto& a = mf.array(mfi);
34 const auto& m = mask.const_array(mfi);
35 const int lo_t = bx.smallEnd(tdir);
36 const int hi_t = bx.bigEnd(tdir);
37 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
38 {
39 const int t = (tdir == 0) ? i : j;
40 const int t0 = amrex::coarsen(t, nr) * nr;
41 // one thread per group, and only where the whole group is in this box
42 if (t != t0 || t0 < lo_t || t0 + nr - 1 > hi_t) { return; }
43 Real sum = Real(0.0);
44 int cnt = 0;
45 for (int s = t0; s < t0 + nr; ++s) {
46 const int si = (tdir == 0) ? s : i;
47 const int sj = (tdir == 0) ? j : s;
48 if (m(si,sj,k) != set_mask) { continue; }
49 sum += a(si,sj,k); ++cnt;
50 }
51 if (cnt == 0) { return; }
52 const Real avg = sum / Real(cnt);
53 for (int s = t0; s < t0 + nr; ++s) {
54 const int si = (tdir == 0) ? s : i;
55 const int sj = (tdir == 0) ? j : s;
56 if (m(si,sj,k) != set_mask) { continue; }
57 a(si,sj,k) = avg;
58 }
59 });
60 }
61}

131/**
132 * The parent's barotropic flux per unit edge length (Dubar, Dvbar), interpolated in time onto
133 * this level's interface faces and, with cf_flux_pc, made piecewise constant along them; zero
134 * on every other face.
135 *
136 * It depends only on the data the parent registered, which register_coarse_data replaces
137 * (bumping cf_crse_gen), the time and the layout. The fast steps of one fine step all ask
138 * for it at that step's t_old, so the result is kept per level and made again only when the
139 * generation, the time or the layout changes. remora.cf_2d_fill_cache = 0 makes it on every

144 * @return the u- and v-face fluxes, valid until the next call for this level

209/**
210 * Set the normal barotropic velocity on this level's coarse-fine interface from the parent's
211 * time-averaged mass flux, as ROMS does in put_refine2d (nesting.F):
212 *
213 * ubar_f = Dubar_c / D_f, D_f = 0.5*(h + zeta)_{i-1} + 0.5*(h + zeta)_i
214 *
215 * D comes from know; ROMS instead uses one time index for both sides: u2dbc_im.F's nested branch
216 * builds D from zeta(kout) and writes ubar(kout), and put_refine2d uses indx1 for both. Since
217 * the next half-step forms DUon = ubar(krhs)*D(krhs) with krhs equal to this knew, ROMS
218 * recovers Dubar_parent exactly while this carries an extra D(knew)/D(know). Measured at most
219 * 1.2e-07 on the step-1 Channel_Test interface velocity, against a 2.4e-03 artifact, with no
220 * change to Dogbone volume drift or the Channel_Test blow-up.
221 *
222 * Momentum only. setup_step resets all three zeta components to Zt_avg1, so what a finer
223 * level interpolates for the free surface is already the parent's fast-time average -- as in
224 * ROMS, where set_zeta runs ahead of put_refine2d.
225 *
226 * @param[in] lev level of refinement
227 * @param[in] time simulation time to interpolate the parent's flux to
228 * @param[in] know zeta time component to take the sea surface height from
