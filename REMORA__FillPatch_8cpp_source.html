28/** \brief number of cells where mask is set and a(comp) and b(comp) differ in value
29 * (two NaNs count as equal), over all ranks */
30Long count_mask_diff (const MultiFab& a, const MultiFab& b, const iMultiFab& mask, int comp)
31{
32 ReduceOps<ReduceOpSum> reduce_op;
33 ReduceData<Long> reduce_data(reduce_op);
34 using ReduceTuple = typename decltype(reduce_data)::Type;
35 for (MFIter mfi(a); mfi.isValid(); ++mfi) {
36 auto x = a.const_array(mfi, comp);
37 auto y = b.const_array(mfi, comp);
38 auto m = mask.const_array(mfi);
39 reduce_op.eval(mfi.fabbox(), reduce_data,
40 [=] AMREX_GPU_DEVICE (int i, int j, int k) -> ReduceTuple
41 {
42 // Outside the mask the cells may never have been written (a signalling NaN in a
43 // debug build), so do not even compare them
44 if (!m(i,j,k)) { return { Long(0) }; }
45 const Real xv = x(i,j,k);
46 const Real yv = y(i,j,k);
47 const bool same = (xv == yv) || (xv != xv && yv != yv);
48 return { same ? Long(0) : Long(1) };
49 });
50 }
51 Long n = amrex::get<0>(reduce_data.value(reduce_op));
52 ParallelDescriptor::ReduceLongSum(n);
53 return n;
54}

56/**
57 * ba with each box extended outward by ng across the non-periodic domain faces it touches.
58 * The result is kept and handed out again for the same ba, so a MultiFab built on it has the
59 * same BoxArray from call to call and FillBoundary reuses its communication pattern rather
60 * than rebuilding it on every fill. The few layouts a run uses at a time are kept; the cache
61 * is emptied at finalize, before AMReX is torn down.
62 */
63BoxArray
64extended_boxarray (const BoxArray& ba, const Geometry& geom, const IntVect& ng)
65{
66 struct Entry { BoxArray ba; Box dom; IntVect ng; BoxArray ext; };
67 static Vector<Entry> cache;
68 static bool registered = false;
69 if (!registered) {
70 amrex::ExecOnFinalize([] () { cache.clear(); });
71 registered = true;
72 }
73
74 const Box dom = amrex::convert(geom.Domain(), ba.ixType());
75 for (const auto& e : cache) {
76 if (e.ng == ng && e.dom == dom && e.ba == ba) { return e.ext; }
77 }
78
79 BoxList bl(ba.ixType());
80 for (int i = 0; i < ba.size(); ++i) {
81 Box b = ba[i];
82 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
83 if (geom.isPeriodic(d)) { continue; }
84 if (b.smallEnd(d) <= dom.smallEnd(d)) { b.growLo(d, ng[d]); }
85 if (b.bigEnd(d) >= dom.bigEnd(d)) { b.growHi(d, ng[d]); }
86 }
87 bl.push_back(b);
88 }
89 BoxArray ext(std::move(bl));
90
91 constexpr int max_entries = 16;
92 if (cache.size() >= max_entries) { cache.erase(cache.begin()); }
93 cache.push_back({ba, dom, ng, ext});
94 return ext;
95}

97/**
98 * A copy of mf whose ghost cells beyond a non-periodic domain face agree with their periodic
99 * images, for a copy that reads ghost cells with periodicity.
100 *
101 * FillBoundary copies only from valid cells, so a ghost cell that lies outside the domain in
102 * a non-periodic direction and beyond a periodic edge -- a corner where a periodic seam meets
103 * a wall, or the vertical ghosts at a seam -- keeps whatever the boundary conditions put there,
104 * which need not match the cell it images. Face data normal to a periodic direction has the
105 * same problem one step earlier: below a wall, the faces at the two ends of the periodic width
106 * are both owned by a box and need not agree either. InterpFromCoarseLevel copies coarse ghost
107 * cells with periodicity, so such a cell then has two sources, the box's own ghost and its
108 * periodic image. ParallelCopy does not say which one wins. On a CPU the order is fixed, but on
109 * a GPU whichever thread gets there first wins, so the result varies from run to run (#768).
110 *
111 * Here each box is extended outward across its non-periodic domain faces, so those ghosts
112 * become valid cells of a temporary. OverrideSync makes the shared periodic faces agree, and
113 * FillBoundary then fills the images. The result goes into scratch rather than back into mf,
114 * so the coarse level's own ghost cells are left as they were.
115 *
116 * Doing this with AMReX calls alone gets most of the way. EnforcePeriodicity takes its
117 * sources from each box grown by its ghost cells -- Periodicity::Domain() is unbounded in the
118 * non-periodic directions -- so on a copy of mf, FillBoundary followed by EnforcePeriodicity
119 * already makes cell-centered ghosts beyond a wall match their images. On Channel_Test_ML_seam
120 * that gives a plotfile bitwise identical to this function's. It does not cover face data
121 * normal to a periodic direction: faces 0 and nx both lie inside the nodal periodic domain, so
122 * EnforcePeriodicity never writes them, and OverrideSync reconciles only their valid copies.
123 * Below a wall the two then still differ (up to 3e-15 for xvel in that test), leaving two
124 * sources. The extended temporary makes those faces valid so OverrideSync reaches them.
125 *
126 * @param[in ] mf data to read, its ghost cells already FillBoundary'd
127 * @param[in ] geom geometry of mf's level
128 * @param[inout] scratch storage for the copy, used only when one is needed
129 * @return mf itself when no ghost cell can have two sources, otherwise scratch
130 */
131const MultiFab&
132periodic_consistent_ghosts (const MultiFab& mf, const Geometry& geom, MultiFab& scratch)
133{
134 const IntVect ng = mf.nGrowVect();
135 bool any_periodic = false;
136 bool any_outside = false;
137 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
138 if (geom.isPeriodic(d)) {
139 any_periodic = any_periodic || ng[d] > 0;
140 } else {
141 any_outside = any_outside || ng[d] > 0;
142 }
143 }
144 if (!any_periodic || !any_outside) { return mf; }
145
146 const BoxArray& ba = mf.boxArray();
147 IntVect per_ng(0);
148 bool nodal_periodic = false;
149 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
150 if (geom.isPeriodic(d)) {
151 per_ng[d] = ng[d];
152 nodal_periodic = nodal_periodic || mf.ixType().nodeCentered(d);
153 }
154 }
155
156 // The temporary's ghosts start from mf's too: FillBoundary leaves alone a ghost with no
157 // valid cell to copy from -- one at a coarse-fine interface of a level that does not span
158 // the domain -- and that must come out as it went in, not as uninitialized memory.
159 MultiFab ext(extended_boxarray(ba, geom, ng), mf.DistributionMap(), mf.nComp(), per_ng);
160 const int ncomp = mf.nComp();
161 for (MFIter mfi(ext); mfi.isValid(); ++mfi) {
162 const auto& e = ext.array(mfi);
163 const auto& a = mf.const_array(mfi);
164 ParallelFor(mfi.fabbox(), ncomp, [=] AMREX_GPU_DEVICE (int i, int j, int k, int n)
165 {
166 e(i,j,k,n) = a(i,j,k,n);
167 });
168 }
169 if (nodal_periodic) { ext.OverrideSync(geom.periodicity()); }
170 ext.FillBoundary(geom.periodicity());
171
172 // Ghosts the temporary does not cover -- those of a box away from the domain faces --
173 // come from mf unchanged.
174 scratch.define(ba, mf.DistributionMap(), ncomp, ng);
175 MultiFab::Copy(scratch, mf, 0, 0, ncomp, ng);
176 for (MFIter mfi(ext); mfi.isValid(); ++mfi) {
177 const auto& e = ext.const_array(mfi);
178 const auto& a = scratch.array(mfi);
179 ParallelFor(mfi.fabbox(), ncomp, [=] AMREX_GPU_DEVICE (int i, int j, int k, int n)
180 {
181 a(i,j,k,n) = e(i,j,k,n);
182 });
183 }
184 return scratch;
185}

188/**
189 * Coarse data ready for a coarse-to-fine interpolation (FillPatchTwoLevels,
190 * InterpFromCoarseLevel), which copies the coarse cells, ghost cells included, with
191 * periodicity. The ghost cells are exchanged across grids and periodic edges, and those that
192 * also lie beyond a non-periodic domain face are made to agree with their periodic images
193 * (periodic_consistent_ghosts), so that no cell of the copy has two different sources (#768,
194 * #774, #776). Every coarse-to-fine fill goes through here.
195 *
196 * The ghost cells beyond a wall are otherwise taken as the coarse level's own fill left them;
197 * they are not refilled from the boundary conditions here.

205REMORA::coarse_for_interp (int crse_lev, MultiFab& crse, MultiFab& scratch)

207 crse.FillBoundary(geom[crse_lev].periodicity());
208 const MultiFab& src = periodic_consistent_ghosts(crse, geom[crse_lev], scratch);
209 return (&src == &crse) ? crse : scratch;

807 * @param[inout] mf_crse MultiFab of coarse data
808 * @param[in ] bccomp index into domain_bcs_type
809 * @param[in ] bdy_var_type when filling from NetCDF, which boundary data
810 * @param[in ] icomp component to fill
811 * @param[in ] fill_all whether to fill all components
812 * @param[in ] icomp_calc component to use in RHS boundary calculation
813 * @param[in ] dt_lev time step at this level
814 * @param[in ] mf_calc data to use in RHS boundary calculation
815 */
816void
817REMORA::FillCoarsePatchMap (int lev, Real time, MultiFab* mf_to_fill, MultiFab* mf_crse,
818 const int bccomp,
819#ifdef REMORA_USE_NETCDF
820 const int bdy_var_type,
821#else
822 const int /*bdy_var_type*/,
823#endif
824 const int icomp,
825 const bool fill_all,
826 const int n_not_fill,
827 const int icomp_calc,
828 const Real dt_lev,
829 const MultiFab& mf_calc,
830 amrex::Interpolater* mapper)
831{
832 BL_PROFILE_VAR("FillCoarsePatch()",FillCoarsePatch);
833 AMREX_ASSERT(lev > 0);
834
835 MultiFab* mask = nullptr;
836
837 int ncomp;
838 if (fill_all) {
839 ncomp = mf_to_fill->nComp();
840 } else {
841 ncomp = 1;
842 }
843
844 Box box_mf(mf_to_fill->boxArray()[0]);
845
846 if (box_mf.ixType() == IndexType(IntVect(0,0,0)))
847 {
848 mask = vec_mskr[lev].get();
849 }
850 else if (box_mf.ixType() == IndexType(IntVect(1,0,0)))
851 {
852 mask = vec_msku[lev].get();
853 }
854 else if (box_mf.ixType() == IndexType(IntVect(0,1,0)))
855 {
856 mask = vec_mskv[lev].get();
857 }
858 else if (box_mf.ixType() == IndexType(IntVect(0,0,1)))
859 {
860 mask = vec_mskr[lev].get();
861 } else {
862 amrex::Abort("Dont recognize this box type in REMORA_FillPatch");
863 }
864
865 if (mapper == nullptr) {
866 if (box_mf.ixType() == IndexType(IntVect(0,0,0)))
867 {
868 mapper = &cell_cons_interp;
869 }
870 else if (box_mf.ixType() == IndexType(IntVect(1,0,0)))
871 {
872 mapper = &face_cons_linear_interp;
873 }
874 else if (box_mf.ixType() == IndexType(IntVect(0,1,0)))
875 {
876 mapper = &face_cons_linear_interp;
877 }
878 else if (box_mf.ixType() == IndexType(IntVect(0,0,1)))
879 {
880 mapper = &face_cons_linear_interp;
881 } else {
882 amrex::Abort("Dont recognize this box type in REMORA_FillPatch");
883 }
884 }
885
886 // InterpFromCoarseLevel copies the coarse ghost cells with periodicity. When the levels
887 // are first made the parent has had no FillPatch yet, so its ghosts beyond a wall and
888 // beyond a periodic edge still hold initial values rather than their periodic images.
889 // Where the new patch touches the seam the two then compete for one cell of the copy,
890 // and on a GPU the winner varies from run to run (Channel_Test_ML_seam, issue #768).
891 MultiFab crse_scratch;
892 const MultiFab& crse_src = coarse_for_interp(lev-1, *mf_crse, crse_scratch);
893 amrex::InterpFromCoarseLevel(*mf_to_fill, mf_to_fill->nGrowVect(), IntVect(0,0,0),
894 crse_src, 0, icomp, ncomp, geom[lev-1], geom[lev],
895 refRatio(lev-1),
896 mapper, domain_bcs_type, bccomp);
897
898 // ***************************************************************************
899 // Physical bc's at domain boundary
900 // ***************************************************************************
901
902 // The radiation conditions compare the field with its previous state (mf_calc),
903 // which a level being created does not have. Give them the interpolated field
904 // itself: the ghost cells then keep the coarse values and the nudging term,
905 // with no time step, is zero. Only the radiation conditions read it, so the
906 // copy is made only when one of them is set on this field, and only of the
907 // components being filled. It cannot be an alias of the field: the radiation
908 // kernels read the along-boundary neighbours of the column they write, so on a
909 // GPU a thread could read a cell another thread has already overwritten.
910 MultiFab calc_copy;
911 const MultiFab* calc = &mf_calc;
912 int icomp_calc_used = icomp_calc;
913 if (!mf_calc.isDefined()) {
914 bool radiation = false;
915 for (int n = 0; n < ncomp; ++n) {
916 const BCRec& bcr = domain_bcs_type[bccomp+n];
917 for (int dir = 0; dir < 2; ++dir) {
918 for (int bc : {bcr.lo(dir), bcr.hi(dir)}) {
919 radiation = radiation || bc == REMORABCType::orlanski_rad
920 || bc == REMORABCType::orlanski_rad_nudge;
921 }
922 }
923 }
924 if (radiation) {
925 calc_copy.define(mf_to_fill->boxArray(), mf_to_fill->DistributionMap(),
926 ncomp, mf_to_fill->nGrowVect());
927 MultiFab::Copy(calc_copy, *mf_to_fill, icomp, 0, ncomp,
928 mf_to_fill->nGrowVect());
929 calc = &calc_copy;
930 icomp_calc_used = 0;
931 }
932 }
933
934 // Enforce physical boundary conditions
935 (*physbcs[lev])(*mf_to_fill,*mask,icomp,ncomp,mf_to_fill->nGrowVect(),
936 time,bccomp,n_not_fill,*calc,
937 *vec_msku[lev].get(), *vec_mskv[lev].get());
938
939#ifdef REMORA_USE_NETCDF
940 // Fill the data which is stored in the boundary data read from netcdf files
941 if ( (solverChoice.boundary_from_netcdf) && (bdy_var_type != BdyVars::null) )
942 {
943 fill_from_bdyfiles(lev, *mf_to_fill,*mask,time,bccomp,bdy_var_type, icomp,icomp_calc_used,*calc,dt_lev);
944 }
945#else
946 amrex::ignore_unused(icomp_calc_used, dt_lev);
947#endif
948 // Fill corners of the domain with periodic data
949 if ( box_mf.ixType() == IndexType(IntVect(0,0,0)) ) {
950 mf_to_fill->EnforcePeriodicity(geom[lev].periodicity());
951 } else {
952 // The face-centred analogue. InterpFromCoarseLevel fills a periodic ghost from the
953 // coarse level, which for a face tangential to the periodic direction does not
954 // reproduce the fine value it wraps to: on a new level of Channel_Test, vbar's x
955 // ghosts at the seam differed from their images by 2e-4 to 6e-4 where zeta's,
956 // corrected above, were exact. The first barotropic substep then sees different vbar
957 // at the two copies of the periodic u-face, and that one-shot asymmetry amplifies into
958 // a blow-up. Copying the ghosts from this level's own valid data, as every later
959 // FillPatch does, removes it.
960 mf_to_fill->FillBoundary(geom[lev].periodicity());
961 }
962
963 // Enforce free-slip at top boundary (on xvel or yvel)
964 if ( (box_mf.ixType() == IndexType(IntVect(1,0,0))) ||
965 (box_mf.ixType() == IndexType(IntVect(0,1,0))) )
966 {
967 int khi = geom[lev].Domain().bigEnd(2);
968#ifdef AMREX_USE_OMP
969#pragma omp parallel if (Gpu::notInLaunchRegion())
970#endif
971 for (MFIter mfi(*mf_to_fill); mfi.isValid(); ++mfi)
972 {
973 Box gbx = mfi.growntilebox(); // Note this is face-centered since vel is
974 gbx.setSmall(2,khi+1);
975 if (gbx.ok()) {
976 Array4<Real> vel_arr = mf_to_fill->array(mfi);
977 ParallelFor(gbx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
978 {
979 vel_arr(i,j,k) = vel_arr(i,j,khi);
980 });
981 }
982 }
983 }

1008/**
1009 * Fill boundary ghost cells for cell-centered velocities. Each component takes
1010 * the sign implied by its own BC, which with remora.boundary_per_variable can
1011 * differ between u, v and w on one face.
1012 *
1013 * @param[in ] lev level to fill at
1014 * @param[inout] mf_cc_vel MultiFab of cell-centered velocities
1015 */
1016void
1017REMORA::FillBdyCCVels (int lev, MultiFab& mf_cc_vel)
1018{
1019 // Impose bc's at domain boundaries
1020 Box domain(Geom(lev).Domain());
1021
1022 int ihi = domain.bigEnd(0);
1023 int jhi = domain.bigEnd(1);
1024 int khi = domain.bigEnd(2);
1025
1026 // Impose periodicity first
1027 mf_cc_vel.FillBoundary(geom[lev].periodicity());
1028
1029 // Sign multipliers per face, per velocity component
1030 GpuArray<GpuArray<Real,AMREX_SPACEDIM>,AMREX_SPACEDIM*2> mult;
1031 for (OrientationIter oit; oit; ++oit) {
1032 Orientation ori = oit();
1033 for (int n = 0; n < AMREX_SPACEDIM; ++n) {
1034 mult[ori][n] = cc_vel_bdy_mult(phys_bc_type[xvel_bc()+n][ori],
1035 n == ori.coordDir());
1036 }
1037 }
1038
1039 // Callers pass different ghost widths (the plotfile ones carry none in z)
1040 const IntVect ng = mf_cc_vel.nGrowVect();
1041
1042#ifdef _OPENMP
1043#pragma omp parallel if (Gpu::notInLaunchRegion())
1044#endif
1045 for (MFIter mfi(mf_cc_vel, TilingIfNotGPU()); mfi.isValid(); ++mfi)
1046 {
1047 // Note that we don't fill corners here -- only the cells that share a face
1048 // with interior cells -- this is all that is needed to calculate vorticity
1049 const Box& bx = mfi.tilebox();
1050 const Array4<Real>& vel_arr = mf_cc_vel.array(mfi);
1051
1052 if (!Geom(lev).isPeriodic(0) && ng[0] > 0) {
1053 // Low-x side
1054 if (bx.smallEnd(0) <= domain.smallEnd(0)) {
1055 auto const m = mult[Orientation(Direction::x,Orientation::low)];
1056 ParallelFor(makeSlab(bx,0,0), [=] AMREX_GPU_DEVICE(int , int j, int k) noexcept
1057 {
