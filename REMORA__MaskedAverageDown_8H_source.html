9/**
10 * Two-way nesting's fine-to-coarse update, weighted by the land/sea mask.
11 *
12 * Ported from ROMS fine2coarse2d/fine2coarse3d in Nonlinear/nesting.F, the AreaAvg = .FALSE.
13 * branch that every call site there uses:
14 *
15 * num = sum of (fine value * fine mask) over the fine points under the coarse one
16 * den = sum of min(1, fine mask) over the same
17 * crse = num/den * coarse mask
18 *
19 * Dividing by the count of wet fine points rather than by how many there are means a coarse
20 * point only partly covered by water takes the mean of that water, not a value diluted toward
21 * zero by land. That is not conservative; ROMS makes the same trade deliberately. The coarse
22 * mask has the final say on wet or dry.
23 *
24 * Where every fine point is wet the mask-weighted kernels have to reproduce amrex_avgdown and
25 * amrex_avgdown_faces bit for bit, or every existing multilevel answer moves in its last bit.
26 * avgdown_weighted with a volume weight (cf_avgdown_volume_weighted) is the exception: it
27 * changes the answer by design.
28 */

47/** Cell-centered fields: the weighted mean of every fine cell under the coarse one. */
48template <typename Weight>
49AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
50void avgdown_weighted (int i, int j, int k, int n,
51 amrex::Array4<amrex::Real> const& crse,
52 amrex::Array4<amrex::Real const> const& fine,
53 Weight const& weight,
54 amrex::Array4<amrex::Real const> const& cmsk,
55 int ccomp, int fcomp, amrex::IntVect const& ratio) noexcept
56{
57 using amrex::Real;
58 const int facx = ratio[0];
59 const int facy = ratio[1];
60 const int facz = ratio[2];
61 const int ii = i*facx;
62 const int jj = j*facy;
63 const int kk = k*facz;
64
65 Real num = Real(0.0);
66 Real den = Real(0.0);
67 // AMReX's loop order, and the mask multiplying the value before it is accumulated rather
68 // than after (x * 1.0 == x), are what make the all-wet case bit-identical to amrex_avgdown.
69 for (int kref = 0; kref < facz; ++kref) {
70 for (int jref = 0; jref < facy; ++jref) {
71 for (int iref = 0; iref < facx; ++iref) {
72 const Real m = weight(ii+iref,jj+jref,kk+kref);
73 num += fine(ii+iref,jj+jref,kk+kref,n+fcomp) * m;
74 den += m;
75 }}}
76
77 // Multiplication by the reciprocal, not num/den: this is AMReX's volfrac, and dividing
78 // would differ in the last bit. den == 0 under a wet coarse point cannot happen once
79 // check_mask_consistency passes, so write zero rather than diagnose it from a kernel.
80 crse(i,j,k,n+ccomp) = (den > Real(0.0))
81 ? num * (Real(1.0)/den) * cmsk(i,j,0)
82 : Real(0.0);
83}
