14// Number of active tracers, i.e. the ones that feed back on the density field:
15// temperature and salinity. ROMS carries vertical mixing coefficients for these two
16// alone, so Tracer_comp doubles as the count of them.
17#define NAT Tracer_comp
18
19/**
20 * The vertical diffusivity component that cons component icomp is mixed with.
21 *
22 * Akt is stored for the active tracers only, and every passive tracer -- dye and biology
23 * alike -- mixes with the salinity coefficient. This is ROMS's ltrc=MIN(itrc,NAT) in
24 * pre_step3d and step3d_t, and it is why vec_Akt has NAT components rather than ncons.
25 * Horizontal diffusivity is a separate story: tnu2 really is per tracer.
26 */
27AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
28int akt_comp (int icomp) noexcept { return (icomp < NAT) ? icomp : Salt_comp; }

94/**
95 * \brief These are only for the variables read in from netcdf boundary condition files
96 *
97 * The cell-centered tracers occupy a contiguous block of slots starting at t, laid
98 * out in the same order as the components of the cons MultiFab. So slot t is temp,
99 * slot t+1 is salt, and slot t+n is cons component n for every additional scalar
100 * (passive tracers, biology tracers). Use cons(icomp) rather than hardcoding this,
101 * and note that the logic in fill_from_bdyfiles relies on the block being contiguous
102 * and in component order.
103 *
104 * Like BCVars, everything above the tracer block is a function of ncons rather than a
105 * compile-time constant. With ncons == 2 these are ubar 4, vbar 5, zeta 6, NumTypes 7.
106 */
107namespace BdyVars {
108 static constexpr int null = -1;
109 static constexpr int u = 0;
110 static constexpr int v = 1;
111 static constexpr int t = 2; ///< cons component Temp_comp
112 static constexpr int s = 3; ///< cons component Salt_comp
113
114 inline int cons (int icomp) noexcept { return t + icomp; }
115 inline int ubar (int ncons) noexcept { return t + ncons; }
116 inline int vbar (int ncons) noexcept { return t + ncons + 1; }
117 inline int zeta (int ncons) noexcept { return t + ncons + 2; }
118 inline int NumTypes (int ncons) noexcept { return t + ncons + 3; }
