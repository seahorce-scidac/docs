10/**
11 * Mass per unit physical volume, by cloud-in-cell deposit in index space. Horizontally the
12 * weights come from the particle's position in the uniform coordinates it moves in, and
13 * vertically from its level index s (cell k spans s in [k, k+1]), so the deposit lands in
14 * the cells around the particle whatever the stretching of the levels. Weight that would
15 * fall below the bed, above the free surface or across a non-periodic side of the domain
16 * goes to the edge cell instead, so no mass is lost. The deposited mass is then divided by
17 * each cell's volume, Hz/(pm*pn).
18 *
19 * @param a_mf Multifab to calculate over
20 * @param a_lev level to calculate on
21 * @param a_Hz cell thickness on this level
22 * @param a_pm inverse grid spacing in x on this level
23 * @param a_pn inverse grid spacing in y on this level
24 * @param a_comp component to calculate
25 */

55 // Lower cell of the stencil and the weight of the upper one, from the
56 // fractional cell-centred index a_xc
57 auto cic = [] (Real a_xc, int a_lo, int a_hi, bool a_clamp, int& a_i0, Real& a_w1)
58 {
59 a_i0 = static_cast<int>(amrex::Math::floor(a_xc));
60 a_w1 = a_xc - static_cast<Real>(a_i0);
61 if (a_clamp) {
62 if (a_i0 < a_lo) {
63 a_i0 = a_lo; a_w1 = zero;
64 } else if (a_i0 >= a_hi) {
65 a_i0 = a_hi; a_w1 = zero;
66 }
67 }
68 };
69
70 int i0, j0, k0;
71 Real wx, wy, wz;
72 cic((static_cast<Real>(p.pos(0)) - plo[0])*dxi[0] - myhalf, dlo.x, dhi.x, clamp_x, i0, wx);
73 cic((static_cast<Real>(p.pos(1)) - plo[1])*dxi[1] - myhalf, dlo.y, dhi.y, clamp_y, j0, wy);
74 cic(static_cast<Real>(ptd.m_rdata[REMORAParticlesRealIdxSoA::s][i]) - myhalf,
75 dlo.z, dhi.z, true, k0, wz);
76
77 for (int kk = 0; kk <= 1; ++kk) {
78 const Real fz = kk ? wz : one - wz;
79 if (fz == zero) { continue; }
80 for (int jj = 0; jj <= 1; ++jj) {
81 const Real fy = jj ? wy : one - wy;
82 if (fy == zero) { continue; }
83 for (int ii = 0; ii <= 1; ++ii) {
84 const Real fx = ii ? wx : one - wx;
85 if (fx == zero) { continue; }
86 Gpu::Atomic::AddNoRet(&rho(i0+ii, j0+jj, k0+kk, a_comp), fx*fy*fz*mass);
87 }
88 }
89 }
90 });
91
92 // Mass per cell to mass per unit volume
93#ifdef AMREX_USE_OMP
94#pragma omp parallel if (Gpu::notInLaunchRegion())
95#endif
96 for (MFIter mfi(a_mf, TilingIfNotGPU()); mfi.isValid(); ++mfi)
97 {
98 const Box& bx = mfi.tilebox();
99 const Array4<Real> rho = a_mf.array(mfi);
100 const Array4<Real const> Hz = a_Hz.const_array(mfi);
101 const Array4<Real const> pm = a_pm.const_array(mfi);
102 const Array4<Real const> pn = a_pn.const_array(mfi);
103 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
104 {
105 rho(i,j,k,a_comp) = (Hz(i,j,k) > zero)
106 ? rho(i,j,k,a_comp) * pm(i,j,0) * pn(i,j,0) / Hz(i,j,k) : zero;
107 });
108 }
109
110 return;
111}

202/**
203 * Write the particles of one species, for a checkpoint or a plotfile. With a_xy null they are
204 * written as they are, in the uniform coordinates of the AMReX geometry they move in. With a_xy
205 * given (plotfiles), x and y are replaced by the physical coordinates of the cell corners (psi
206 * points) interpolated bilinearly to each particle. These are the corner positions the
207 * plotfile's nodal displacement gives a viewer, so each particle is drawn inside the cell that
208 * holds it. z is already the physical height. The uniform x and y are saved before the
209 * conversion and restored bit for bit after the write, so writing a plotfile does not change
210 * the run.
211 *
212 * Each particle stays filed under the grid that physically holds it, though its physical
213 * position need not map to that grid through the uniform geometry, so pc.OK() does not hold
214 * during the write. AMReX's particle writers do not require it (AMReX-Codes/amrex#6175).
215 *
216 * @param a_fname checkpoint or plotfile directory
217 * @param a_name particle species name
218 * @param a_xy null, or per level the physical x and y of the psi points
219 * (REMORA::plot_node_coords)
220 */
221void REMORAPC::WriteOutput ( const std::string& a_fname, const std::string& a_name,
222 const Vector<MultiFab>* a_xy )
223{
224 BL_PROFILE("REMORAPC::WriteOutput()");
225
226 if (a_xy == nullptr) {
227 Checkpoint(a_fname, a_name, true, varNames());
228 return;
229 }
230
231 // The uniform x and y of each tile, keyed like the particle tiles by (grid, tile). Every
232 // entry is made here, so the threaded loops below only look entries up.
233 Vector<std::map<std::pair<int,int>, Gpu::DeviceVector<ParticleReal>>> saved_xy(finestLevel()+1);
234 for (int lev = 0; lev <= finestLevel(); ++lev) {
235 for (const auto& kv : GetParticles(lev)) {
236 saved_xy[lev][kv.first].resize(2 * kv.second.numParticles());
237 }
238 }
239
240 for (int lev = 0; lev <= finestLevel(); ++lev)
241 {
242 const Geometry& geom = m_gdb->Geom(lev);
243 const auto plo = geom.ProbLoArray();
244 const auto dxi = geom.InvCellSizeArray();
245
246#ifdef AMREX_USE_OMP
247#pragma omp parallel if (Gpu::notInLaunchRegion())
248#endif
249 for (ParIterType pti(*this, lev); pti.isValid(); ++pti)
250 {
251 auto& aos = pti.GetArrayOfStructs();
252 const int np = aos.numParticles();
253 auto* p_pbox = aos().data();
254 auto* p_save = saved_xy[lev].at(pti.GetPairIndex()).data();
255 const auto xy = (*a_xy)[lev].const_array(pti.index());
256 const Box& vbx = pti.validbox();
257 const int ilo = vbx.smallEnd(0);
258 const int ihi = vbx.bigEnd(0);
259 const int jlo = vbx.smallEnd(1);
260 const int jhi = vbx.bigEnd(1);
261
262 ParallelFor(np, [=] AMREX_GPU_DEVICE (int n)
263 {
264 auto& p = p_pbox[n];
265 p_save[2*n ] = p.pos(0);
266 p_save[2*n+1] = p.pos(1);
267 if (p.id() <= 0) { return; }
268
269 // Nodal index coordinates, kept in ParticleReal: rounding them to a
270 // single-precision Real first can carry a particle at the high edge of its grid
271 // into the next cell. The cell is clamped to the particle's grid, so both corner
272 // nodes are always inside the grid's nodal box (a_xy has no ghost cells).
273 const ParticleReal lx = (p.pos(0)-plo[0])*dxi[0];
274 const ParticleReal ly = (p.pos(1)-plo[1])*dxi[1];
275 const int i = amrex::max(ilo, amrex::min(static_cast<int>(amrex::Math::floor(lx)), ihi));
276 const int j = amrex::max(jlo, amrex::min(static_cast<int>(amrex::Math::floor(ly)), jhi));
277 const auto sx = static_cast<Real>(lx - static_cast<ParticleReal>(i));
278 const auto sy = static_cast<Real>(ly - static_cast<ParticleReal>(j));
279
280 for (int d = 0; d < 2; ++d) {
281 p.pos(d) = static_cast<ParticleReal>(
282 xy(i ,j ,0,d) * (one-sx) * (one-sy) + xy(i+1,j ,0,d) * sx * (one-sy) +
283 xy(i ,j+1,0,d) * (one-sx) * sy + xy(i+1,j+1,0,d) * sx * sy);
284 }
285 });
286 }
287 }
288
289 Checkpoint(a_fname, a_name, true, varNames());
290
291 for (int lev = 0; lev <= finestLevel(); ++lev)
292 {
293#ifdef AMREX_USE_OMP
294#pragma omp parallel if (Gpu::notInLaunchRegion())
295#endif
296 for (ParIterType pti(*this, lev); pti.isValid(); ++pti)
297 {
298 auto& aos = pti.GetArrayOfStructs();
299 const int np = aos.numParticles();
300 auto* p_pbox = aos().data();
301 const auto* p_save = saved_xy[lev].at(pti.GetPairIndex()).data();
302 ParallelFor(np, [=] AMREX_GPU_DEVICE (int n)
303 {
304 p_pbox[n].pos(0) = p_save[2*n ];
305 p_pbox[n].pos(1) = p_save[2*n+1];
306 });
307 }
308 }
309 // saved_xy is freed on return. The restore kernels may run on several streams, so wait for
310 // the whole device rather than relying on ParIter's own synchronization.
311 Gpu::synchronize();
312}
