6/** \brief Copy the freshly filled leapfrog component into the other two, in the coarse-fine
7 * ghost band only.
8 *
9 * FillPatch writes only component knew. The solver updates every component inside the fine
10 * grid but nothing writes the coarse-fine ghost band, so its other components stay a full
11 * parent step stale -- and the 2D solver reads zeta(krhs) and ubar(kstp) from exactly those
12 * cells. ROMS's put_refine2d sets the contact points it is about to read, so make every
13 * component agree with the parent state just filled.
14 *
15 * The band is found by marking the valid region and calling FillBoundary: ghosts another box
16 * on this level covers pick up a 1, so the remaining zeros are the coarse-fine and
17 * domain-boundary ghosts. Domain ghosts are excluded so the physical boundary conditions,
18 * applied after the FillPatch, are not overwritten.
19 */
20void fill_ghost_kcomps (MultiFab& mf, int knew, const Geometry& geom)
21{
22 MultiFab valid(mf.boxArray(), mf.DistributionMap(), 1, mf.nGrowVect());
23 valid.setVal(Real(0.0));
24 valid.setVal(Real(1.0), 0, 1, 0);
25 valid.FillBoundary(geom.periodicity());
26
27 const Box dom = amrex::convert(geom.Domain(), mf.boxArray().ixType());
28
29 const bool per_x = geom.isPeriodic(0);
30 const bool per_y = geom.isPeriodic(1);
31
32#ifdef AMREX_USE_OMP
33#pragma omp parallel if (Gpu::notInLaunchRegion())
34#endif
35 for (MFIter mfi(mf, TilingIfNotGPU()); mfi.isValid(); ++mfi) {
36 Box gbx = mfi.growntilebox();
37 const auto& a = mf.array(mfi);
38 const auto& v = valid.const_array(mfi);
39 const auto dlo = amrex::lbound(dom);
40 const auto dhi = amrex::ubound(dom);
41 ParallelFor(gbx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
42 {
43 if (v(i,j,k) > Real(0.5)) { return; }
44 // Skip ghosts outside the domain only where the domain really ends. A ghost beyond
45 // a periodic edge is not a physical-boundary cell: if the level wraps onto itself
46 // the FillBoundary above already marked it valid, and if it does not -- a patch
47 // covering part of the periodic width -- it is an ordinary coarse-fine ghost that
48 // FillPatch just filled from the parent, needing the same synchronisation as any
49 // other.
50 if (!per_x && (i < dlo.x || i > dhi.x)) { return; }
51 if (!per_y && (j < dlo.y || j > dhi.y)) { return; }
52 const Real val = a(i,j,k,knew);
53 a(i,j,k,0) = val;
54 a(i,j,k,1) = val;
55 a(i,j,k,2) = val;
56 });
