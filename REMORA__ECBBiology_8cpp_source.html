48void
49REMORABiology::ECBParameters::init_params (const std::string& remora_prefix)
50{
51 ParmParse pp(remora_prefix + ".ecb");
52
53 // ECB is ported with one ROMS option set; reject the Fennel-style switches
54 // rather than ignore them.
55 for (const char* option : {"carbon", "oxygen", "bio_sediment", "denitrification", "po4",
56 "talk_nonconserv", "co2_schmidt", "oxygen_schmidt",
57 "pco2air_type", "pco2_rz", "sediment_proxy",
58 "net_sulfate_reduction", "wet_dry"}) {
59 if (pp.contains(option)) {
60 amrex::Abort("ECB implements a fixed ROMS option set; unsupported option: " +
61 remora_prefix + ".ecb." + option);
62 }
63 }
64
65 // The ROMS Fortran kernel and the parity diagnostics exist only for Fennel,
66 // so asking for either with ECB is an error rather than a silent C++ run.
67 {
68 ParmParse pp_remora(remora_prefix);
69 int use_cpp = 1;
70 int debug = 0;
71 pp_remora.query("use_biology_cpp_answer", use_cpp);
72 pp_remora.query("biology_debug", debug);
73 if (use_cpp != 1) {
74 amrex::Abort(remora_prefix + ".use_biology_cpp_answer = 0 selects the ROMS Fortran "
75 "kernel, which exists only for Fennel, not ECB.");
76 }
77 if (debug > 0) {
78 amrex::Abort(remora_prefix + ".biology_debug > 0 requests the parity diagnostics, "
79 "which exist only for Fennel, not ECB.");
80 }
81 }
82
83 pp.queryAdd("BioIter", BioIter);
84 pp.queryAdd("AttSW", AttSW);
85 pp.queryAdd("AttChl", AttChl);
86 pp.queryAdd("PARfrac", PARfrac);
87 pp.queryAdd("Vp0", Vp0);
88 pp.queryAdd("I_thNH4", I_thNH4);
89 pp.queryAdd("D_p5NH4", D_p5NH4);
90 pp.queryAdd("NitriR", NitriR);
91 pp.queryAdd("K_NO3", K_NO3);
92 pp.queryAdd("K_NH4", K_NH4);
93 pp.queryAdd("K_PO4", K_PO4);
94 pp.queryAdd("K_Phy", K_Phy);
95 pp.queryAdd("Chl2C_m", Chl2C_m);
96 pp.queryAdd("ChlMin", ChlMin);
97 pp.queryAdd("PhyCN", PhyCN);
98 pp.queryAdd("R_P2N", R_P2N);
99 pp.queryAdd("PhyIS", PhyIS);
100 pp.queryAdd("PhyMin", PhyMin);
101 pp.queryAdd("PhyMR", PhyMR);
102 pp.queryAdd("ElDON", ElDON);
103 pp.queryAdd("EsDON", EsDON);
104 pp.queryAdd("deltN", deltN);
105 pp.queryAdd("deltC", deltC);
106 pp.queryAdd("gammaC", gammaC);
107 pp.queryAdd("slCexc", slCexc);
108 pp.queryAdd("ZooAE_N", ZooAE_N);
109 pp.queryAdd("ZooBM", ZooBM);
110 pp.queryAdd("ZooCN", ZooCN);
111 pp.queryAdd("ZooER", ZooER);
112 pp.queryAdd("ZooGR", ZooGR);
113 pp.queryAdd("ZooMin", ZooMin);
114 pp.queryAdd("ZooMR", ZooMR);
115 pp.queryAdd("LDeRRN", LDeRRN);
116 pp.queryAdd("LDeRRC", LDeRRC);
117 pp.queryAdd("CoagR", CoagR);
118 pp.queryAdd("SDeRRN", SDeRRN);
119 pp.queryAdd("SDeRRC", SDeRRC);
120 pp.queryAdd("RDeRRN", RDeRRN);
121 pp.queryAdd("RDeRRC", RDeRRC);
122 pp.queryAdd("wPhy", wPhy);
123 pp.queryAdd("wLDet", wLDet);
124 pp.queryAdd("wSDet", wSDet);
125 pp.queryAdd("pCO2air", pCO2air);
126
127 // Reject values that divide by zero in the kernel, where they would
128 // otherwise surface as NaN in the tracers with no diagnostic.
129 const std::string prefix = remora_prefix + ".ecb";
130 // Light attenuation (AttSW + AttChl*chl + DOC term) * dz may be zero, which the
131 // kernel handles, but a negative coefficient would amplify light with depth.
132 require_parameter(AttSW >= Real(0.0), prefix, "AttSW", "non-negative", AttSW);
133 require_parameter(AttChl >= Real(0.0), prefix, "AttChl", "non-negative", AttChl);
134 // Carbon excess uptake divides by slCexc.
135 require_parameter(slCexc > Real(0.0), prefix, "slCexc", "positive", slCexc);
136 // Grazing divides by K_Phy + phyt^2, which is K_Phy when phyt is zero.
137 require_parameter(K_Phy > Real(0.0), prefix, "K_Phy", "positive", K_Phy);
138 // A non-positive atmospheric pCO2 reverses the sign of the air-sea CO2 flux.
139 require_parameter(pCO2air > Real(0.0), prefix, "pCO2air", "positive", pCO2air);
140}

150void
151REMORABiology::ECBBiology::advance (const BiologyAdvanceContext& context) const
152{
153 const auto parms = m_parameters;
154 if (parms.BioIter <= 0) {
155 return;
156 }
157
158 auto const& mf_cons_old = context.old_state;
159 auto& mf_cons_new = context.new_state;
160 const bool do_bulk_flux = context.solver_choice.bulk_fluxes;
161 const int N = context.top_layer + 1; // number of vertical layers
162 const Real dt_lev = context.dt;
163 const Real rho0 = context.solver_choice.rho0;
164
165 if (context.hz == nullptr || context.z_w == nullptr || context.mask == nullptr ||
166 context.shortwave == nullptr ||
167 context.bottom_stress_u == nullptr || context.bottom_stress_v == nullptr ||
168 (do_bulk_flux && (context.wind_u == nullptr || context.wind_v == nullptr)) ||
169 (!do_bulk_flux && (context.stress_u == nullptr || context.stress_v == nullptr))) {
170 amrex::Abort("ECB biology requires Hz, z_w, rmask, srflx, bottom stresses, and "
171 "either uwind/vwind (bulk_fluxes) or sustr/svstr");
172 }
173 if (N < 2) {
174 amrex::Abort("ECB sinking requires at least two vertical layers");
175 }
176 if (dt_lev <= Real(0.0)) {
177 amrex::Abort("ECB requires a positive timestep when BioIter > 0");
178 }
179
180 // Set time-stepping according to the number of iterations.
181 const Real dtdays = dt_lev * Real(1.0 / 86400.0) / Real(parms.BioIter);
182
183 // Air-sea exchange always uses the Wanninkhof (2014) relations (ROMS
184 // RW14_OXYGEN_SC and RW14_CO2_SC).
185 const SchmidtCoefficients o2_schmidt = GasExchange::O2Wanninkhof2014;
186 const SchmidtCoefficients co2_schmidt = GasExchange::CO2Wanninkhof2014;
187
188 // Set biological tracer component identifiers, looked up by name.
189 auto const& decl = m_declaration;
190 auto comp = [&] (const char* name) {
191 const int c = decl.component_index(name, context.component_start);
192 AMREX_ALWAYS_ASSERT(c >= 0);
193 return c;
194 };
195 const int c_no3 = comp("NO3");
196 const int c_nh4 = comp("NH4");
197 const int c_chlo = comp("chlorophyll");
198 const int c_phyt = comp("phytoplankton");
199 const int c_zoop = comp("zooplankton");
200 const int c_lden = comp("LdetritusN");
201 const int c_sden = comp("SdetritusN");
202 const int c_don = comp("DON");
203 const int c_po4 = comp("PO4");
204 const int c_ldec = comp("LdetritusC");
205 const int c_sdec = comp("SdetritusC");
206 const int c_tic = comp("TIC");
207 const int c_talk = comp("alkalinity");
208 const int c_doc = comp("DOC");
209 const int c_oxyg = comp("oxygen");
210 const int c_rdon = comp("Rdon");
211 const int c_rdoc = comp("Rdoc");
212 const int c_calc = comp("calcium");
213
214 // Scratch arrays mirror the ROMS Bio, qc, bR, bL, WR, WL, and
215 // inverse-thickness work arrays, but use per-tile FArrayBox storage.
216 int ncell = 0;
217 const int sc_no3 = ncell++;
218 const int sc_nh4 = ncell++;
219 const int sc_chlo = ncell++;
220 const int sc_phyt = ncell++;
221 const int sc_zoop = ncell++;
222 const int sc_lden = ncell++;
223 const int sc_sden = ncell++;
224 const int sc_don = ncell++;
225 const int sc_po4 = ncell++;
226 const int sc_ldec = ncell++;
227 const int sc_sdec = ncell++;
228 const int sc_tic = ncell++;
229 const int sc_talk = ncell++;
230 const int sc_doc = ncell++;
231 const int sc_oxyg = ncell++;
232 const int sc_rdon = ncell++;
233 const int sc_rdoc = ncell++;
234 const int sc_calc = ncell++;
235 const int sc_temp = ncell++;
236 const int sc_salt = ncell++;
237 const int sc_inv_hz = ncell++;
238 const int sc_inv_hz2 = ncell++;
239 const int sc_inv_hz3 = ncell++;
240 const int sc_qc = ncell++;
241 const int sc_bR = ncell++;
242 const int sc_bL = ncell++;
243 const int sc_WR = ncell++;
244 const int sc_WL = ncell++;
245
246 int nw = 0;
247 const int sw_FC = nw++;
248
249#ifdef AMREX_USE_OMP
250#pragma omp parallel if (Gpu::notInLaunchRegion())
251#endif
252 for (MFIter mfi(mf_cons_new, TilingIfNotGPU()); mfi.isValid(); ++mfi) {
253 Box bx = mfi.tilebox();
254 Box bx2d = bx;
255 bx2d.makeSlab(2, 0);
256 Box wbx = surroundingNodes(bx, 2);
257
258 // Every scratch value is written before it is read, so it is not zeroed.
259 FArrayBox fab_cell(bx, ncell, amrex::The_Async_Arena());
260 FArrayBox fab_w(wbx, nw, amrex::The_Async_Arena());
261
262 auto no3 = fab_cell.array(sc_no3);
263 auto nh4 = fab_cell.array(sc_nh4);
264 auto chlo = fab_cell.array(sc_chlo);
265 auto phyt = fab_cell.array(sc_phyt);
266 auto zoop = fab_cell.array(sc_zoop);
267 auto lden = fab_cell.array(sc_lden);
268 auto sden = fab_cell.array(sc_sden);
269 auto don = fab_cell.array(sc_don);
270 auto po4 = fab_cell.array(sc_po4);
271 auto ldec = fab_cell.array(sc_ldec);
272 auto sdec = fab_cell.array(sc_sdec);
273 auto tic = fab_cell.array(sc_tic);
274 auto talk = fab_cell.array(sc_talk);
275 auto doc = fab_cell.array(sc_doc);
276 auto oxyg = fab_cell.array(sc_oxyg);
277 auto rdon_s = fab_cell.array(sc_rdon);
278 auto rdoc_s = fab_cell.array(sc_rdoc);
279 auto calc = fab_cell.array(sc_calc);
280 auto temp = fab_cell.array(sc_temp);
281 auto salt = fab_cell.array(sc_salt);
282 auto inv_hz = fab_cell.array(sc_inv_hz);
283 auto inv_hz2 = fab_cell.array(sc_inv_hz2);
284 auto inv_hz3 = fab_cell.array(sc_inv_hz3);
285 auto qc = fab_cell.array(sc_qc);
286 auto bR = fab_cell.array(sc_bR);
287 auto bL = fab_cell.array(sc_bL);
288 auto WR = fab_cell.array(sc_WR);
289 auto WL = fab_cell.array(sc_WL);
290 auto FC = fab_w.array(sw_FC);
291
292 Array4<Real const> const& state_old = mf_cons_old.const_array(mfi);
293 Array4<Real> const& state_new = mf_cons_new.array(mfi);
294 Array4<Real const> const& Hz = context.hz->const_array(mfi);
295 Array4<Real const> const& z_w = context.z_w->const_array(mfi);
296 Array4<Real const> const& srflx = context.shortwave->const_array(mfi);
297 Array4<Real const> const& mskr = context.mask->const_array(mfi);
298 Array4<Real const> const& bustr = context.bottom_stress_u->const_array(mfi);
299 Array4<Real const> const& bvstr = context.bottom_stress_v->const_array(mfi);
300 // ROMS takes the gas-exchange transfer velocity from Uwind/Vwind
301 // under BULK_FLUXES and from the surface stress otherwise; only the
302 // selected pair is read.
303 Array4<Real const> sustr;
304 Array4<Real const> svstr;
305 Array4<Real const> uwind;
306 Array4<Real const> vwind;
307 if (do_bulk_flux) {
308 uwind = context.wind_u->const_array(mfi);
309 vwind = context.wind_v->const_array(mfi);
310 } else {
311 sustr = context.stress_u->const_array(mfi);
312 svstr = context.stress_v->const_array(mfi);
313 }
314
315 ParallelFor(bx2d, [=] AMREX_GPU_DEVICE (int i, int j, int) noexcept
316 {
317 constexpr Real zero = Real(0.0);
318 // Land columns contribute nothing: the increment below is scaled by rmask.
319 if (mskr(i,j,0) == zero) return;
320
321 constexpr Real one = Real(1.0);
322 constexpr Real two = Real(2.0);
323 constexpr Real eps = Real(1.0e-20);
324 constexpr Real minval = Real(1.0e-6);
325
326 constexpr Real rOxNO3 = Real(8.625); // 138/16
327 constexpr Real rOxNH4 = Real(6.625); // 106/16
328 constexpr Real rOxNH4Denit = Real(115.0) / Real(16.0);
329 constexpr Real denitrification_NH4_fraction = Real(4.0) / Real(16.0);
330 constexpr Real K_Nitri = Real(1.0); // O2 half-saturation, nitrification
331 constexpr Real K_Denit = K_Nitri; // O2 half-saturation, denitrification
332 constexpr Real K_wNO3 = Real(0.5); // NO3 half-saturation, denitrification
333 constexpr Real K_bO2 = Real(26.5); // O2 half-saturation, sediment denitrification
334 constexpr Real TIC_min = Real(1.0);
335 constexpr Real TIC_max = Real(3000.0);
336
337 // Compute inverse thickness to avoid repeated divisions.
338 for (int k = 0; k < N; ++k) {
339 inv_hz(i,j,k) = one / Hz(i,j,k);
340 }
341 for (int k = 0; k < N - 1; ++k) {
342 inv_hz2(i,j,k) = one / (Hz(i,j,k) + Hz(i,j,k+1));
343 }
344 for (int k = 1; k < N - 1; ++k) {
345 inv_hz3(i,j,k) = one / (Hz(i,j,k-1) + Hz(i,j,k) + Hz(i,j,k+1));
346 }
347
348 // Extract biological variables from tracer arrays, place them into
349 // scratch arrays, and restrict their values to be positive definite.
350 // TIC is also bounded, as in ROMS.
351 for (int k = 0; k < N; ++k) {
352 no3(i,j,k) = amrex::max(zero, state_old(i,j,k,c_no3));
353 nh4(i,j,k) = amrex::max(zero, state_old(i,j,k,c_nh4));
354 chlo(i,j,k) = amrex::max(zero, state_old(i,j,k,c_chlo));
355 phyt(i,j,k) = amrex::max(zero, state_old(i,j,k,c_phyt));
356 zoop(i,j,k) = amrex::max(zero, state_old(i,j,k,c_zoop));
357 lden(i,j,k) = amrex::max(zero, state_old(i,j,k,c_lden));
358 sden(i,j,k) = amrex::max(zero, state_old(i,j,k,c_sden));
359 don(i,j,k) = amrex::max(zero, state_old(i,j,k,c_don));
360 po4(i,j,k) = amrex::max(zero, state_old(i,j,k,c_po4));
361 ldec(i,j,k) = amrex::max(zero, state_old(i,j,k,c_ldec));
362 sdec(i,j,k) = amrex::max(zero, state_old(i,j,k,c_sdec));
363 tic(i,j,k) = amrex::max(amrex::min(amrex::max(zero, state_old(i,j,k,c_tic)),
364 TIC_max), TIC_min);
365 talk(i,j,k) = amrex::max(zero, state_old(i,j,k,c_talk));
366 doc(i,j,k) = amrex::max(zero, state_old(i,j,k,c_doc));
367 oxyg(i,j,k) = amrex::max(zero, state_old(i,j,k,c_oxyg));
368 rdon_s(i,j,k) = amrex::max(zero, state_old(i,j,k,c_rdon));
369 rdoc_s(i,j,k) = amrex::max(zero, state_old(i,j,k,c_rdoc));
370 calc(i,j,k) = amrex::max(zero, state_old(i,j,k,c_calc));
371 temp(i,j,k) = amrex::min(state_old(i,j,k,Temp_comp), Real(35.0));
372 salt(i,j,k) = amrex::max(state_old(i,j,k,Salt_comp), zero);
373 }
374
375 // Calculate surface Photosynthetically Available Radiation (PAR).
376 // REMORA stores srflx in W/m2, so only PARfrac applies. ROMS ecb.h
377 // multiplies by rho0*Cp because its srflx is kinematic (degC m/s).
378 const Real PARsur = parms.PARfrac * srflx(i,j,0);
379
380 // The surface gas-exchange factors depend only on the surface
381 // temperature, salinity and wind, which the iterations do not change.
382 const Real u10squ = u10_squared(do_bulk_flux, uwind, vwind, sustr, svstr, i, j, rho0);
383 const Real O2_transfer = transfer_velocity(transfer_rate(o2_schmidt, dtdays), u10squ,
384 schmidt_number(o2_schmidt, temp(i,j,N-1)));
385 const Real O2satu = o2_saturation(temp(i,j,N-1), salt(i,j,N-1));
386 const Real CO2_transfer = transfer_velocity(transfer_rate(co2_schmidt, dtdays), u10squ,
387 schmidt_number(co2_schmidt, temp(i,j,N-1)));
388 const Real CO2_sol = co2_solubility(temp(i,j,N-1), salt(i,j,N-1));
389
390 // Subcycle the complete ordered ECB reaction/column calculation.
391 for (int iter = 0; iter < parms.BioIter; ++iter) {
392
393 // Primary production, exudation and nitrification. Light is
394 // attenuated from the surface down, so PAR at a depth depends
395 // on the whole column above it.
396 if (PARsur > zero) {
397 Real PAR = PARsur;
398 for (int k = N - 1; k >= 0; --k) {
399 const Real attfac = amrex::max(doc(i,j,k) + rdoc_s(i,j,k) - Real(70.819), zero) *
400 Real(3.786e-3);
401 const Real Att = (parms.AttSW + parms.AttChl * chlo(i,j,k) + attfac) *
402 (z_w(i,j,k+1) - z_w(i,j,k));
403 const Real ExpAtt = std::exp(-Att);
404 const Real Itop = PAR;
405 // Mean PAR over the cell; Itop when nothing attenuates.
406 PAR = (Att > zero) ? Itop * (one - ExpAtt) / Att : Itop;
407
408 const Real cff = parms.PhyCN * Real(12.0);
409 const Real Chl2C = amrex::min(chlo(i,j,k) / (phyt(i,j,k) * cff + eps), parms.Chl2C_m);
410 const Real Vp = amrex::max(parms.Vp0 * std::exp(Real(0.08065) * temp(i,j,k)), Real(4.00));
411 const Real fac1 = PAR * parms.PhyIS;
412 const Real Epp = Vp / std::sqrt(Vp * Vp + fac1 * fac1);
413 const Real t_PPmax = Epp * fac1;
414
415 // Nutrient limitation.
416 Real cff1 = nh4(i,j,k) * parms.K_NH4;
417 const Real cff2 = no3(i,j,k) * parms.K_NO3;
418 const Real inhNH4 = one / (one + cff1);
419 const Real L_NH4 = cff1 / (one + cff1);
420 const Real L_NO3 = cff2 * inhNH4 / (one + cff2);
421 const Real LTOT = L_NO3 + L_NH4;
422 const Real cff3 = po4(i,j,k) * parms.K_PO4;
423 const Real L_PO4 = cff3 / (one + cff3);
424 const Real LMIN = amrex::min(LTOT, L_PO4);
425
426 // Nutrient uptake.
427 const Real mu = dtdays * t_PPmax * LMIN;
428 const Real cff4 = mu * phyt(i,j,k) * L_NO3 /
429 amrex::max(minval, LTOT) / amrex::max(minval, no3(i,j,k));
430 const Real cff5 = mu * phyt(i,j,k) * L_NH4 /
431 amrex::max(minval, LTOT) / amrex::max(minval, nh4(i,j,k));
432 const Real cff6 = parms.R_P2N * mu * phyt(i,j,k) / amrex::max(minval, po4(i,j,k));
433 no3(i,j,k) = no3(i,j,k) / (one + cff4);
434 nh4(i,j,k) = nh4(i,j,k) / (one + cff5);
435 po4(i,j,k) = po4(i,j,k) / (one + cff6);
436 const Real N_Flux_NewProd = no3(i,j,k) * cff4;
437 const Real N_Flux_RegProd = nh4(i,j,k) * cff5;
438
439 // Phytoplankton growth and DON exudation.
440 Real f_ntr = oxyg(i,j,k) / (oxyg(i,j,k) + K_Nitri);
441 const Real f_dnf = K_Denit / (oxyg(i,j,k) + K_Denit) * no3(i,j,k) /
442 (no3(i,j,k) + K_wNO3);
443 phyt(i,j,k) = phyt(i,j,k) + (one - parms.EsDON - (f_ntr + f_dnf) * parms.ElDON) *
444 (N_Flux_NewProd + N_Flux_RegProd);
445 chlo(i,j,k) = chlo(i,j,k) + (dtdays * t_PPmax * t_PPmax * LMIN * LMIN *
446 parms.Chl2C_m * chlo(i,j,k)) /
447 (parms.PhyIS * amrex::max(Chl2C, eps) * PAR + eps);
448 don(i,j,k) = don(i,j,k) + parms.EsDON * (N_Flux_NewProd + N_Flux_RegProd);
449 nh4(i,j,k) = nh4(i,j,k) + (f_ntr + f_dnf) * parms.ElDON *
450 (N_Flux_NewProd + N_Flux_RegProd);
451 talk(i,j,k) = talk(i,j,k) + (f_ntr + f_dnf) * parms.ElDON *
452 (N_Flux_NewProd + N_Flux_RegProd);
