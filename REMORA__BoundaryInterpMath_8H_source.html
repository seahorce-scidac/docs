17/**
18 * For fine cell f along a cell-centered strip refined by r, find the coarse cell c below it
19 * and the weight w of coarse cell c+1, so that the fine value is
20 * crse[c] + w*(crse[c+1] - crse[c]). Fine cell f is centered at (f+0.5)/r coarse cell widths
21 * and coarse cell c at c+0.5. When w is zero, crse[c+1] need not exist.
22 *
23 * @param[in ] f fine cell index
24 * @param[in ] r refinement ratio along the strip
25 * @param[ out] c lower coarse cell index
26 * @param[ out] w weight of coarse cell c+1, in [0, 1)
27 */
28AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
29void
30cell_centered_weight (int f, int r, int& c, amrex::Real& w)
31{
32 // 2r times the fine center's position, measured in coarse cells from the center of
33 // coarse cell 0. Integer arithmetic gives the coarse index exactly, and the weight with a
34 // single rounding.
35 const int num = 2*f + 1 - r;
36 const int den = 2*r;
37 c = (num >= 0) ? num / den : -((-num + den - 1) / den);
38 w = static_cast<amrex::Real>(num - c*den) / static_cast<amrex::Real>(den);
39}

41/**
42 * For fine node f along a nodal strip refined by r, find the coarse node c at or below it and
43 * the weight w of coarse node c+1, as for cell_centered_weight. Fine node f lies at f/r coarse
44 * cell widths and coarse node c at c, so w is zero where the nodes coincide.
45 *
46 * @param[in ] f fine node index
47 * @param[in ] r refinement ratio along the strip
48 * @param[ out] c lower coarse node index
49 * @param[ out] w weight of coarse node c+1, in [0, 1)
50 */
51AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
52void
53nodal_weight (int f, int r, int& c, amrex::Real& w)
54{
55 c = (f >= 0) ? f / r : -((-f + r - 1) / r);
56 w = static_cast<amrex::Real>(f - c*r) * (amrex::Real(1.0) / static_cast<amrex::Real>(r));
57}
