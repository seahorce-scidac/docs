13/**
14 * Surface-water pCO2 [ppmv] from temperature [degC], salinity, total
15 * inorganic carbon and total alkalinity [mmol/m3], by bracketing and
16 * bisection for [H+]. This is ROMS pCO2_water with phosphate and silicate
17 * set to zero, shared by the Fennel and ECB kernels, whose ROMS sources carry
18 * identical copies.
19 */
20AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
21amrex::Real pco2_water (amrex::Real T, amrex::Real S, amrex::Real TIC,
22 amrex::Real TAlk) noexcept
23{
24 constexpr Real zero = Real(0.0);
25 constexpr int IbrackMax = 30;
26
27 // Determine coefficients for surface carbon chemistry.
28 const Real Tk = T + Real(273.15);
29 const Real centiTk = Real(0.01) * Tk;
30 const Real invTk = Real(1.0) / Tk;
31 const Real logTk = std::log(Tk);
32 const Real sqrtS = std::sqrt(S);
33 const Real SO4 = Real(19.924) * S / (Real(1000.0) - Real(1.005) * S);
34 const Real sqrtSO4 = std::sqrt(SO4);
35 const Real scl = S / Real(1.80655);
36
37 const Real alk = TAlk * Real(0.000001);
38 const Real dic = TIC * Real(0.000001);
39 const Real phos = zero;
40 const Real sili = zero;
41
42 // Correction term for non-ideality, ff=k0*(1-pH2O). Equation 13 with
43 // table 6 values from Weiss and Price (1980, Mar. Chem., 8, 347-359).
44 const Real ff = std::exp(-Real(162.8301) +
45 Real(218.2968) / centiTk +
46 std::log(centiTk) * Real(90.9241) -
47 centiTk * centiTk * Real(1.47696) +
48 S * (Real(0.025695) -
49 centiTk * (Real(0.025225) -
50 centiTk * Real(0.0049867))));
51
52 // Compute first (K1) and second (K2) dissociation constant of carboinic
53 // acid:
54 // K1 = [H][HCO3]/[H2CO3]
55 // K2 = [H][CO3]/[HCO3]
56 // From Millero (1995; page 664) using Mehrbach et al. (1973) data on
57 // seawater scale.
58 const Real K1 = std::pow(Real(10.0), Real(62.008) -
59 invTk * Real(3670.7) -
60 logTk * Real(9.7944) +
61 S * (Real(0.0118) - S * Real(0.000116)));
62 const Real K2 = std::pow(Real(10.0), -Real(4.777) -
63 invTk * Real(1394.7) +
64 S * (Real(0.0184) - S * Real(0.000118)));
65
66 // Compute dissociation constant of boric acid, Kb=[H][BO2]/[HBO2].
67 // From Millero (1995; page 669) using data from Dickson (1990).
68 const Real Kb = std::exp(-invTk * (Real(8966.90) +
69 sqrtS * (Real(2890.53) +
70 sqrtS * (Real(77.942) -
71 sqrtS * (Real(1.728) -
72 sqrtS * Real(0.0996))))) -
73 logTk * (Real(24.4344) +
74 sqrtS * (Real(25.085) + sqrtS * Real(0.2474))) +
75 Tk * (sqrtS * Real(0.053105)) +
76 Real(148.0248) +
77 sqrtS * (Real(137.1942) + sqrtS * Real(1.62142)));
78
79 // Compute first (K1p), second (K2p), and third (K3p) dissociation
80 // constant of phosphoric acid:
81 // K1p = [H][H2PO4]/[H3PO4]
82 // K2p = [H][HPO4]/[H2PO4]
83 // K3p = [H][PO4]/[HPO4]
84 // From DOE (1994) equations 7.2.20, 7.2.23, and 7.2.26, respectively.
85 // With footnote using data from Millero (1974).
86 const Real K1p = std::exp(Real(115.525) -
87 invTk * Real(4576.752) -
88 logTk * Real(18.453) +
89 sqrtS * (Real(0.69171) - invTk * Real(106.736)) -
90 S * (Real(0.01844) + invTk * Real(0.65643)));
91 const Real K2p = std::exp(Real(172.0883) -
92 invTk * Real(8814.715) -
93 logTk * Real(27.927) +
94 sqrtS * (Real(1.3566) - invTk * Real(160.340)) -
95 S * (Real(0.05778) - invTk * Real(0.37335)));
96 const Real K3p = std::exp(-Real(18.141) -
97 invTk * Real(3070.75) +
98 sqrtS * (Real(2.81197) + invTk * Real(17.27039)) -
99 S * (Real(0.09984) + invTk * Real(44.99486)));
100
101 // Compute dissociation constant of silica, Ksi=[H][SiO(OH)3]/[Si(OH)4].
102 // From Millero (1995; page 671) using data from Yao and Millero (1995).
103 const Real Ksi = std::exp(Real(117.385) -
104 invTk * Real(8904.2) -
105 logTk * Real(19.334) +
106 sqrtSO4 * (Real(3.5913) - invTk * Real(458.79)) -
107 SO4 * (Real(1.5998) - invTk * Real(188.74) -
108 SO4 * (Real(0.07871) - invTk * Real(12.1652))) +
109 std::log(Real(1.0) - Real(0.001005) * S));
110
111 // Compute ion product of whater, Kw = [H][OH].
112 // From Millero (1995; page 670) using composite data.
113 const Real Kw = std::exp(Real(148.9652) -
114 invTk * Real(13847.26) -
115 logTk * Real(23.6521) -
116 sqrtS * (Real(5.977) - invTk * Real(118.67) -
117 logTk * Real(1.0495)) -
118 S * Real(0.01615));
119
120 // Compute salinity constant of hydrogen sulfate, Ks = [H][SO4]/[HSO4].
121 // From Dickson (1990, J. chem. Thermodynamics 22, 113).
122 const Real Ks = std::exp(Real(141.328) -
123 invTk * Real(4276.1) -
124 logTk * Real(23.093) +
125 sqrtSO4 * (Real(324.57) - invTk * Real(13856.0) -
126 logTk * Real(47.986) - SO4 * invTk * Real(2698.0)) -
127 SO4 * (Real(771.54) - invTk * Real(35474.0) -
128 logTk * Real(114.723) - SO4 * invTk * Real(1776.0)) +
129 std::log(Real(1.0) - Real(0.001005) * S));
130
131 // Compute stability constant of hydrogen fluorid, Kf = [H][F]/[HF].
132 // From Dickson and Riley (1979) -- change pH scale to total.
133 const Real Kf = std::exp(-Real(12.641) +
134 invTk * Real(1590.2) +
135 sqrtSO4 * Real(1.525) +
136 std::log(Real(1.0) - Real(0.001005) * S) +
137 std::log(Real(1.0) + Real(0.1400) * scl / (Real(96.062) * Ks)));
138
139 // Calculate concentrations for borate (Uppstrom, 1974), sulfate (Morris
140 // and Riley, 1966), and fluoride (Riley, 1965).
141 const Real borate = Real(0.000232) * scl / Real(10.811);
142 const Real sulfate = Real(0.14) * scl / Real(96.062);
143 const Real fluoride = Real(0.000067) * scl / Real(18.9984);
144
145 // Bracket and bisection method.
146 Real X_lo = std::pow(Real(10.0), -Real(10.0));
147 Real X_hi = std::pow(Real(10.0), -Real(5.0));
148 Real X_mid = Real(0.5) * (X_lo + X_hi);
149 Real X = X_mid;
150 const Real K12 = K1 * K2;
151 const Real K12p = K1p * K2p;
152 const Real K123p = K12p * K3p;
153 const Real invKb = Real(1.0) / Kb;
154 const Real invKs = Real(1.0) / Ks;
155 const Real invKsi = Real(1.0) / Ksi;
156 Real fni1 = zero;
157 Real fni3 = zero;
158
159 for (int Ibrack = 0; Ibrack < IbrackMax; ++Ibrack) {
160 for (int Hstep = 1; Hstep <= 3; ++Hstep) {
161 if (Hstep == 1) { X = X_hi; }
162 if (Hstep == 2) { X = X_lo; }
163 if (Hstep == 3) { X = X_mid; }
164
165 // Set some common combinations of parameters used in the iterative
166 // [H+] solver.
167 const Real X2 = X * X;
168 const Real X3 = X2 * X;
169 const Real invX = Real(1.0) / X;
170 const Real A = X * (K12p + X * (K1p + X)) + K123p;
171 const Real B = X * (K1 + X) + K12;
172 const Real C = Real(1.0) / (Real(1.0) + sulfate * invKs);
173
174 // Evaluate f([H+]) for bracketing and mid-value cases.
175 const Real fni = dic * (K1 * X + Real(2.0) * K12) / B +
176 borate / (Real(1.0) + X * invKb) +
177 Kw * invX +
178 phos * (K12p * X + Real(2.0) * K123p - X3) / A +
179 sili / (Real(1.0) + X * invKsi) -
180 X * C -
181 sulfate / (Real(1.0) + Ks * invX * C) -
182 fluoride / (Real(1.0) + Kf * invX) - alk;
183 if (Hstep == 1) { fni1 = fni; }
184 if (Hstep == 3) { fni3 = fni; }
185 }
186
187 // Now, bracket solution within two of three.
188 if (fni3 == zero) {
189 break;
190 }
191 const Real ftest = fni1 / fni3;
192 if (ftest > zero) {
193 X_hi = X_mid;
194 } else {
195 X_lo = X_mid;
196 }
197 X_mid = Real(0.5) * (X_lo + X_hi);
198 }
199
200 // Last iteration gives value.
201 X = X_mid;
202
203 // Determine pCO2. Total Hydrogen ion concentration, Htotal = [H+].
204 const Real Htotal = X;
205 const Real Htotal2 = Htotal * Htotal;
206
207 // Calculate [CO2*] (mole/m3) as defined in DOE Methods Handbook 1994
208 // Version 2, ORNL/CDIAC-74, Dickson and Goyet, Eds. (Chapter 2,
209 // page 10, Eq A.49).
210 const Real CO2star = dic * Htotal2 / (Htotal2 + K1 * Htotal + K1 * K2);
211
212 // Add output argument for storing pCO2surf.
213 return CO2star * Real(1000000.0) / ff;
214}
