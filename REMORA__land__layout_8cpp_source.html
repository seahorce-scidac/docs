33/**
34 * Fill mskr, a rho-point mask on any layout of level lev, with the mask set_masks would give
35 * a level on that layout, before such a level exists. On level 0 every ghost cell the level's
36 * own mask carries is set, as set_masks sets it; on a finer level only the valid cells are,
37 * from the parent's mask (piecewise constant, as set_masks injects it), which is all a weight
38 * needs.
39 *
40 * @param[in ] lev level the layout belongs to
41 * @param[inout] mskr mask to fill, on 2D slab boxes, initially anything
42 * @return false when the mask cannot be known before the level is made: level 0 with
43 * remora.hires_grid_level >= 0 at initialization, whose mask is coarsened from
44 * full-domain data that init_only builds
45 */
46bool
47REMORA::probe_rho_mask (int lev, MultiFab& mskr)
48{
49 mskr.setVal(one);
50 if (solverChoice.mask_type == MaskType::none) {
51 return true;
52 }
53
54 if (lev <= hires_grid_level) {
55 if (lev >= vec_mskr_full_domain.size() || !vec_mskr_full_domain[lev]) {
56 return false;
57 }
58 // As set_masks_averaged_down
59 mskr.ParallelCopy(*vec_mskr_full_domain[lev], 0, 0, 1,
60 vec_mskr_full_domain[lev]->nGrowVect(), mskr.nGrowVect());
61 mskr.FillBoundary(geom[lev].periodicity());
62 return true;
63 }
64
65 if (lev == 0) {
66 if (solverChoice.mask_type == MaskType::analytic) {
67 prob->init_analytic_masks(lev, geom[lev], solverChoice, *this, mskr);
68 mskr.FillBoundary(geom[lev].periodicity());
69 return true;
70 }
71#ifdef REMORA_USE_NETCDF
72 fill_rho_mask_from_netcdf(lev, mskr);
73 return true;
74#else
75 return false;
76#endif
77 }
78
79 // A finer level's valid cells take the parent's mask, piecewise constant
80 AMREX_ALWAYS_ASSERT(lev-1 < vec_mskr.size() && vec_mskr[lev-1]);
81 const IntVect rr = refRatio(lev-1);
82 MultiFab ctmp(amrex::coarsen(mskr.boxArray(), rr), mskr.DistributionMap(), 1, 0);
83 ctmp.setVal(one);
84 ctmp.ParallelCopy(*vec_mskr[lev-1], 0, 0, 1);
85 const int rx = rr[0], ry = rr[1];
86#ifdef AMREX_USE_OMP
87#pragma omp parallel if (Gpu::notInLaunchRegion())
88#endif
89 for (MFIter mfi(mskr, TilingIfNotGPU()); mfi.isValid(); ++mfi) {
90 Array4< Real> const& f = mskr.array(mfi);
91 Array4<const Real> const& c = ctmp.const_array(mfi);
92 ParallelFor(mfi.tilebox(), [=] AMREX_GPU_DEVICE (int i, int j, int k) noexcept
93 {
94 f(i,j,k) = c(amrex::coarsen(i, rx), amrex::coarsen(j, ry), 0);
95 });
96 }
97 Gpu::streamSynchronize();
98 return true;
99}

101/**
102 * Wet rho cells in each box of a prospective layout of level lev: the box's wet columns times
103 * its vertical extent. The same on every rank.
104 *
105 * @param[in ] lev level the layout belongs to
106 * @param[in ] ba the layout's (3D) boxes
107 * @param[ out] wet wet cells per box of ba
108 * @return false when the mask cannot be known yet (see probe_rho_mask)
109 */
110bool
111REMORA::wet_cells_per_box (int lev, const BoxArray& ba, Vector<Long>& wet)
112{
113 const BoxArray ba2d = slab_boxes(ba);
114 MultiFab mskr(ba2d, DistributionMapping(ba2d), 1, IntVect(NGROW+1,NGROW+1,0));
115 if (!probe_rho_mask(lev, mskr)) { return false; }
116
117 wet.assign(ba.size(), 0);
118#ifdef AMREX_USE_OMP
119#pragma omp parallel if (Gpu::notInLaunchRegion())
120#endif
121 for (MFIter mfi(mskr); mfi.isValid(); ++mfi) {
122 // The mask is exactly 0 or 1, so the sum counts wet columns exactly
123 const Real ncol = mskr[mfi].sum<RunOn::Device>(mfi.validbox(), 0);
124 wet[mfi.index()] = static_cast<Long>(ncol) * ba[mfi.index()].length(2);
125 }
126 ParallelAllReduce::Sum(wet.data(), static_cast<int>(wet.size()),
127 ParallelContext::CommunicatorSub());
128 return true;
129}

131/**
132 * The distribution mapping of a new BoxArray on level lev (overrides AmrMesh). Called by AMReX
133 * for every new layout -- initialization, regrid, a level made from coarse -- and by the
134 * checkpoint reader. remora.load_balance picks the strategy; each box weighs its wet cells
135 * plus remora.load_balance_land_weight times its land cells. Where every box weighs its cell
136 * count (no land, or a land weight of one) the SFC strategy is AMReX's own mapping, exactly as
137 * remora.load_balance = amrex gives it. Ownership never changes the answer.
138 *
139 * @param[in ] lev level of the BoxArray
140 * @param[in ] ba the BoxArray to distribute
141 */
142DistributionMapping
143REMORA::MakeDistributionMap (int lev, BoxArray const& ba)
144{
145 BL_PROFILE("REMORA::MakeDistributionMap()");
146 if (load_balance == LoadBalance::amrex) {
147 return AmrCore::MakeDistributionMap(lev, ba);
148 }
149
150 // A land cell weighing as much as a wet one makes every weight the cell count, so the mask
151 // is not needed (nor read from a grid file again).
152 Vector<Long> wet;
153 const bool known = (load_balance_land_weight != one) && wet_cells_per_box(lev, ba, wet);
154
155 // Weights in units of 1/scale of a wet cell, so a fractional land weight keeps its value
156 // to 1/scale (2^20, about 1e-6); a box of 1e7 cells weighs 1e13 at most, far inside Long.
157 constexpr Long scale = Long(1) << 20;
158 const Long land_w = std::llround(Real(scale) * load_balance_land_weight);
159 std::vector<Long> wgts(ba.size());
160 Long total_cells = 0, total_wet = 0;
161 bool cell_count_weights = true;
162 for (int b = 0; b < ba.size(); ++b) {
163 const Long npts = ba[b].numPts();
164 const Long nwet = known ? wet[b] : npts;
165 wgts[b] = scale * nwet + land_w * (npts - nwet);
166 cell_count_weights = cell_count_weights && (wgts[b] == scale * npts);
167 total_cells += npts;
168 total_wet += nwet;
169 }
170
171 if (!known && load_balance_land_weight != one) {
172 amrex::Print() << "Level " << lev << ": the land mask is not known before the level is"
173 << " made (remora.hires_grid_level >= " << lev << "), so its boxes are"
174 << " weighted by cell count for remora.load_balance.\n";
175 }
176
177 if (cell_count_weights && load_balance == LoadBalance::sfc) {
178 return AmrCore::MakeDistributionMap(lev, ba);
179 }
180
181 const int nprocs = ParallelContext::NProcsSub();
182 DistributionMapping dm;
183 if (load_balance == LoadBalance::sfc) {
184 dm.SFCProcessorMap(ba, wgts, nprocs);
185 } else if (load_balance == LoadBalance::knapsack) {
186 dm.KnapSackProcessorMap(wgts, nprocs);
187 } else {
188 dm.RoundRobinProcessorMap(wgts, nprocs);
189 }
190
191 // How even the result is, by the weights balanced and by cell count
192 std::vector<Long> rank_w(nprocs, 0), rank_c(nprocs, 0), rank_n(nprocs, 0);
193 for (int b = 0; b < ba.size(); ++b) {
194 const int r = ParallelContext::global_to_local_rank(dm[b]);
195 rank_w[r] += wgts[b];
196 rank_c[r] += ba[b].numPts();
197 rank_n[r] += 1;
198 }
199 auto max_over_mean = [nprocs] (const std::vector<Long>& v) {
200 Long tot = 0, mx = 0;
201 for (auto x : v) { tot += x; mx = std::max(mx, x); }
202 return (tot > 0) ? Real(mx) * nprocs / Real(tot) : one;
203 };
204 amrex::Print() << "Level " << lev << " load balance (" << load_balance_name()
205 << ", land weight " << load_balance_land_weight << "): " << ba.size()
206 << " boxes, " << total_wet << " of " << total_cells
207 << (known ? " cells wet;" : " cells (mask unknown, all weighed);")
208 << " max/mean rank load " << max_over_mean(rank_w)
209 << " by weight, " << max_over_mean(rank_c) << " by cell count; boxes per"
210 << " rank " << *std::min_element(rank_n.begin(), rank_n.end()) << " to "
211 << *std::max_element(rank_n.begin(), rank_n.end()) << "\n";
212 return dm;
213}

230/**
231 * The level-0 cells whose boxes a finer level needs kept: every refinement indicator's region,
232 * in level-0 index space, as AMReX's grid generation sees it. AMReX buffers the tagged cells by
233 * amr.n_error_buf, rounds them to blocks of amr.blocking_factor (of level 1) over the refinement
234 * ratio, and clears every tag within amr.n_proper such blocks of a cell no level-0 box covers.
235 * So a dropped box that close to an indicator would clip the fine grids the full-domain run
236 * makes. The region is therefore the indicator's cells plus the buffer, rounded out to whole
237 * blocks, grown by n_proper + 1 blocks, and then by the coarse stencil of the coarse-fine fills
238 * and the flux register. A box touching it is never dropped, so the fine grids, and the coarse
239 * data under and around them, are those of the full-domain run. Level 1 only: drop_dry_level0_boxes
240 * drops nothing when amr.max_level > 1, since a level-2 indicator grows level 1 by its own
241 * buffer and proper nesting, which this does not follow.
242 *
243 * @param[ out] unbounded true when an indicator has no region, so a finer level may go anywhere
244 */
245BoxList
246REMORA::level0_refine_footprint (bool& unbounded) const
247{
248 BoxList bl;
249 unbounded = false;
250 if (max_level < 1 || ref_tags.empty()) { return bl; }
251 unbounded = refine_region_unbounded;
252 if (unbounded) { return bl; }
253
254 const IntVect rr = ref_ratio[0];
255 IntVect bf(1), stencil(0);
256 for (int d = 0; d < 2; ++d) {
257 bf[d] = std::max(1, blocking_factor[1][d] / rr[d]);
258 stencil[d] = (NGROW + 2 + rr[d] - 1) / rr[d] + 1;
259 }
260 const Box& dom = geom[0].Domain();
261 const Real* plo = geom[0].ProbLo();
262 const Real* dx = geom[0].CellSize();
263 for (const auto& rb : refine_regions) {
264 IntVect lo = dom.smallEnd(), hi = dom.bigEnd();
265 for (int d = 0; d < 2; ++d) {
266 lo[d] = static_cast<int>(std::floor((rb.lo(d) - plo[d]) / dx[d])) - n_error_buf[0][d];
267 hi[d] = static_cast<int>(std::floor((rb.hi(d) - plo[d]) / dx[d])) + n_error_buf[0][d];
268 }
269 Box b(lo, hi);
270 // Whole blocks, then n_proper + 1 more of them, then the coarse stencil
271 b.coarsen(bf);
272 b.grow(IntVect(n_proper + 1, n_proper + 1, 0));
273 b.refine(bf);
274 b.grow(stencil);
275 b &= dom;
276 if (b.ok()) { bl.push_back(b); }
277 }
278 return bl;
279}

281/**
282 * The boxes of a level-0 BoxArray that remora.drop_dry_boxes keeps: every box with a wet rho
283 * cell in its footprint -- its valid cells and every ghost cell the mask carries, up to one
284 * cell beyond a non-periodic domain edge -- and every
285 * box a finer level may need (level0_refine_footprint). The rest are land over all of that, so
286 * no wet cell's stencil reaches them; partly wet boxes are kept whole.
287 *
288 * @param[in ] ba candidate level-0 BoxArray
289 * @return the kept boxes, in the order of ba; ba itself when nothing is dropped
290 */
291BoxArray
292REMORA::drop_dry_level0_boxes (const BoxArray& ba)
293{
294 BL_PROFILE("REMORA::drop_dry_level0_boxes()");
295 if (solverChoice.mask_type == MaskType::none) {
296 amrex::Print() << "remora.drop_dry_boxes: no land mask (remora.mask_type = none), so no"
297 << " box is dropped.\n";
298 return ba;
299 }
300
301#ifdef REMORA_USE_PARTICLES
302 constexpr bool with_particles = true;
303#else
304 constexpr bool with_particles = false;
305#endif
306#ifdef REMORA_USE_MOAB
307 constexpr bool with_moab = true;
308#else
309 constexpr bool with_moab = false;
310#endif
311 if (with_particles) {
312 // Tracer particles are seeded over the whole domain, and a particle in a dropped box
313 // would belong to no box. Untested, so not allowed.
314 amrex::Print() << "remora.drop_dry_boxes: built with particles, so no level-0 box is"
315 << " dropped.\n";
316 return ba;
317 }
318 if (with_moab) {
319 // InitMOABMesh builds its mesh from level 0 and takes it to cover the domain
320 amrex::Print() << "remora.drop_dry_boxes: built with MOAB, so no level-0 box is dropped.\n";
321 return ba;
322 }
323 if (max_level > 1) {
324 // A level-2 indicator makes level 1 grow by its own buffer and proper nesting, and
325 // AMReX would clear the level-1 tags near a hole without a word; the footprint below
326 // covers level 1 only.
327 amrex::Print() << "remora.drop_dry_boxes: amr.max_level > 1, so no level-0 box is"
328 << " dropped.\n";
329 return ba;
330 }
331 if (running_with_coupling_driver) {
332 // The driver builds its remap from level 0's layout, which it takes to cover the domain
333 amrex::Print() << "remora.drop_dry_boxes: coupled to an atmosphere driver, so no level-0"
334 << " box is dropped.\n";
335 return ba;
336 }
337
338 bool unbounded = false;
339 const BoxList protect = level0_refine_footprint(unbounded);
340 if (unbounded) {
341 amrex::Print() << "remora.drop_dry_boxes: a refinement indicator has no box, so a finer"
342 << " level may cover any land; no level-0 box is dropped.\n";
343 return ba;
344 }
345
346 const BoxArray ba2d = slab_boxes(ba);
347 MultiFab mskr(ba2d, DistributionMapping(ba2d), 1, IntVect(NGROW+1,NGROW+1,0));
348 if (!probe_rho_mask(0, mskr)) {
349 amrex::Print() << "remora.drop_dry_boxes: the level-0 mask comes from remora.hires_grid_level"
350 << " and is not known before the level is made; no box is dropped.\n";
351 return ba;
352 }
353
354 // The footprint stops one cell beyond a non-periodic domain edge: a grid file holds the
355 // mask there and no further, and the rings beyond it are physical-boundary ghost cells,
356 // filled from the inside rather than exchanged, so they keep the all-water placeholder.
357 Box reach = makeSlab(geom[0].Domain(), 2, 0);
358 for (int d = 0; d < 2; ++d) {
359 reach.grow(d, geom[0].isPeriodic(d) ? NGROW+1 : 1);
360 }
361 Vector<int> keep(ba.size(), 0);
362#ifdef AMREX_USE_OMP
363#pragma omp parallel if (Gpu::notInLaunchRegion())
364#endif
365 for (MFIter mfi(mskr); mfi.isValid(); ++mfi) {
366 const Real wet = mskr[mfi].max<RunOn::Device>(mfi.fabbox() & reach, 0);
367 keep[mfi.index()] = (wet > zero) ? 1 : 0;
368 }
369 ParallelAllReduce::Max(keep.data(), static_cast<int>(keep.size()),
370 ParallelContext::CommunicatorSub());
371
372 BoxList kept;
373 Long ndrop = 0, cells_drop = 0;
374 for (int b = 0; b < ba.size(); ++b) {
375 bool k = keep[b] != 0;
376 if (!k) {
377 for (Box p : protect) {
378 p.setRange(2,0);
379 if (p.intersects(ba2d[b])) { k = true; break; }
380 }
381 }
382 if (k) {
383 kept.push_back(ba[b]);
384 } else {
385 ++ndrop;
386 cells_drop += ba[b].numPts();
387 }
388 }
389 if (kept.isEmpty()) {
390 amrex::Abort("remora.drop_dry_boxes: every level-0 box is land; there is no water to model");
391 }
392
393 amrex::Print() << "remora.drop_dry_boxes: level 0 keeps " << kept.size() << " of " << ba.size()
394 << " boxes; dropped " << ndrop << " all-land boxes, " << cells_drop << " of "
395 << ba.numPts() << " cells (" << Real(100.0)*Real(cells_drop)/Real(ba.numPts())
396 << "%)\n";
397 if (ndrop == 0) { return ba; }
398 return BoxArray(std::move(kept));
399}

412/**
413 * Abort if level 1, or the coarse cells its fills and averaging use, reaches a dropped level-0
414 * box. level0_refine_footprint keeps those boxes when the run starts; this catches a grid that
415 * outgrew that footprint at a regrid.
416 */
417void
418REMORA::check_dry_box_nesting () const
419{
420 if (finest_level < 1 || !level0_has_holes()) { return; }
421 const IntVect rr = ref_ratio[0];
422 IntVect ng(0);
423 for (int d = 0; d < 2; ++d) { ng[d] = (NGROW + 2 + rr[d] - 1) / rr[d] + 1; }
424 BoxArray need = amrex::coarsen(grids[1], rr);
425 need.grow(ng);
426 BoxList bl;
427 for (int b = 0; b < need.size(); ++b) {
428 Box bx = need[b] & geom[0].Domain();
429 if (bx.ok()) { bl.push_back(bx); }
430 }
431 if (!grids[0].contains(BoxArray(std::move(bl)))) {
432 amrex::Abort("remora.drop_dry_boxes: level 1, or the level-0 cells it needs, reaches a"
433 " dropped level-0 box. Set remora.drop_dry_boxes = false for this run.");
434 }
435}

447/**
448 * Level 0's mask, as set_masks built it on the kept boxes, has to be the mask the boxes were
449 * dropped on; otherwise the decision rested on a different coastline. Compared everywhere the
450 * mask is stored, ghost cells included.
451 */
452void
453REMORA::check_dropped_level0_mask ()
454{
455 const MultiFab& mskr = *vec_mskr[0];
456 MultiFab probe(mskr.boxArray(), mskr.DistributionMap(), 1, mskr.nGrowVect());
457 AMREX_ALWAYS_ASSERT(probe_rho_mask(0, probe));
458 MultiFab::Subtract(probe, mskr, 0, 0, 1, mskr.nGrowVect());
459 if (probe.norminf(0, 1, mskr.nGrowVect()) != zero) {
460 amrex::Abort("remora.drop_dry_boxes: level 0's mask differs from the one the boxes were"
461 " dropped on");
462 }
463}

528 * Write val into every cell of mf, a field on the level-0 layout, that lies inside the domain
529 * but in no box: the ghost cells of kept boxes inside dropped ones, which no exchange fills. No
530 * op when level 0 covers the domain.
531 *
532 * @return the number of values written on this rank
533 */
534Long
535REMORA::set_dry_ghost_cells (MultiFab& mf, Real val)
536{
537 if (!level0_has_holes()) { return 0; }
538 const BoxArray& ba = mf.boxArray();
539 const IntVect ng = mf.nGrowVect();
540 auto& cache = dry_ghost_cache();
541 DryGhostHoles* entry = nullptr;
542 for (auto& e : cache) {
543 if (e.ng == ng && e.ba == ba) { entry = &e; break; }
544 }
545 if (!entry) {
546 cache.push_back(DryGhostHoles{ba, ng, {}});
547 entry = &cache.back();
548 }
549 // The domain in the field's index type; a 2D field lives on slabs at k = 0
550 Box dom = amrex::convert(geom[0].Domain(), mf.ixType());
551 const Box mb = ba.minimalBox();
552 dom.setRange(2, mb.smallEnd(2), mb.length(2));
553
554 // The hole boxes of every local box first, serially, since this fills the cache
555 for (MFIter mfi(mf); mfi.isValid(); ++mfi) {
556 if (entry->holes.count(mfi.index()) == 0) {
557 Vector<Box> hv;
558 const Box fab = mfi.fabbox() & dom;
559 if (fab.ok()) {
560 for (const Box& h : ba.complementIn(fab)) { hv.push_back(h); }
561 }
562 entry->holes.emplace(mfi.index(), std::move(hv));
563 }
564 }
565
566 Long n = 0;
567 const int nc = mf.nComp();
568 const auto& holes = entry->holes;
569#ifdef AMREX_USE_OMP
570#pragma omp parallel if (Gpu::notInLaunchRegion()) reduction(+:n)
571#endif
572 for (MFIter mfi(mf); mfi.isValid(); ++mfi) {
573 Array4<Real> const& a = mf.array(mfi);
574 for (const Box& h : holes.at(mfi.index())) {
575 n += h.numPts() * nc;
576 ParallelFor(h, nc, [=] AMREX_GPU_DEVICE (int i, int j, int k, int c) noexcept
577 {
578 a(i,j,k,c) = val;
579 });
580 }
581 }
582 Gpu::streamSynchronize();
583 return n;
584}

586/**
587 * With level-0 boxes dropped, the ghost cells of a kept box that lie in a dropped box are never
588 * filled by an exchange, and the land stencils of the kept box read them. What those stencils
589 * compute stays on land only while the values there are finite: a wet cell is at least NGROW+2
590 * land cells away and the masks multiply the land values out (the regression tests write 7
591 * there and compare), but 0 * NaN is NaN, so a NaN there spreads into the water within a few
592 * steps. A fresh FArrayBox holds NaN in a debug build (amrex.init_snan), so these cells are
593 * zeroed in every level-0 field as soon as the fields exist, before anything reads them.
594 * Fields made later on the level-0 layout (lon/lat/angle from the grid file, the rho-point
595 * surface stress) zero theirs when they are allocated.
596 *
597 * The scratch MultiFabs the step allocates on the level-0 layout (for example in setup_step,
598 * gls, bulk_fluxes and the prestep) are not zeroed there. Nothing that the step reads from
599 * them in a dropped region reaches a kept cell's result: the Debug CI build (amrex.init_snan)
600 * and the dropped-vs-full feature checks with signalling NaNs and the invalid, zero and
601 * overflow traps on all complete with bitwise water points. A new kernel that reads such
602 * scratch in its ghost cells and lets the value reach a wet cell would break that; zero the
603 * scratch with set_dry_ghost_cells then.
604 */
605void
606REMORA::zero_level0_dry_ghost_cells ()
607{
608 if (!level0_has_holes()) { return; }
609 for (MultiFab* mf : level0_fields(true)) {
610 set_dry_ghost_cells(*mf, zero);
611 }
612}

614/**
615 * remora.dry_ghost_test (debug). After initialization, overwrite the ghost cells of the kept
616 * level-0 boxes that lie in dropped boxes, in every evolving field (remora.dry_ghost_fields =
617 * dynamic) or every field (all): 1 writes a signalling NaN, which with amrex.fpe_trap_invalid
618 * = 1 stops the run at the first arithmetic read of one; 2 writes remora.dry_ghost_value
619 * (default 1e30), so a plotted field that still matches the full-domain run does not depend on
620 * those cells while they hold that finite value.
621 */
622void
623REMORA::poison_dry_ghost_cells ()
624{
625 if (dry_ghost_test == 0 || !level0_has_holes()) { return; }
626 std::string which = "dynamic";
627 ParmParse pp("remora");
628 pp.query("dry_ghost_fields", which);
629 if (which != "dynamic" && which != "all") {
630 amrex::Abort("remora.dry_ghost_fields must be dynamic or all, not " + which);
631 }
632 Real big = Real(1.e30);
633 pp.query("dry_ghost_value", big);
634 const Real val = (dry_ghost_test == 1) ? std::numeric_limits<Real>::signaling_NaN() : big;
635 const Vector<MultiFab*> f = level0_fields(which == "all");
636 Long npoison = 0;
637 for (MultiFab* mf : f) {
638 npoison += set_dry_ghost_cells(*mf, val);
639 }
640 ParallelDescriptor::ReduceLongSum(npoison);
641 amrex::Print() << "remora.dry_ghost_test = " << dry_ghost_test << ": overwrote " << npoison
642 << " uncovered ghost values in " << f.size() << " level-0 fields\n";
643}
