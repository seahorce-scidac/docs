74/**
75 * Whether each species moves with the flow on level a_lev, and whether any does. One with
76 * no particles there needs no flow fields: particles are seeded on level 0 and usually stay
77 * there, so this skips them on the finer levels. The count is a global reduction, so every
78 * rank agrees.
79 */
80static bool advecting_species (const Vector<REMORAPC*>& a_species, int a_lev,
81 Vector<int>& a_advect)
82{
83 a_advect.resize(a_species.size());
84 bool any_advect = false;
85 for (int n = 0; n < a_species.size(); ++n) {
86 a_advect[n] = a_species[n]->advectsWithFlow()
87 && a_species[n]->NumberOfParticlesAtLevel(a_lev) > 0;
88 any_advect = any_advect || a_advect[n];
89 }
90 return any_advect;
91}

93/**
94 * Particles are advected in terrain-following coordinates, so they need omega, the velocity
95 * across the w-levels, rather than a Cartesian vertical velocity. As in ROMS floats, it comes
96 * from the time-averaged fluxes update_massflux_3d leaves, the ones the tracers move with,
97 * rather than the diagnostic omega from the current velocities. advance_3d integrates those
98 * fluxes into W, the same volume flux compute_omega would build, its first ghost column
99 * included, so this keeps W scaled by pm*pn instead of integrating the columns again. The
100 * scaling and fill are those of compute_omega, so the result is the same to the bit.
101 *
102 * @param[in] lev level
103 * @param[in] mf_W advance_3d's W, on the valid region and the first ghost column
104 */
105void REMORA::store_particle_omega (int lev, const MultiFab& mf_W)
106{
107 if (!m_use_tracer_particles && !m_use_hydro_particles) { return; }
108
109 if (int(particle_omega_stored.size()) <= lev) {
110 particle_omega_stored.resize(lev+1, 0);
111 particle_advect.resize(lev+1);
112 }
113 const Vector<REMORAPC*> species = particle_species(particleData, m_use_tracer_particles,
114 m_use_hydro_particles);
115 particle_omega_stored[lev] = 1;
116 if (!advecting_species(species, lev, particle_advect[lev])) { return; }
117
118 if (int(particle_omega.size()) <= lev) { particle_omega.resize(lev+1); }
119 const BoxArray ba = convert(grids[lev],IntVect(0,0,1));
120 const IntVect ng(NGROW,NGROW,0);
121 auto& mf = particle_omega[lev];
122 if (!mf || !mf->boxArray().CellEqual(ba) || mf->boxArray().ixType() != ba.ixType()
123 || mf->DistributionMap() != dmap[lev] || mf->nGrowVect() != ng) {
124 mf = std::make_unique<MultiFab>(ba, dmap[lev], 1, ng);
125 }
126 MultiFab& mf_omega = *mf;
127 const int N = Geom(lev).Domain().size()[2]-1;
128
129 mf_omega.setVal(zero);
130#ifdef _OPENMP
131#pragma omp parallel if (Gpu::notInLaunchRegion())
132#endif
133 for (MFIter mfi(mf_omega, TilingIfNotGPU()); mfi.isValid(); ++mfi)
134 {
135 Array4<Real > const& omega = mf_omega.array(mfi);
136 Array4<Real const> const& W = mf_W.const_array(mfi);
137 Array4<Real const> const& pm = vec_pm[lev]->const_array(mfi);
138 Array4<Real const> const& pn = vec_pn[lev]->const_array(mfi);
139
140 // The tile, with the first ghost column where it touches the edge of the grid
141 const Box bx2d = makeSlab(mfi.growntilebox(IntVect(1,1,0)), 2, 0);
142 ParallelFor(bx2d, N, [=] AMREX_GPU_DEVICE (int i, int j, int, int kk)
143 {
144 omega(i,j,kk+1) = W(i,j,kk+1) * pm(i,j,0) * pn(i,j,0);
145 });
146 }
147
148 mf_omega.FillBoundary(geom[lev].periodicity());
149}

151/*! Evolve tracers and hydro particles for one time step*/
152void REMORA::evolveTracers (int a_lev, Real a_dt_lev)
153{
154 if (!m_use_tracer_particles && !m_use_hydro_particles) { return; }
155
156 const Vector<REMORAPC*> species = particle_species(particleData, m_use_tracer_particles,
157 m_use_hydro_particles);
158
159 // Normally advance_3d has just stored omega and the species flags (store_particle_omega).
160 // Otherwise compute them here.
161 const bool stored = a_lev < int(particle_omega_stored.size()) && particle_omega_stored[a_lev];
162 Vector<int> advect_here;
163 bool any_advect;
164 if (stored) {
165 advect_here = particle_advect[a_lev];
166 any_advect = false;
167 for (int a : advect_here) { any_advect = any_advect || a; }
168 particle_omega_stored[a_lev] = 0;
169 } else {
170 any_advect = advecting_species(species, a_lev, advect_here);
171 }
172
173 if (any_advect) {
174 // Kept between steps rather than allocated on each one, and redefined when the
175 // level's grids change. store_particle_omega (or compute_omega) and the fills below
176 // overwrite them entirely.
177 if (int(particle_omega.size()) <= a_lev) { particle_omega.resize(a_lev+1); }
178 if (int(particle_u.size()) <= a_lev) {
179 particle_u.resize(a_lev+1);
180 particle_v.resize(a_lev+1);
181 }
182 auto fit = [&] (std::unique_ptr<MultiFab>& mf, const BoxArray& ba, const IntVect& ng)
183 {
184 if (!mf || !mf->boxArray().CellEqual(ba) || mf->boxArray().ixType() != ba.ixType()
185 || mf->DistributionMap() != dmap[a_lev] || mf->nGrowVect() != ng) {
186 mf = std::make_unique<MultiFab>(ba, dmap[a_lev], 1, ng);
187 }
188 };
189 if (!stored) {
190 fit(particle_omega[a_lev], convert(grids[a_lev],IntVect(0,0,1)), IntVect(NGROW,NGROW,0));
191 compute_omega(a_lev, *particle_omega[a_lev], *vec_Huon[a_lev], *vec_Hvom[a_lev]);
192 }
193 fit(particle_u[a_lev], xvel_new[a_lev]->boxArray(), xvel_new[a_lev]->nGrowVect());
194 fit(particle_v[a_lev], yvel_new[a_lev]->boxArray(), yvel_new[a_lev]->nGrowVect());
195 MultiFab& mf_u = *particle_u[a_lev];
196 MultiFab& mf_v = *particle_v[a_lev];
197
198 // A particle within half a cell of a wall or a coarse/fine boundary interpolates u and
199 // v from ghost cells, which so far this step have only been filled across grids and
200 // periodic boundaries. Fill copies with the physical and coarse/fine boundary
201 // conditions at the end of the step. The model's own velocity ghosts are left alone:
202 // refilling them with the physical boundary conditions here changes the answer on
203 // open boundaries (see timeStepML).
204 //
205 // As at initialization and regrid, each copy is its own "calc" field, with no time
206 // step: the NetCDF radiation conditions read the calc field unguarded. With calc
207 // equal to the field, a radiation (orlanski_rad*) boundary leaves the ghost as it
208 // was, so there the particles still see the start-of-step ghost value.
209 MultiFab::Copy(mf_u, *xvel_new[a_lev], 0, 0, 1, mf_u.nGrowVect());
210 MultiFab::Copy(mf_v, *yvel_new[a_lev], 0, 0, 1, mf_v.nGrowVect());
211 FillPatch(a_lev, t_new[a_lev], mf_u, xvel_new, xvel_bc(), BdyVars::u, 0, true, false,
212 0, 0, zero, mf_u, xvel_old, xvel_new);
213 FillPatch(a_lev, t_new[a_lev], mf_v, yvel_new, yvel_bc(), BdyVars::v, 0, true, false,
214 0, 0, zero, mf_v, yvel_old, yvel_new);
215 }
216
217 for (int n = 0; n < species.size(); ++n) {
218 if (advect_here[n]) {
219 Vector<MultiFab const*> flowvel = {particle_u[a_lev].get(), particle_v[a_lev].get(),
220 particle_omega[a_lev].get()};
221 species[n]->EvolveParticles(a_lev, a_dt_lev, flowvel, vec_z_phys_nd,
222 *vec_pm[a_lev], *vec_pn[a_lev]);
223 } else {
224 // Nothing to advect on this level. A species that does not advect still keeps
225 // its level index, so its heights follow the levels; then redistribute, as
226 // EvolveParticles does.
227 if (!species[n]->advectsWithFlow()) {
228 species[n]->SetHeightsFromLevelIndex(a_lev, a_dt_lev, vec_z_phys_nd[a_lev]);
229 }
230 species[n]->Redistribute();
231 }
232 }
233 return;
234}
