65/*! Evolve tracers and hydro particles for one time step*/
66void REMORA::evolveTracers (int a_lev, Real a_dt_lev)
67{
68 if (!m_use_tracer_particles && !m_use_hydro_particles) { return; }
69
70 Vector<REMORAPC*> species;
71 if (m_use_tracer_particles) { species.push_back(particleData[REMORAParticleNames::tracers]); }
72 if (m_use_hydro_particles) { species.push_back(particleData[REMORAParticleNames::hydro]); }
73
74 // Whether each species moves with the flow on this level. One with no particles here needs
75 // no flow fields: particles are seeded on level 0 and usually stay there, so this skips
76 // them on the finer levels. The count is a global reduction, so every rank agrees.
77 Vector<int> advect_here(species.size());
78 bool any_advect = false;
79 for (int n = 0; n < species.size(); ++n) {
80 advect_here[n] = species[n]->advectsWithFlow()
81 && species[n]->NumberOfParticlesAtLevel(a_lev) > 0;
82 any_advect = any_advect || advect_here[n];
83 }
84
85 // Particles are advected in terrain-following coordinates, so they need omega, the
86 // velocity across the w-levels, rather than a Cartesian vertical velocity. As in ROMS
87 // floats, it comes from the time-averaged fluxes update_massflux_3d leaves, the ones the
88 // tracers move with, rather than the diagnostic omega from the current velocities.
89 if (any_advect) {
90 // Kept between steps rather than allocated on each one, and redefined when the
91 // level's grids change. compute_omega and the fills below overwrite them entirely.
92 if (int(particle_omega.size()) <= a_lev) {
93 particle_omega.resize(a_lev+1);
94 particle_u.resize(a_lev+1);
95 particle_v.resize(a_lev+1);
96 }
97 auto fit = [&] (std::unique_ptr<MultiFab>& mf, const BoxArray& ba, const IntVect& ng)
98 {
99 if (!mf || !mf->boxArray().CellEqual(ba) || mf->boxArray().ixType() != ba.ixType()
100 || mf->DistributionMap() != dmap[a_lev] || mf->nGrowVect() != ng) {
101 mf = std::make_unique<MultiFab>(ba, dmap[a_lev], 1, ng);
102 }
103 };
104 fit(particle_omega[a_lev], convert(grids[a_lev],IntVect(0,0,1)), IntVect(NGROW,NGROW,0));
105 fit(particle_u[a_lev], xvel_new[a_lev]->boxArray(), xvel_new[a_lev]->nGrowVect());
106 fit(particle_v[a_lev], yvel_new[a_lev]->boxArray(), yvel_new[a_lev]->nGrowVect());
107 MultiFab& mf_omega = *particle_omega[a_lev];
108 MultiFab& mf_u = *particle_u[a_lev];
109 MultiFab& mf_v = *particle_v[a_lev];
110
111 compute_omega(a_lev, mf_omega, *vec_Huon[a_lev], *vec_Hvom[a_lev]);
112
113 // A particle within half a cell of a wall or a coarse/fine boundary interpolates u and
114 // v from ghost cells, which so far this step have only been filled across grids and
115 // periodic boundaries. Fill copies with the physical and coarse/fine boundary
116 // conditions at the end of the step. The model's own velocity ghosts are left alone:
117 // refilling them with the physical boundary conditions here changes the answer on
118 // open boundaries (see timeStepML).
119 //
120 // As at initialization and regrid, each copy is its own "calc" field, with no time
121 // step: the NetCDF radiation conditions read the calc field unguarded. With calc
122 // equal to the field, a radiation (orlanski_rad*) boundary leaves the ghost as it
123 // was, so there the particles still see the start-of-step ghost value.
124 MultiFab::Copy(mf_u, *xvel_new[a_lev], 0, 0, 1, mf_u.nGrowVect());
125 MultiFab::Copy(mf_v, *yvel_new[a_lev], 0, 0, 1, mf_v.nGrowVect());
126 FillPatch(a_lev, t_new[a_lev], mf_u, xvel_new, xvel_bc(), BdyVars::u, 0, true, false,
127 0, 0, zero, mf_u, xvel_old, xvel_new);
128 FillPatch(a_lev, t_new[a_lev], mf_v, yvel_new, yvel_bc(), BdyVars::v, 0, true, false,
129 0, 0, zero, mf_v, yvel_old, yvel_new);
130 }
131
132 for (int n = 0; n < species.size(); ++n) {
133 if (advect_here[n]) {
134 Vector<MultiFab const*> flowvel = {particle_u[a_lev].get(), particle_v[a_lev].get(),
135 particle_omega[a_lev].get()};
136 species[n]->EvolveParticles(a_lev, a_dt_lev, flowvel, vec_z_phys_nd,
137 *vec_pm[a_lev], *vec_pn[a_lev]);
138 } else {
139 // Nothing to advect on this level; still redistribute, as EvolveParticles does
140 species[n]->Redistribute();
141 }
142 }
143 return;
144}
