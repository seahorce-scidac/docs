65/*! Evolve tracers and hydro particles for one time step*/
66void REMORA::evolveTracers (int a_lev, Real a_dt_lev)
67{
68 if (!m_use_tracer_particles && !m_use_hydro_particles) { return; }
69
70 // Particles are advected in terrain-following coordinates, so they need omega, the
71 // velocity across the w-levels, rather than a Cartesian vertical velocity. As in ROMS
72 // floats, it comes from the time-averaged fluxes update_massflux_3d leaves, the ones the
73 // tracers move with, rather than the diagnostic omega from the current velocities. It is
74 // only needed if some species moves with the flow (hydro particles do not by default).
75 bool any_advect = false;
76 if (m_use_tracer_particles) {
77 any_advect = any_advect || particleData[REMORAParticleNames::tracers]->advectsWithFlow();
78 }
79 if (m_use_hydro_particles) {
80 any_advect = any_advect || particleData[REMORAParticleNames::hydro]->advectsWithFlow();
81 }
82 MultiFab mf_omega;
83 MultiFab mf_u, mf_v;
84 if (any_advect) {
85 mf_omega.define(convert(grids[a_lev],IntVect(0,0,1)), dmap[a_lev], 1, IntVect(NGROW,NGROW,0));
86 compute_omega(a_lev, mf_omega, *vec_Huon[a_lev], *vec_Hvom[a_lev]);
87
88 // A particle within half a cell of a wall or a coarse/fine boundary interpolates u and
89 // v from ghost cells, which so far this step have only been filled across grids and
90 // periodic boundaries. Fill copies with the physical and coarse/fine boundary
91 // conditions at the end of the step. The model's own velocity ghosts are left alone:
92 // refilling them with the physical boundary conditions here changes the answer on
93 // open boundaries (see timeStepML).
94 //
95 // As at initialization and regrid, each copy is its own "calc" field, with no time
96 // step: the NetCDF radiation conditions read the calc field unguarded. With calc
97 // equal to the field, a radiation (orlanski_rad*) boundary leaves the ghost as it
98 // was, so there the particles still see the start-of-step ghost value.
99 mf_u.define(xvel_new[a_lev]->boxArray(), dmap[a_lev], 1, xvel_new[a_lev]->nGrowVect());
100 mf_v.define(yvel_new[a_lev]->boxArray(), dmap[a_lev], 1, yvel_new[a_lev]->nGrowVect());
101 MultiFab::Copy(mf_u, *xvel_new[a_lev], 0, 0, 1, mf_u.nGrowVect());
102 MultiFab::Copy(mf_v, *yvel_new[a_lev], 0, 0, 1, mf_v.nGrowVect());
103 FillPatch(a_lev, t_new[a_lev], mf_u, xvel_new, xvel_bc(), BdyVars::u, 0, true, false,
104 0, 0, zero, mf_u, xvel_old, xvel_new);
105 FillPatch(a_lev, t_new[a_lev], mf_v, yvel_new, yvel_bc(), BdyVars::v, 0, true, false,
106 0, 0, zero, mf_v, yvel_old, yvel_new);
107 }
108
109 Vector<MultiFab const*> flowvel = {any_advect ? &mf_u : xvel_new[a_lev],
110 any_advect ? &mf_v : yvel_new[a_lev], &mf_omega};
111
112 if (m_use_tracer_particles) {
113 particleData[REMORAParticleNames::tracers]->EvolveParticles( a_lev,
114 a_dt_lev,
115 flowvel,
116 vec_z_phys_nd,
117 *vec_pm[a_lev],
118 *vec_pn[a_lev] );
119 }
120 if (m_use_hydro_particles) {
121 particleData[REMORAParticleNames::hydro]->EvolveParticles( a_lev,
122 a_dt_lev,
123 flowvel,
124 vec_z_phys_nd,
125 *vec_pm[a_lev],
126 *vec_pn[a_lev] );
127 }
128 return;
129}
