47/** Whether a units attribute names degrees east (or north) */
48bool units_say (const ncutils::NCVar& var, const std::string& direction)
49{
50 if (!var.has_attr("units")) { return false; }
51 std::string u = var.get_attr("units");
52 std::transform(u.begin(), u.end(), u.begin(), [] (unsigned char c) { return std::tolower(c); });
53 return u.find("degree") != std::string::npos &&
54 (u.find(direction) != std::string::npos ||
55 u.find(direction == "east" ? "_e" : "_n") != std::string::npos);
56}

77/**
78 * Find and read the coordinate axes of a (time, lat, lon) variable. The names come from
79 * the inputs when given; otherwise from the variable's CF coordinates attribute, each name
80 * identified by its units (degrees east or north) or, failing that, by its leading
81 * "lon"/"lat" or, failing that, by order as ROMS's get_varcoords takes them; otherwise from
82 * the CF coordinate variables of its last two dimensions (longitude(longitude), as ERA5 and
83 * most reanalyses write them). Each axis must be 1D and dimensioned like the variable's
84 * matching dimension. Collective; aborts on every rank with the reason, followed by
85 * `context`, when the axes cannot be found.
86 */
87SourceGrid
88find_source_grid (const std::string& file_name, const std::string& field_name,
89 const std::vector<long long>& field_shape,
90 const LatLonRegridOptions& options, const std::string& context)
91{
92 SourceGrid g;
93 std::string error;
94
95 if (ParallelDescriptor::IOProcessor())
96 {
97 auto ncf = ncutils::NCFile::open_local(file_name);
98 const auto var = ncf.var(field_name);
99 const int rank = var.ndim();
100
101 g.lon_name = options.lon_name;
102 g.lat_name = options.lat_name;
103 if (g.lon_name.empty() || g.lat_name.empty()) {
104 if (!var.has_attr("coordinates")) {
105 if (has_dimension_coordinate(ncf, var, rank-1) &&
106 has_dimension_coordinate(ncf, var, rank-2)) {
107 if (g.lon_name.empty()) { g.lon_name = dim_name(ncf, var, rank-1); }
108 if (g.lat_name.empty()) { g.lat_name = dim_name(ncf, var, rank-2); }
109 } else {
110 error = "it has no coordinates attribute naming its longitude and latitude, "
111 "its last two dimensions have no coordinate variables, and "
112 "remora.frc_lon_varname and remora.frc_lat_varname are not both set";
113 }
114 } else {
115 std::istringstream tokens(var.get_attr("coordinates"));
116 std::vector<std::string> names;
117 for (std::string t; tokens >> t; ) {
118 if (ncf.has_var(t) && ncf.var(t).ndim() >= 1 &&
119 dim_name(ncf, ncf.var(t), 0) != dim_name(ncf, var, 0)) {
120 names.push_back(t); // skip the time coordinate
121 }
122 }
123 std::string by_units_lon, by_units_lat, by_name_lon, by_name_lat;
124 for (const auto& t : names) {
125 if (units_say(ncf.var(t), "east")) { by_units_lon = t; }
126 if (units_say(ncf.var(t), "north")) { by_units_lat = t; }
127 if (t.rfind("lon", 0) == 0) { by_name_lon = t; }
128 if (t.rfind("lat", 0) == 0) { by_name_lat = t; }
129 }
130 std::string lon = !by_units_lon.empty() ? by_units_lon : by_name_lon;
131 std::string lat = !by_units_lat.empty() ? by_units_lat : by_name_lat;
132 if ((lon.empty() || lat.empty()) && names.size() == 2) {
133 lon = names[0];
134 lat = names[1];
135 }
136 if (g.lon_name.empty()) { g.lon_name = lon; }
137 if (g.lat_name.empty()) { g.lat_name = lat; }
138 if (g.lon_name.empty() || g.lat_name.empty()) {
139 error = "its coordinates attribute '" + var.get_attr("coordinates") +
140 "' does not name a longitude and a latitude variable; set "
141 "remora.frc_lon_varname and remora.frc_lat_varname";
142 }
143 }
144 }
145
146 // Each axis 1D, along the variable's last (lon) or next-to-last (lat) dimension
147 const std::pair<std::string, int> axes[2] = {{g.lon_name, rank-1}, {g.lat_name, rank-2}};
148 for (const auto& [name, idim] : axes) {
149 if (!error.empty()) { break; }
150 if (!ncf.has_var(name)) {
151 error = "its coordinate variable '" + name + "' is not in the file";
152 } else if (ncf.var(name).ndim() != 1) {
153 error = "its coordinate variable '" + name + "' is " +
154 std::to_string(ncf.var(name).ndim()) + "D; only 1D (rectilinear) "
155 "longitude and latitude axes are supported";
156 } else if (dim_name(ncf, ncf.var(name), 0) != dim_name(ncf, var, idim) ||
157 ncf.var(name).shape()[0] != field_shape[idim]) {
158 error = "its coordinate variable '" + name + "' does not run along its " +
159 std::string(idim == rank-1 ? "last" : "next-to-last") + " dimension '" +
160 dim_name(ncf, var, idim) + "'; the variable must be (time, lat, lon)";
161 }
162 }
163
164 int has_scale = 0, has_offset = 0;
165 for (const auto& [attr, has, val] : { std::make_tuple("_FillValue", &g.has_fill, &g.fill_value),
166 std::make_tuple("missing_value", &g.has_missing, &g.missing_value),
167 std::make_tuple("scale_factor", &has_scale, &g.scale_factor),
168 std::make_tuple("add_offset", &has_offset, &g.add_offset) })
169 {
170 if (var.has_attr(attr)) {
171 std::vector<double> v;
172 var.get_attr(attr, v);
173 if (!v.empty()) { *has = 1; *val = v[0]; }
174 }
175 }
176 ncf.close();
177 }
178
179 bcast_string(error);
180 if (!error.empty()) {
181 Abort("Cannot interpolate '" + field_name + "' in " + file_name +
182 " onto the model grid: " + error + "." + context);
183 }
184 bcast_string(g.lon_name);
185 bcast_string(g.lat_name);
186 const int ioproc = ParallelDescriptor::IOProcessorNumber();
187 ParallelDescriptor::Bcast(&g.has_fill, 1, ioproc);
188 ParallelDescriptor::Bcast(&g.has_missing, 1, ioproc);
189 ParallelDescriptor::Bcast(&g.fill_value, 1, ioproc);
190 ParallelDescriptor::Bcast(&g.missing_value, 1, ioproc);
191 ParallelDescriptor::Bcast(&g.scale_factor, 1, ioproc);
192 ParallelDescriptor::Bcast(&g.add_offset, 1, ioproc);
193
194 Vector<NDArray<double>> arrays(2);
195 ReadNetCDFFile(file_name, {g.lon_name, g.lat_name}, arrays);
196 if (ParallelDescriptor::IOProcessor()) {
197 g.lon.assign(arrays[0].get_data(), arrays[0].get_data() + arrays[0].get_vshape()[0]);
198 g.lat.assign(arrays[1].get_data(), arrays[1].get_data() + arrays[1].get_vshape()[0]);
199 }
200 bcast_vector(g.lon);
201 bcast_vector(g.lat);
202 return g;
203}

205/**
206 * Weights already built, for the fields that follow on the same source grid. An entry is
207 * found again by the source axes as the file stores them, the target points and the forcing
208 * layout, which together fix the weights. It holds them only weakly, so they are freed with
209 * the last field that uses them; the cache is emptied at finalize.
210 */
211struct WeightsEntry
212{
213 std::vector<double> lon, lat;
214 Box domain;
215 BoxArray ba;
216 DistributionMapping dm;
217 const MultiFab* target_lon = nullptr;
218 const MultiFab* target_lat = nullptr;
219 std::weak_ptr<const MultiFab> weights;
220 std::weak_ptr<const iMultiFab> indices;
221};
222
223std::vector<WeightsEntry>& weights_cache ()
224{
225 static std::vector<WeightsEntry> cache;
226 static bool registered = false;
227 if (!registered) {
228 amrex::ExecOnFinalize([] () { weights_cache().clear(); });
229 registered = true;
230 }
231 return cache;
232}

238void
239LatLonRegrid::define (const std::string& file_name, const std::string& field_name,
240 const std::vector<long long>& field_shape,
241 const MultiFab& target_lon, const MultiFab& target_lat,
242 const MultiFab& like, const Box& a_domain,
243 const LatLonRegridOptions& options, const std::string& context)
244{
245 const SourceGrid g = find_source_grid(file_name, field_name, field_shape, options, context);
246 lon_name = g.lon_name;
247 lat_name = g.lat_name;
248 lon_file.assign(g.lon.begin(), g.lon.end());
249 lat_file.assign(g.lat.begin(), g.lat.end());
250 nlon = static_cast<int>(lon_file.size());
251 nlat = static_cast<int>(lat_file.size());
252 has_fill = g.has_fill;
253 has_missing = g.has_missing;
254 fill_value = g.fill_value;
255 missing_value = g.missing_value;
256 scale_factor = g.scale_factor;
257 add_offset = g.add_offset;
258 fill_max_rings = options.fill_max_rings;
259 domain = makeSlab(a_domain, 2, 0);
260
261 const std::string where = "'" + field_name + "' in " + file_name;
262 if (nlon < 2 || nlat < 2) {
263 Abort("Cannot interpolate " + where + ": its source grid is " + std::to_string(nlon) +
264 " x " + std::to_string(nlat) + "; it needs at least two points along each axis.");
265 }
266
267 // Longitude made continuous, so an axis crossing the branch cut (170 .. 180, -180 .. -170)
268 // is monotone; then both axes put in increasing order.
269 lon_inc = lon_file;
270 latlon_regrid::unwrap_longitudes(lon_inc);
271 lat_inc = lat_file;
272 for (auto [axis, flip, name] : { std::make_tuple(&lon_inc, &flip_lon, lon_name),
273 std::make_tuple(&lat_inc, &flip_lat, lat_name) })
274 {
275 *flip = axis->back() < axis->front();
276 if (*flip) { std::reverse(axis->begin(), axis->end()); }
277 for (std::size_t m = 1; m < axis->size(); ++m) {
278 if (!((*axis)[m] > (*axis)[m-1])) {
279 Abort("Cannot interpolate " + where + ": its axis '" + name +
280 "' is not strictly monotone (at index " + std::to_string(m) + ").");
281 }
282 }
283 }
284
285 // A longitude axis whose spacing closes the circle covers the globe
286 const double span = latlon_regrid::longitude_span(lon_inc);
287 periodic_lon = latlon_regrid::is_periodic(lon_inc);
288 if (span > 360.0 + 1.e-9) {
289 Abort("Cannot interpolate " + where + ": its longitude axis spans " + std::to_string(span) +
290 " degrees, more than once around the globe.");
291 }
292
293 // Every rank holds the same axes and layout, so every rank finds the same entry, or none
294 auto& cache = weights_cache();
295 cache.erase(std::remove_if(cache.begin(), cache.end(),
296 [] (const WeightsEntry& e) { return e.weights.expired(); }),
297 cache.end());
298 for (const auto& e : cache) {
299 if (e.lon == lon_file && e.lat == lat_file && e.domain == domain &&
300 e.ba == like.boxArray() && e.dm == like.DistributionMap() &&
301 e.target_lon == &target_lon && e.target_lat == &target_lat) {
302 weights = e.weights.lock();
303 indices = e.indices.lock();
304 if (weights && indices) {
305 source_box = find_source_box();
306 amrex::Print() << "Interpolating " << field_name << " from " << summary()
307 << ", with the weights of an earlier field on it" << std::endl;
308 return;
309 }
310 }
311 }
312
313 auto w = std::make_shared<MultiFab>(like.boxArray(), like.DistributionMap(), stencil_size,
314 IntVect(1,1,0));
315 auto idx = std::make_shared<iMultiFab>(like.boxArray(), like.DistributionMap(),
316 3*stencil_size, IntVect(1,1,0));
317 w->setVal(0.0);
318 idx->setVal(-1);
319
320 // Target coordinates on the forcing layout, with the ring
321 MultiFab lon(like.boxArray(), like.DistributionMap(), 1, IntVect(1,1,0));
322 MultiFab lat(like.boxArray(), like.DistributionMap(), 1, IntVect(1,1,0));
323 lon.ParallelCopy(target_lon, 0, 0, 1, IntVect(1,1,0), IntVect(1,1,0));
324 lat.ParallelCopy(target_lat, 0, 0, 1, IntVect(1,1,0), IntVect(1,1,0));
325
326 build_weights(lon, lat, *w, *idx);
327 weights = w;
328 indices = idx;
329 cache.push_back({lon_file, lat_file, domain, like.boxArray(), like.DistributionMap(),
330 &target_lon, &target_lat, weights, indices});
331 source_box = find_source_box();
332
333 amrex::Print() << "Interpolating " << field_name << " from " << summary() << std::endl;
334}

336void
337LatLonRegrid::build_weights (const MultiFab& lon, const MultiFab& lat,
338 MultiFab& weights_out, iMultiFab& indices_out) const
339{
340 Gpu::DeviceVector<Real> d_lon(nlon), d_lat(nlat);
341 {
342 std::vector<Real> h_lon(lon_inc.begin(), lon_inc.end());
343 std::vector<Real> h_lat(lat_inc.begin(), lat_inc.end());
344 Gpu::copy(Gpu::hostToDevice, h_lon.begin(), h_lon.end(), d_lon.begin());
345 Gpu::copy(Gpu::hostToDevice, h_lat.begin(), h_lat.end(), d_lat.begin());
346 }
347 const Real* plon = d_lon.data();
348 const Real* plat = d_lat.data();
349 const int Nx = nlon, Ny = nlat;
350 const bool periodic = periodic_lon, fx = flip_lon, fy = flip_lat;
351 const Real lon_ref = static_cast<Real>(latlon_regrid::longitude_reference(lon_inc, periodic));
352 const Real tol = Real(1.e-9);
353
354 ReduceOps<ReduceOpSum, ReduceOpMin, ReduceOpMax, ReduceOpMin, ReduceOpMax> reduce_op;
355 ReduceData<int, Real, Real, Real, Real> reduce_data(reduce_op);
356 using ReduceTuple = typename decltype(reduce_data)::Type;
357
358 for (MFIter mfi(weights_out); mfi.isValid(); ++mfi) {
359 const Box bx = target_box(mfi);
360 auto const& lon_a = lon.const_array(mfi);
361 auto const& lat_a = lat.const_array(mfi);
362 auto const& w = weights_out.array(mfi);
363 auto const& idx = indices_out.array(mfi);
364
365 reduce_op.eval(bx, reduce_data, [=] AMREX_GPU_DEVICE (int i, int j, int k) -> ReduceTuple
366 {
367 const Real lt = latlon_regrid::wrap_longitude(lon_a(i,j,k), lon_ref);
368 const Real pt = lat_a(i,j,k);
369 const latlon_regrid::Stencil st =
370 latlon_regrid::bilinear_stencil(plon, Nx, plat, Ny, periodic, fx, fy, lt, pt, tol);
371 for (int s = 0; s < 4; ++s) {
372 w(i,j,k,s) = st.inside ? st.w[s] : Real(0.0);
373 idx(i,j,k,3*s) = st.inside ? st.i[s] : -1;
374 idx(i,j,k,3*s+1) = st.inside ? st.j[s] : -1;
375 idx(i,j,k,3*s+2) = 0;
376 }
377
378 return { st.inside ? 0 : 1, lt, lt, pt, pt };
379 });
380 }
381
382 ReduceTuple r = reduce_data.value(reduce_op);
383 int n_out = amrex::get<0>(r);
384 Real lon_min = amrex::get<1>(r), lon_max = amrex::get<2>(r);
385 Real lat_min = amrex::get<3>(r), lat_max = amrex::get<4>(r);
386 ParallelDescriptor::ReduceIntSum(n_out);
387 ParallelDescriptor::ReduceRealMin(lon_min);
388 ParallelDescriptor::ReduceRealMax(lon_max);
389 ParallelDescriptor::ReduceRealMin(lat_min);
390 ParallelDescriptor::ReduceRealMax(lat_max);
391
392 if (n_out > 0) {
393 std::ostringstream os;
394 os << "The forcing grid does not contain the model grid: " << n_out << " rho points of"
395 << " level 0 and its boundary ring lie outside it. Forcing grid: " << summary()
396 << ". Model grid (in the forcing grid's longitude convention): lon " << lon_min
397 << " to " << lon_max << ", lat " << lat_min << " to " << lat_max
398 << ". Supply forcing that covers the domain with its boundary ring.";
399 Abort(os.str());
400 }
401}

403Box
404LatLonRegrid::find_source_box () const
405{
406 // Corner 0 of the stencil of every target cell this rank fills. The other corners lie
407 // within one cell of it, and so does the cell the nearest-valid search centers on, so
408 // the cells read lie within fill_max_rings + 1 of it. Along a periodic longitude the
409 // corners are also taken half way round, so that a range across the seam (as on a
410 // 0 .. 360 axis near Greenwich) is found as one range there.
411 const int half = nlon / 2, N = nlon;
412 ReduceOps<ReduceOpMin, ReduceOpMax, ReduceOpMin, ReduceOpMax, ReduceOpMin, ReduceOpMax> reduce_op;
413 ReduceData<int, int, int, int, int, int> reduce_data(reduce_op);
414 using ReduceTuple = typename decltype(reduce_data)::Type;
415 const int big = std::numeric_limits<int>::max();
416 for (MFIter mfi(*indices); mfi.isValid(); ++mfi) {
417 const Box bx = target_box(mfi);
418 auto const& idx = indices->const_array(mfi);
419 reduce_op.eval(bx, reduce_data, [=] AMREX_GPU_DEVICE (int i, int j, int k) -> ReduceTuple
420 {
421 const int i0 = idx(i,j,k,0), j0 = idx(i,j,k,1);
422 if (i0 < 0) { return { big, -big, big, -big, big, -big }; }
423 const int is = (i0 + half) % N;
424 return { i0, i0, j0, j0, is, is };
425 });
426 }
427 ReduceTuple r = reduce_data.value(reduce_op);
428 int ilo = amrex::get<0>(r), ihi = amrex::get<1>(r);
429 int jlo = amrex::get<2>(r), jhi = amrex::get<3>(r);
430 const int islo = amrex::get<4>(r), ishi = amrex::get<5>(r);
431 if (ilo > ihi) { return Box(); }
432 if (periodic_lon && ishi - islo < ihi - ilo) {
433 ilo = islo - half;
434 ihi = ishi - half;
435 }
436
437 const int R = fill_max_rings + 1;
438 ilo -= R; ihi += R; jlo -= R; jhi += R;
439 jlo = amrex::max(jlo, 0); jhi = amrex::min(jhi, nlat-1);
440 if (!periodic_lon) {
441 ilo = amrex::max(ilo, 0); ihi = amrex::min(ihi, nlon-1);
442 } else if (ihi - ilo + 1 >= nlon) {
443 ilo = 0; ihi = nlon-1;
444 } else if (ilo < 0) {
445 // Starting in [0, nlon); a box past the last longitude wraps to the first
446 ilo += nlon; ihi += nlon;
447 }
448 return Box(IntVect(ilo, jlo, 0), IntVect(ihi, jhi, 0));
449}

451void
452LatLonRegrid::apply (const std::string& file_name, const std::string& field_name, int itime,
453 MultiFab& dst) const
454{
455 // Each rank reads its own box of the record, (time, lat, lon) in the file: one piece,
456 // or two when the box runs past the last longitude and wraps to the first. Every rank
457 // joins both collective reads, with an empty count for a piece it does not have.
458 const bool any = source_box.ok();
459 const int ilo = source_box.smallEnd(0), jlo = source_box.smallEnd(1);
460 const int ni = any ? source_box.length(0) : 0, nj = any ? source_box.length(1) : 0;
461 const int ni_east = std::min(ni, nlon - ilo); // columns ilo .. nlon-1
462 const int ni_wrap = ni - ni_east; // columns 0 .. ni_wrap-1, after them
463 std::vector<Real> east(static_cast<std::size_t>(ni_east) * nj);
464 std::vector<Real> wrap(static_cast<std::size_t>(ni_wrap) * nj);
465 {
466 auto ncf = ncutils::NCFile::open(file_name, NC_NOWRITE);
467 const auto var = ncf.var(field_name);
468 const auto shape = var.shape();
469 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(shape.size() == 3 && shape[1] == nlat && shape[2] == nlon &&
470 itime >= 0 && itime < shape[0],
471 "Lat-lon forcing record of '" + field_name + "' has an unexpected shape");
472 const MPI_Offset row = any ? jlo : 0;
473 var.get_all(east.data(), {itime, row, ni_east > 0 ? ilo : 0}, {ni_east > 0 ? 1 : 0, nj, ni_east});
474 var.get_all(wrap.data(), {itime, row, 0}, {ni_wrap > 0 ? 1 : 0, nj, ni_wrap});
475 ncf.close();
476 }
477
478 // Missing-data markers are compared exactly and in the stored (packed) values, as CF
479 // defines them, then the data unpacked. A marker becomes NaN, which the interpolation
480 // sets aside. (lat, lon) row-major in the file is (i = lon, j = lat) in a FAB, indexed
481 // as the source grid is, so the stencil's indices apply unchanged.
482 const double nan = std::numeric_limits<double>::quiet_NaN();
483 std::vector<Real> buf(static_cast<std::size_t>(ni) * nj);
484 for (int n = 0; n < nj; ++n) {
485 for (int m = 0; m < ni; ++m) {
486 const double v = static_cast<double>(m < ni_east ? east[n*ni_east + m]
487 : wrap[n*ni_wrap + m - ni_east]);
488 const bool marker = (has_fill && v == fill_value) || (has_missing && v == missing_value);
489 buf[n*ni + m] = static_cast<Real>(marker ? nan : v * scale_factor + add_offset);
490 }
491 }
492
493 FArrayBox src(any ? source_box : Box(IntVect(0), IntVect(0)), 1);
494 if (any) { Gpu::copy(Gpu::hostToDevice, buf.begin(), buf.end(), src.dataPtr()); }
495
496 interpolate(src, dst);
497
498 // A target point with no valid source value within reach is left NaN
499 ReduceOps<ReduceOpSum> op;
500 ReduceData<int> data(op);
501 for (MFIter mfi(dst); mfi.isValid(); ++mfi) {
502 const Box bx = target_box(mfi);
503 auto const& d = dst.const_array(mfi);
504 op.eval(bx, data, [=] AMREX_GPU_DEVICE (int i, int j, int k) -> GpuTuple<int>
505 {
506 return { (d(i,j,k) == d(i,j,k)) ? 0 : 1 };
507 });
508 }
509 int n_missing = amrex::get<0>(data.value(op));
510 ParallelDescriptor::ReduceIntSum(n_missing);
511 if (n_missing > 0) {
512 Abort("Interpolating '" + field_name + "' record " + std::to_string(itime) + " of " +
513 file_name + " onto the model grid: " + std::to_string(n_missing) +
514 " rho points have no valid source value within " + std::to_string(fill_max_rings) +
515 " cells (the source is _FillValue or missing_value there). Raise "
516 "remora.frc_fill_max_rings or fill the forcing over the ocean.");
517 }
518}

520void
521LatLonRegrid::interpolate (const FArrayBox& src_fab, MultiFab& dst) const
522{
523 auto const& src = src_fab.const_array();
524 const int Nx = nlon, Ny = nlat, R = fill_max_rings;
525 // A column before the box's first is one it reads past the last longitude
526 const int i_first = src_fab.box().smallEnd(0);
527 const bool periodic = periodic_lon, fx = flip_lon, fy = flip_lat;
528 const Real nan = std::numeric_limits<Real>::quiet_NaN();
529
530 for (MFIter mfi(dst); mfi.isValid(); ++mfi) {
531 const Box bx = target_box(mfi);
532 auto const& d = dst.array(mfi);
533 auto const& w = weights->const_array(mfi);
534 auto const& idx = indices->const_array(mfi);
535
536 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
537 {
538 if (idx(i,j,k,0) < 0) { d(i,j,k) = nan; return; }
539
540 Real sum = Real(0.0), wsum = Real(0.0);
541 for (int s = 0; s < stencil_size; ++s) {
542 const int m = idx(i,j,k,3*s);
543 const Real v = src(m < i_first ? m + Nx : m, idx(i,j,k,3*s+1), 0);
544 if (w(i,j,k,s) > Real(0.0) && is_valid(v)) {
545 sum += w(i,j,k,s) * v;
546 wsum += w(i,j,k,s);
547 }
548 }
549 if (wsum > Real(0.0)) { d(i,j,k) = sum / wsum; return; }
550
551 // Every corner with weight is missing: take the nearest valid source value. The
552 // target's fractional source index follows from the stencil: corner 0 is its
553 // lower-left cell, and the fractions are p = w1 + w2 and q = w2 + w3, counted
554 // backward along an axis the file stores decreasing.
555 const Real p = w(i,j,k,1) + w(i,j,k,2);
556 const Real q = w(i,j,k,2) + w(i,j,k,3);
557 const Real xf = Real(idx(i,j,k,0)) + (fx ? -p : p);
558 const Real yf = Real(idx(i,j,k,1)) + (fy ? -q : q);
559 const int ci = static_cast<int>(std::round(xf));
560 const int cj = static_cast<int>(std::round(yf));
561 Real best = nan, best_d2 = std::numeric_limits<Real>::max();
562 for (int dj = -R; dj <= R; ++dj) {
563 const int n = cj + dj;
564 if (n < 0 || n >= Ny) { continue; }
565 for (int di = -R; di <= R; ++di) {
566 int m = ci + di;
567 if (periodic) { m = ((m % Nx) + Nx) % Nx; }
568 else if (m < 0 || m >= Nx) { continue; }
569 const Real v = src(m < i_first ? m + Nx : m, n, 0);
570 if (!is_valid(v)) { continue; }
571 Real dx = Real(m) - xf;
572 if (periodic) { dx -= Real(Nx) * std::round(dx / Real(Nx)); }
573 const Real dy = Real(n) - yf;
574 const Real d2 = dx*dx + dy*dy;
575 if (d2 < best_d2) { best_d2 = d2; best = v; }
576 }
577 }
578 d(i,j,k) = best;
579 });
580 }
581}

583bool
584LatLonRegrid::same_source_grid (const std::string& file_name, const std::string& field_name,
585 const std::vector<long long>& field_shape) const
586{
587 LatLonRegridOptions opts;
588 opts.lon_name = lon_name;
589 opts.lat_name = lat_name;
590 const SourceGrid g = find_source_grid(file_name, field_name, field_shape, opts, "");
591 auto same = [] (const std::vector<double>& a, const std::vector<double>& b) {
592 if (a.size() != b.size()) { return false; }
593 for (std::size_t m = 0; m < a.size(); ++m) {
594 if (std::abs(a[m] - b[m]) > 1.e-9 * std::max(1.0, std::abs(a[m]))) { return false; }
595 }
596 return true;
597 };
598 return same(g.lon, lon_file) && same(g.lat, lat_file) &&
599 g.has_fill == static_cast<int>(has_fill) && g.has_missing == static_cast<int>(has_missing) &&
600 (!has_fill || g.fill_value == fill_value) && (!has_missing || g.missing_value == missing_value) &&
601 g.scale_factor == scale_factor && g.add_offset == add_offset;
602}
