44/** Whether a units attribute names degrees east (or north) */
45bool units_say (const ncutils::NCVar& var, const std::string& direction)
46{
47 if (!var.has_attr("units")) { return false; }
48 std::string u = var.get_attr("units");
49 std::transform(u.begin(), u.end(), u.begin(), [] (unsigned char c) { return std::tolower(c); });
50 return u.find("degree") != std::string::npos &&
51 (u.find(direction) != std::string::npos ||
52 u.find(direction == "east" ? "_e" : "_n") != std::string::npos);

74/**
75 * Find and read the coordinate axes of a (time, lat, lon) variable. The names come from
76 * the inputs when given; otherwise from the variable's CF coordinates attribute, each name
77 * identified by its units (degrees east or north) or, failing that, by its leading
78 * "lon"/"lat" or, failing that, by order as ROMS's get_varcoords takes them; otherwise from
79 * the CF coordinate variables of its last two dimensions (longitude(longitude), as ERA5 and
80 * most reanalyses write them). Each axis must be 1D and dimensioned like the variable's
81 * matching dimension. Collective; aborts on every rank with the reason, followed by
82 * `context`, when the axes cannot be found.
83 */
84SourceGrid
85find_source_grid (const std::string& file_name, const std::string& field_name,
86 const std::vector<long long>& field_shape,
87 const LatLonRegridOptions& options, const std::string& context)
88{
89 SourceGrid g;
90 std::string error;
91
92 auto ncf = ncutils::NCFile::open(file_name, NC_NOCLOBBER);
93 ncmpi_begin_indep_data(ncf.ncid);
94 if (ParallelDescriptor::IOProcessor())
95 {
96 const auto var = ncf.var(field_name);
97 const int rank = var.ndim();
98
99 g.lon_name = options.lon_name;
100 g.lat_name = options.lat_name;
101 if (g.lon_name.empty() || g.lat_name.empty()) {
102 if (!var.has_attr("coordinates")) {
103 if (has_dimension_coordinate(ncf, var, rank-1) &&
104 has_dimension_coordinate(ncf, var, rank-2)) {
105 if (g.lon_name.empty()) { g.lon_name = dim_name(ncf, var, rank-1); }
106 if (g.lat_name.empty()) { g.lat_name = dim_name(ncf, var, rank-2); }
107 } else {
108 error = "it has no coordinates attribute naming its longitude and latitude, "
109 "its last two dimensions have no coordinate variables, and "
110 "remora.frc_lon_varname and remora.frc_lat_varname are not both set";
111 }
112 } else {
113 std::istringstream tokens(var.get_attr("coordinates"));
114 std::vector<std::string> names;
115 for (std::string t; tokens >> t; ) {
116 if (ncf.has_var(t) && ncf.var(t).ndim() >= 1 &&
117 dim_name(ncf, ncf.var(t), 0) != dim_name(ncf, var, 0)) {
118 names.push_back(t); // skip the time coordinate
119 }
120 }
121 std::string by_units_lon, by_units_lat, by_name_lon, by_name_lat;
122 for (const auto& t : names) {
123 if (units_say(ncf.var(t), "east")) { by_units_lon = t; }
124 if (units_say(ncf.var(t), "north")) { by_units_lat = t; }
125 if (t.rfind("lon", 0) == 0) { by_name_lon = t; }
126 if (t.rfind("lat", 0) == 0) { by_name_lat = t; }
127 }
128 std::string lon = !by_units_lon.empty() ? by_units_lon : by_name_lon;
129 std::string lat = !by_units_lat.empty() ? by_units_lat : by_name_lat;
130 if ((lon.empty() || lat.empty()) && names.size() == 2) {
131 lon = names[0];
132 lat = names[1];
133 }
134 if (g.lon_name.empty()) { g.lon_name = lon; }
135 if (g.lat_name.empty()) { g.lat_name = lat; }
136 if (g.lon_name.empty() || g.lat_name.empty()) {
137 error = "its coordinates attribute '" + var.get_attr("coordinates") +
138 "' does not name a longitude and a latitude variable; set "
139 "remora.frc_lon_varname and remora.frc_lat_varname";
140 }
141 }
142 }
143
144 // Each axis 1D, along the variable's last (lon) or next-to-last (lat) dimension
145 const std::pair<std::string, int> axes[2] = {{g.lon_name, rank-1}, {g.lat_name, rank-2}};
146 for (const auto& [name, idim] : axes) {
147 if (!error.empty()) { break; }
148 if (!ncf.has_var(name)) {
149 error = "its coordinate variable '" + name + "' is not in the file";
150 } else if (ncf.var(name).ndim() != 1) {
151 error = "its coordinate variable '" + name + "' is " +
152 std::to_string(ncf.var(name).ndim()) + "D; only 1D (rectilinear) "
153 "longitude and latitude axes are supported";
154 } else if (dim_name(ncf, ncf.var(name), 0) != dim_name(ncf, var, idim) ||
155 ncf.var(name).shape()[0] != field_shape[idim]) {
156 error = "its coordinate variable '" + name + "' does not run along its " +
157 std::string(idim == rank-1 ? "last" : "next-to-last") + " dimension '" +
158 dim_name(ncf, var, idim) + "'; the variable must be (time, lat, lon)";
159 }
160 }
161
162 int has_scale = 0, has_offset = 0;
163 for (const auto& [attr, has, val] : { std::make_tuple("_FillValue", &g.has_fill, &g.fill_value),
164 std::make_tuple("missing_value", &g.has_missing, &g.missing_value),
165 std::make_tuple("scale_factor", &has_scale, &g.scale_factor),
166 std::make_tuple("add_offset", &has_offset, &g.add_offset) })
167 {
168 if (var.has_attr(attr)) {
169 std::vector<double> v;
170 var.get_attr(attr, v);
171 if (!v.empty()) { *has = 1; *val = v[0]; }
172 }
173 }
174 }
175 ncf.close();
176
177 bcast_string(error);
178 if (!error.empty()) {
179 Abort("Cannot interpolate '" + field_name + "' in " + file_name +
180 " onto the model grid: " + error + "." + context);
181 }
182 bcast_string(g.lon_name);
183 bcast_string(g.lat_name);
184 const int ioproc = ParallelDescriptor::IOProcessorNumber();
185 ParallelDescriptor::Bcast(&g.has_fill, 1, ioproc);
186 ParallelDescriptor::Bcast(&g.has_missing, 1, ioproc);
187 ParallelDescriptor::Bcast(&g.fill_value, 1, ioproc);
188 ParallelDescriptor::Bcast(&g.missing_value, 1, ioproc);
189 ParallelDescriptor::Bcast(&g.scale_factor, 1, ioproc);
190 ParallelDescriptor::Bcast(&g.add_offset, 1, ioproc);
191
192 Vector<NDArray<double>> arrays(2);
193 ReadNetCDFFile(file_name, {g.lon_name, g.lat_name}, arrays);
194 if (ParallelDescriptor::IOProcessor()) {
195 g.lon.assign(arrays[0].get_data(), arrays[0].get_data() + arrays[0].get_vshape()[0]);
196 g.lat.assign(arrays[1].get_data(), arrays[1].get_data() + arrays[1].get_vshape()[0]);
197 }
198 bcast_vector(g.lon);
199 bcast_vector(g.lat);
200 return g;
201}

203/** Largest m with a[m] <= x in an increasing array, or -1 */
204AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
205int bracket (const Real* a, int n, Real x)
206{
207 int lo = -1, hi = n; // a[lo] <= x < a[hi], with a[-1] = -inf and a[n] = +inf
208 while (hi - lo > 1) {
209 const int mid = (lo + hi) / 2;
210 if (a[mid] <= x) { lo = mid; } else { hi = mid; }
211 }
212 return lo;
213}

224void
225LatLonRegrid::define (const std::string& file_name, const std::string& field_name,
226 const std::vector<long long>& field_shape,
227 const MultiFab& target_lon, const MultiFab& target_lat,
228 const MultiFab& like, const Box& a_domain,
229 const LatLonRegridOptions& options, const std::string& context)
230{
231 const SourceGrid g = find_source_grid(file_name, field_name, field_shape, options, context);
232 lon_name = g.lon_name;
233 lat_name = g.lat_name;
234 lon_file.assign(g.lon.begin(), g.lon.end());
235 lat_file.assign(g.lat.begin(), g.lat.end());
236 nlon = static_cast<int>(lon_file.size());
237 nlat = static_cast<int>(lat_file.size());
238 has_fill = g.has_fill;
239 has_missing = g.has_missing;
240 fill_value = g.fill_value;
241 missing_value = g.missing_value;
242 scale_factor = g.scale_factor;
243 add_offset = g.add_offset;
244 fill_max_rings = options.fill_max_rings;
245 domain = makeSlab(a_domain, 2, 0);
246
247 const std::string where = "'" + field_name + "' in " + file_name;
248 if (nlon < 2 || nlat < 2) {
249 Abort("Cannot interpolate " + where + ": its source grid is " + std::to_string(nlon) +
250 " x " + std::to_string(nlat) + "; it needs at least two points along each axis.");
251 }
252
253 // Longitude made continuous, so an axis crossing the branch cut (170 .. 180, -180 .. -170)
254 // is monotone; then both axes put in increasing order.
255 lon_inc = lon_file;
256 for (int m = 1; m < nlon; ++m) {
257 lon_inc[m] -= 360.0 * std::round((lon_inc[m] - lon_inc[m-1]) / 360.0);
258 }
259 lat_inc = lat_file;
260 for (auto [axis, flip, name] : { std::make_tuple(&lon_inc, &flip_lon, lon_name),
261 std::make_tuple(&lat_inc, &flip_lat, lat_name) })
262 {
263 *flip = axis->back() < axis->front();
264 if (*flip) { std::reverse(axis->begin(), axis->end()); }
265 for (std::size_t m = 1; m < axis->size(); ++m) {
266 if (!((*axis)[m] > (*axis)[m-1])) {
267 Abort("Cannot interpolate " + where + ": its axis '" + name +
268 "' is not strictly monotone (at index " + std::to_string(m) + ").");
269 }
270 }
271 }
272
273 // A longitude axis whose spacing closes the circle covers the globe
274 const double span = lon_inc.back() - lon_inc.front();
275 const double dlon = span / (nlon - 1);
276 periodic_lon = std::abs(span + dlon - 360.0) < 1.e-3 * dlon;
277 if (span > 360.0 + 1.e-9) {
278 Abort("Cannot interpolate " + where + ": its longitude axis spans " + std::to_string(span) +
279 " degrees, more than once around the globe.");
280 }
281
282 weights.define(like.boxArray(), like.DistributionMap(), stencil_size, IntVect(1,1,0));
283 indices.define(like.boxArray(), like.DistributionMap(), 3*stencil_size, IntVect(1,1,0));
284 weights.setVal(0.0);
285 indices.setVal(-1);
286
287 // Target coordinates on the forcing layout, with the ring
288 MultiFab lon(like.boxArray(), like.DistributionMap(), 1, IntVect(1,1,0));
289 MultiFab lat(like.boxArray(), like.DistributionMap(), 1, IntVect(1,1,0));
290 lon.ParallelCopy(target_lon, 0, 0, 1, IntVect(1,1,0), IntVect(1,1,0));
291 lat.ParallelCopy(target_lat, 0, 0, 1, IntVect(1,1,0), IntVect(1,1,0));
292
293 build_weights(lon, lat);
294
295 amrex::Print() << "Interpolating " << field_name << " from " << summary() << std::endl;
296}

298void
299LatLonRegrid::build_weights (const MultiFab& lon, const MultiFab& lat)
300{
301 Gpu::DeviceVector<Real> d_lon(nlon), d_lat(nlat);
302 {
303 std::vector<Real> h_lon(lon_inc.begin(), lon_inc.end());
304 std::vector<Real> h_lat(lat_inc.begin(), lat_inc.end());
305 Gpu::copy(Gpu::hostToDevice, h_lon.begin(), h_lon.end(), d_lon.begin());
306 Gpu::copy(Gpu::hostToDevice, h_lat.begin(), h_lat.end(), d_lat.begin());
307 }
308 const Real* plon = d_lon.data();
309 const Real* plat = d_lat.data();
310 const int Nx = nlon, Ny = nlat;
311 const bool periodic = periodic_lon, fx = flip_lon, fy = flip_lat;
312 // A target longitude is taken into [lon_ref, lon_ref + 360): from the axis's first point
313 // when it covers the globe, and otherwise centered on the axis, so that a point outside
314 // it is reported on the side it misses.
315 const Real lon_ref = periodic ? static_cast<Real>(lon_inc.front())
316 : static_cast<Real>(0.5 * (lon_inc.front() + lon_inc.back()) - 180.0);
317 const Real tol = Real(1.e-9);
318
319 ReduceOps<ReduceOpSum, ReduceOpMin, ReduceOpMax, ReduceOpMin, ReduceOpMax> reduce_op;
320 ReduceData<int, Real, Real, Real, Real> reduce_data(reduce_op);
321 using ReduceTuple = typename decltype(reduce_data)::Type;
322
323 for (MFIter mfi(weights); mfi.isValid(); ++mfi) {
324 const Box bx = target_box(mfi);
325 auto const& lon_a = lon.const_array(mfi);
326 auto const& lat_a = lat.const_array(mfi);
327 auto const& w = weights.array(mfi);
328 auto const& idx = indices.array(mfi);
329
330 reduce_op.eval(bx, reduce_data, [=] AMREX_GPU_DEVICE (int i, int j, int k) -> ReduceTuple
331 {
332 // Into the source's longitude window
333 const Real lt = lon_a(i,j,k) - Real(360.0) * std::floor((lon_a(i,j,k) - lon_ref) / Real(360.0));
334 const Real pt = lat_a(i,j,k);
335
336 int m, m2;
337 Real p;
338 bool inside = true;
339 if (periodic) {
340 m = amrex::max(0, bracket(plon, Nx, lt));
341 m2 = (m + 1) % Nx;
342 const Real right = (m == Nx-1) ? plon[0] + Real(360.0) : plon[m+1];
343 p = (lt - plon[m]) / (right - plon[m]);
344 } else {
345 inside = (lt >= plon[0] - tol) && (lt <= plon[Nx-1] + tol);
346 m = amrex::min(amrex::max(bracket(plon, Nx, lt), 0), Nx-2);
347 m2 = m + 1;
348 p = (lt - plon[m]) / (plon[m2] - plon[m]);
349 }
350 inside = inside && (pt >= plat[0] - tol) && (pt <= plat[Ny-1] + tol);
351 const int n = amrex::min(amrex::max(bracket(plat, Ny, pt), 0), Ny-2);
352 const int n2 = n + 1;
353 Real q = (pt - plat[n]) / (plat[n2] - plat[n]);
354 p = amrex::min(amrex::max(p, Real(0.0)), Real(1.0));
355 q = amrex::min(amrex::max(q, Real(0.0)), Real(1.0));
356
357 // Stencil in the file's own index order
358 auto fi = [=] (int a) { return fx ? Nx-1-a : a; };
359 auto fj = [=] (int b) { return fy ? Ny-1-b : b; };
360 const int si[4] = { fi(m), fi(m2), fi(m2), fi(m) };
361 const int sj[4] = { fj(n), fj(n), fj(n2), fj(n2) };
362 const Real sw[4] = { (Real(1.0)-p)*(Real(1.0)-q), p*(Real(1.0)-q), p*q, (Real(1.0)-p)*q };
363 for (int s = 0; s < 4; ++s) {
364 w(i,j,k,s) = inside ? sw[s] : Real(0.0);
365 idx(i,j,k,3*s) = inside ? si[s] : -1;
366 idx(i,j,k,3*s+1) = inside ? sj[s] : -1;
367 idx(i,j,k,3*s+2) = 0;
368 }
369
370 return { inside ? 0 : 1, lt, lt, pt, pt };
371 });
372 }
373
374 ReduceTuple r = reduce_data.value(reduce_op);
375 int n_out = amrex::get<0>(r);
376 Real lon_min = amrex::get<1>(r), lon_max = amrex::get<2>(r);
377 Real lat_min = amrex::get<3>(r), lat_max = amrex::get<4>(r);
378 ParallelDescriptor::ReduceIntSum(n_out);
379 ParallelDescriptor::ReduceRealMin(lon_min);
380 ParallelDescriptor::ReduceRealMax(lon_max);
381 ParallelDescriptor::ReduceRealMin(lat_min);
382 ParallelDescriptor::ReduceRealMax(lat_max);
383
384 if (n_out > 0) {
385 std::ostringstream os;
386 os << "The forcing grid does not contain the model grid: " << n_out << " rho points of"
387 << " level 0 and its boundary ring lie outside it. Forcing grid: " << summary()
388 << ". Model grid (in the forcing grid's longitude convention): lon " << lon_min
389 << " to " << lon_max << ", lat " << lat_min << " to " << lat_max
390 << ". Supply forcing that covers the domain with its boundary ring.";
391 Abort(os.str());
392 }

395void
396LatLonRegrid::apply (const std::string& file_name, const std::string& field_name, int itime,
397 MultiFab& dst) const
398{
399 Vector<NDArray<Real>> arrays(1);
400 ReadNetCDFFile(file_name, {field_name}, arrays, true, itime);
401
402 std::vector<Real> buf(static_cast<std::size_t>(nlon) * nlat);
403 if (ParallelDescriptor::IOProcessor()) {
404 const auto shape = arrays[0].get_vshape();
405 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(shape.size() == 3 && shape[1] == nlat && shape[2] == nlon,
406 "Lat-lon forcing record of '" + field_name + "' has an unexpected shape");
407 // Missing-data markers are compared exactly and in the stored (packed) values, as CF
408 // defines them, then the data unpacked. A marker becomes NaN, which the interpolation
409 // sets aside.
410 const double nan = std::numeric_limits<double>::quiet_NaN();
411 const Real* raw = arrays[0].get_data();
412 for (std::size_t c = 0; c < buf.size(); ++c) {
413 const double v = static_cast<double>(raw[c]);
414 const bool marker = (has_fill && v == fill_value) || (has_missing && v == missing_value);
415 buf[c] = static_cast<Real>(marker ? nan : v * scale_factor + add_offset);
416 }
417 }
418 ParallelDescriptor::Bcast(buf.data(), buf.size(), ParallelDescriptor::IOProcessorNumber());
419
420 // (lat, lon) row-major in the file is (i = lon, j = lat) in a FAB
421 FArrayBox src(Box(IntVect(0,0,0), IntVect(nlon-1, nlat-1, 0)), 1);
422 Gpu::copy(Gpu::hostToDevice, buf.begin(), buf.end(), src.dataPtr());
423
424 interpolate(src, dst);
425
426 // A target point with no valid source value within reach is left NaN
427 ReduceOps<ReduceOpSum> op;
428 ReduceData<int> data(op);
429 for (MFIter mfi(dst); mfi.isValid(); ++mfi) {
430 const Box bx = target_box(mfi);
431 auto const& d = dst.const_array(mfi);
432 op.eval(bx, data, [=] AMREX_GPU_DEVICE (int i, int j, int k) -> GpuTuple<int>
433 {
434 return { (d(i,j,k) == d(i,j,k)) ? 0 : 1 };
435 });
436 }
437 int n_missing = amrex::get<0>(data.value(op));
438 ParallelDescriptor::ReduceIntSum(n_missing);
439 if (n_missing > 0) {
440 Abort("Interpolating '" + field_name + "' record " + std::to_string(itime) + " of " +
441 file_name + " onto the model grid: " + std::to_string(n_missing) +
442 " rho points have no valid source value within " + std::to_string(fill_max_rings) +
443 " cells (the source is _FillValue or missing_value there). Raise "
444 "remora.frc_fill_max_rings or fill the forcing over the ocean.");
445 }
446}

448void
449LatLonRegrid::interpolate (const FArrayBox& src_fab, MultiFab& dst) const
450{
451 auto const& src = src_fab.const_array();
452 const int Nx = nlon, Ny = nlat, R = fill_max_rings;
453 const bool periodic = periodic_lon, fx = flip_lon, fy = flip_lat;
454 const Real nan = std::numeric_limits<Real>::quiet_NaN();
455
456 for (MFIter mfi(dst); mfi.isValid(); ++mfi) {
457 const Box bx = target_box(mfi);
458 auto const& d = dst.array(mfi);
459 auto const& w = weights.const_array(mfi);
460 auto const& idx = indices.const_array(mfi);
461
462 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
463 {
464 if (idx(i,j,k,0) < 0) { d(i,j,k) = nan; return; }
465
466 Real sum = Real(0.0), wsum = Real(0.0);
467 for (int s = 0; s < stencil_size; ++s) {
468 const Real v = src(idx(i,j,k,3*s), idx(i,j,k,3*s+1), 0);
469 if (w(i,j,k,s) > Real(0.0) && is_valid(v)) {
470 sum += w(i,j,k,s) * v;
471 wsum += w(i,j,k,s);
472 }
473 }
474 if (wsum > Real(0.0)) { d(i,j,k) = sum / wsum; return; }
475
476 // Every corner with weight is missing: take the nearest valid source value. The
477 // target's fractional source index follows from the stencil: corner 0 is its
478 // lower-left cell, and the fractions are p = w1 + w2 and q = w2 + w3, counted
479 // backward along an axis the file stores decreasing.
480 const Real p = w(i,j,k,1) + w(i,j,k,2);
481 const Real q = w(i,j,k,2) + w(i,j,k,3);
482 const Real xf = Real(idx(i,j,k,0)) + (fx ? -p : p);
483 const Real yf = Real(idx(i,j,k,1)) + (fy ? -q : q);
484 const int ci = static_cast<int>(std::round(xf));
485 const int cj = static_cast<int>(std::round(yf));
486 Real best = nan, best_d2 = std::numeric_limits<Real>::max();
487 for (int dj = -R; dj <= R; ++dj) {
488 const int n = cj + dj;
489 if (n < 0 || n >= Ny) { continue; }
490 for (int di = -R; di <= R; ++di) {
491 int m = ci + di;
492 if (periodic) { m = ((m % Nx) + Nx) % Nx; }
493 else if (m < 0 || m >= Nx) { continue; }
494 const Real v = src(m, n, 0);
495 if (!is_valid(v)) { continue; }
496 Real dx = Real(m) - xf;
497 if (periodic) { dx -= Real(Nx) * std::round(dx / Real(Nx)); }
498 const Real dy = Real(n) - yf;
499 const Real d2 = dx*dx + dy*dy;
500 if (d2 < best_d2) { best_d2 = d2; best = v; }
501 }
502 }
503 d(i,j,k) = best;
504 });
505 }

510 const std::vector<long long>& field_shape) const
511{
512 LatLonRegridOptions opts;
513 opts.lon_name = lon_name;
514 opts.lat_name = lat_name;
515 const SourceGrid g = find_source_grid(file_name, field_name, field_shape, opts, "");
516 auto same = [] (const std::vector<double>& a, const std::vector<double>& b) {
517 if (a.size() != b.size()) { return false; }
518 for (std::size_t m = 0; m < a.size(); ++m) {
519 if (std::abs(a[m] - b[m]) > 1.e-9 * std::max(1.0, std::abs(a[m]))) { return false; }
520 }
521 return true;
522 };
523 return same(g.lon, lon_file) && same(g.lat, lat_file) &&
524 g.has_fill == static_cast<int>(has_fill) && g.has_missing == static_cast<int>(has_missing) &&
525 (!has_fill || g.fill_value == fill_value) && (!has_missing || g.missing_value == missing_value) &&
526 g.scale_factor == scale_factor && g.add_offset == add_offset;
527}
