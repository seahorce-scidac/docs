63 // MFIter::nodaltilebox calls surroundingNodes(dir) and only strips the high node when the
64 // tile stops short of the valid high end, so a box on the domain high edge legitimately
65 // ends at domain.bigEnd(dir)+1 -- the same convention as bx_xhi_face.setSmall(0,dom_hi.x+1)
66 // in the boundary conditions. Comparing against the cell-centered domain.bigEnd(dir) would
67 // never match there, and would wrongly match a box whose last cell is domain hi minus one.
68 const int nodal_lo = domain.smallEnd(dir);
69 const int nodal_hi = domain.bigEnd(dir) + 1;
70
71 // Independent tests, not else-if: a box spanning the domain in dir sheds both edge faces.
72 if (bx.smallEnd(dir) == nodal_lo) {
73 bx.growLo(dir,-1);
74 }
75 if (bx.bigEnd(dir) == nodal_hi) {
76 bx.growHi(dir,-1);
77 }
