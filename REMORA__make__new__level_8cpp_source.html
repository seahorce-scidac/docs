16/**
17 * Every box has to span the whole water column; see the amr.max_grid_size_z default in
18 * main.cpp. Abort with the reason rather than fail somewhere in a vertical sweep.
19 */
20void check_full_columns (int lev, const BoxArray& ba, const Geometry& geom)
21{
22 const int nz = geom.Domain().length(2);
23 for (int i = 0; i < ba.size(); ++i) {
24 if (ba[i].length(2) != nz) {
25 amrex::Abort("Level " + std::to_string(lev) + " has a box " + std::to_string(ba[i].length(2))
26 + " cells tall in a column of " + std::to_string(nz) + ". Boxes must"
27 " hold whole water columns: set amr.max_grid_size_z to at least the"
28 " number of vertical cells, which is its default.");
29 }
30 }

33/**
34 * CPU builds tile in 16x16 columns by default (main.cpp), and the threads of a rank share
35 * the tiles of its boxes. When tiling is turned off (remora.omp_tile_size at least the box
36 * size), the threads share whole boxes instead, so a rank with fewer boxes than threads
37 * leaves threads idle. Say so, naming the knobs that fix it, whenever a level's grids
38 * start leaving threads idle on some rank; grids that keep doing so are not reported again.
39 *
40 * @param[inout] warned whether the level's previous grids left threads idle; reset when
41 * the level is removed
42 */
43void warn_if_fewer_boxes_than_threads (int lev, const BoxArray& ba, const DistributionMapping& dm,
44 int& warned)
45{
46#ifdef AMREX_USE_OMP
47 // Tiling is on when a tile is smaller than some box in x or y; inputs often write
48 // "1024 1024 1024" to mean no tiling, so compare with the boxes, not a fixed value.
49 const int nthreads = OpenMP::get_max_threads();
50 const IntVect& tile_size = FabArrayBase::mfiter_tile_size;
51 int nlocal = 0;
52 bool tiled = false;
53 const int myproc = ParallelDescriptor::MyProc();
54 for (int i = 0; i < ba.size(); ++i) {
55 if (dm[i] == myproc) { ++nlocal; }
56 tiled = tiled || ba[i].length(0) > tile_size[0] || ba[i].length(1) > tile_size[1];
57 }
58 // Each rank judges its own boxes against its own threads. Collective: every rank
59 // reaches it, whatever its thread count, so the result is the same on all of them.
60 int idle = (nthreads > 1 && !tiled && nlocal < nthreads) ? 1 : 0;
61 ParallelDescriptor::ReduceIntMax(idle);
62 if (idle && !warned) {
63 amrex::Print() << "WARNING: level " << lev << " leaves OpenMP threads idle: some rank"
64 << " has fewer boxes than threads, and without tiling the threads"
65 << " share whole boxes. Lower amr.max_grid_size (or"
66 << " amr.max_grid_size_x/y) to give every rank at least as many"
67 << " boxes as threads, or set remora.omp_tile_size so the threads"
68 << " share the tiles of a box.\n";
69 }
70 warned = idle;
71#else
72 amrex::ignore_unused(lev, ba, dm, warned);
73#endif
74}

214 FillCoarsePatch(lev, time, vec_ru[lev].get(), &crse_ru, xvel_bc(),

216 FillCoarsePatch(lev, time, vec_rv[lev].get(), &crse_rv, yvel_bc(),
217 BdyVars::null,icomp,false);
218 FillCoarsePatch(lev, time, vec_ru2d[lev].get(), &crse_ru2d, xvel_bc(),
219 BdyVars::null,icomp,false);
220 FillCoarsePatch(lev, time, vec_rv2d[lev].get(), &crse_rv2d, yvel_bc(),
221 BdyVars::null,icomp,false);
222 }
223 scale_rhs_by_area(lev, *vec_ru[lev], 0, false);
224 scale_rhs_by_area(lev, *vec_rv[lev], 1, false);
225 scale_rhs_by_area(lev, *vec_ru2d[lev], 0, false);
226 scale_rhs_by_area(lev, *vec_rv2d[lev], 1, false);
227 }

229 stretch_transform(lev);
230
231 init_set_vmix(lev);
232 set_hmixcoef(lev);
233 set_coriolis(lev);
234
235#ifdef REMORA_USE_NETCDF
236 if (solverChoice.do_rivers) {
237 init_riv_pos_from_netcdf(lev);
238 }
239#endif
240
241 // ********************************************************************************************
242 // If we are making a new level then the FillPatcher for this level hasn't been allocated yet
243 // ********************************************************************************************
244 if (cf_width >= 0) {
245 Construct_REMORAFillPatchers(lev);
246 Define_REMORAFillPatchers(lev);
247 }
248 define_flux_register(lev);

250#ifdef REMORA_USE_PARTICLES
251 // particleData.Redistribute();
252#endif

1128/**
1129 * The metric products the time step needs, computed once from pm and pn rather than
1130 * with divides inside every kernel (ROMS metrics.F). Each value uses the expression, and
1131 * operand order, of the kernels it replaces, so they read identical bits.
1132 *
1133 * @param[in ]lev level to operate on
1134 */
1135void
1136REMORA::set_derived_metrics (int lev)
1137{
1138#ifdef AMREX_USE_OMP
1139#pragma omp parallel if (Gpu::notInLaunchRegion())
1140#endif
1141 for ( MFIter mfi(*vec_metric_r[lev], TilingIfNotGPU()); mfi.isValid(); ++mfi )
1142 {
1143 Array4<const Real> const& pm = vec_pm[lev]->const_array(mfi);
1144 Array4<const Real> const& pn = vec_pn[lev]->const_array(mfi);
1145 Array4<Real> const& mr = vec_metric_r[lev]->array(mfi);
1146 Array4<Real> const& mu = vec_metric_u[lev]->array(mfi);
1147 Array4<Real> const& mv = vec_metric_v[lev]->array(mfi);
1148 Array4<Real> const& mp = vec_metric_p[lev]->array(mfi);
1149
1150 ParallelFor(mfi.growntilebox(vec_metric_r[lev]->nGrowVect()),
1151 [=] AMREX_GPU_DEVICE (int i, int j, int)
1152 {
1153 mr(i,j,0,MetricR::pmn) = pm(i,j,0)*pn(i,j,0);
1154 mr(i,j,0,MetricR::omn) = one/(pm(i,j,0)*pn(i,j,0));
1155 mr(i,j,0,MetricR::om_r) = one/pm(i,j,0);
1156 mr(i,j,0,MetricR::on_r) = one/pn(i,j,0);
1157 mr(i,j,0,MetricR::pmon_r) = pm(i,j,0)/pn(i,j,0);
1158 mr(i,j,0,MetricR::pnom_r) = pn(i,j,0)/pm(i,j,0);
1159 });
1160
1161 ParallelFor(mfi.grownnodaltilebox(0, vec_metric_u[lev]->nGrowVect()),
1162 [=] AMREX_GPU_DEVICE (int i, int j, int)
1163 {
1164 const Real om_u = two/(pm(i,j,0)+pm(i-1,j,0));
1165 const Real on_u = two/(pn(i,j,0)+pn(i-1,j,0));
1166 mu(i,j,0,MetricU::om_u) = om_u;
1167 mu(i,j,0,MetricU::on_u) = on_u;
1168 mu(i,j,0,MetricU::omn_u) = om_u*on_u;
1169 mu(i,j,0,MetricU::pmon_u) = (pm(i-1,j,0)+pm(i,j,0))/(pn(i-1,j,0)+pn(i,j,0));
1170 mu(i,j,0,MetricU::pmn4_u) = (pm(i,j,0)+pm(i-1,j,0))*(pn(i,j,0)+pn(i-1,j,0));
1171 });
1172
1173 ParallelFor(mfi.grownnodaltilebox(1, vec_metric_v[lev]->nGrowVect()),
1174 [=] AMREX_GPU_DEVICE (int i, int j, int)
1175 {
1176 const Real om_v = two/(pm(i,j,0)+pm(i,j-1,0));
1177 const Real on_v = two/(pn(i,j,0)+pn(i,j-1,0));
1178 mv(i,j,0,MetricV::om_v) = om_v;
1179 mv(i,j,0,MetricV::on_v) = on_v;
1180 mv(i,j,0,MetricV::omn_v) = om_v*on_v;
1181 mv(i,j,0,MetricV::pnom_v) = (pn(i,j-1,0)+pn(i,j,0))/(pm(i,j-1,0)+pm(i,j,0));
1182 mv(i,j,0,MetricV::pmn4_v) = (pm(i,j,0)+pm(i,j-1,0))*(pn(i,j,0)+pn(i,j-1,0));
1183 });
1184
1185 Box pbx = mfi.tilebox(IntVect(1,1,0), vec_metric_p[lev]->nGrowVect());
1186 ParallelFor(pbx, [=] AMREX_GPU_DEVICE (int i, int j, int)
1187 {
1188 const Real sum_pm = pm(i-1,j-1,0)+pm(i-1,j,0)+pm(i,j-1,0)+pm(i,j,0);
1189 const Real sum_pn = pn(i-1,j-1,0)+pn(i-1,j,0)+pn(i,j-1,0)+pn(i,j,0);
1190 mp(i,j,0,MetricP::om_p) = Real(4.0)/sum_pm;
1191 mp(i,j,0,MetricP::on_p) = Real(4.0)/sum_pn;
1192 mp(i,j,0,MetricP::pmon_p) = sum_pm/sum_pn;
1193 mp(i,j,0,MetricP::pnom_p) = sum_pn/sum_pm;
1194 });
1195 }
1196}

1198/**
1199 * Fill a refined level's coordinate array from the coarser level, ghost cells included.
1200 *
1201 * FillCoarsePatch would finish with the physical boundary condition pm and pn want, which
1202 * copies the edge value into every ghost cell outside the domain and averages the
1203 * corners; for a coordinate that puts those cells on top of the edge rather than where
1204 * they are. Here the ghost cells outside the domain are interpolated like the rest, from
1205 * the coarse level's own ghost cells, which hold the coordinates continued past the edge
1206 * (extrapolate_metric_to_physical_boundaries with linear on level 0, and this function on
1207 * the levels between). The coarse level has NGROW+1 ghost cells, enough for the stencil
1208 * at any refinement ratio from 2 up. The interpolation is conservative-linear, which
1209 * reproduces a linear field exactly.
1210 *
1211 * The coarse data is copied without periodicity. Across a periodic edge, FillBoundary has
1212 * already put the wrapped values in the coarse ghost cells, but a ghost cell that also lies
1213 * beyond a wall (a corner where the seam meets a non-periodic face) keeps the value continued
1214 * from the file's ring, while its periodic image keeps the other side's. A periodic copy
1215 * would give that cell both as sources, and which one wins is not defined: on a GPU it varies
1216 * from run to run (#775). Copied without periodicity, each cell comes from the coarse level's
1217 * own ghost cell. The fine level's ghost cells across a periodic edge are then refilled from
1218 * its valid cells, wrapped as on level 0.
1219 *
1220 * @param[in ] lev the refined level
1221 * @param[inout] fine its coordinate array
1222 * @param[in ] crse the coarser level's
1223 */
1224void
1225REMORA::interp_grid_coords_from_coarse (int lev, MultiFab& fine, const MultiFab& crse)
1226{
1227 AMREX_ASSERT(lev > 0);
1228 Vector<BCRec> interior_bcs(1);
1229 for (int dir = 0; dir < AMREX_SPACEDIM; ++dir) {
1230 interior_bcs[0].setLo(dir, BCType::int_dir);
1231 interior_bcs[0].setHi(dir, BCType::int_dir);
1232 }
1233 const IntVect ng = fine.nGrowVect();
1234 const Geometry& cg = geom[lev-1];
1235 const Geometry crse_geom_nonperiodic(cg.Domain(), cg.ProbDomain(), cg.Coord(),
1236 Array<int,AMREX_SPACEDIM>{AMREX_D_DECL(0,0,0)});
1237 amrex::InterpFromCoarseLevel(fine, ng, ng, crse, 0, 0, 1, crse_geom_nonperiodic, geom[lev],
1238 ref_ratio[lev-1], &cell_cons_interp, interior_bcs, 0);
1239 fine.FillBoundary(geom[lev].periodicity());
1240}

1390/**
1391 * Fill a refined level's coordinate array, ghost cells included, by bilinear interpolation of
1392 * the parent's array at the same staggering. The parent is read only on the domain plus the
1393 * one-cell ring a grid file supplies (no ring along a nodal or periodic direction); points
1394 * beyond that extrapolate linearly from the two nearest values. A coordinate linear in the
1395 * index is reproduced exactly, so a uniform grid keeps (i+1/2)/pm.
1396 *
1397 * @param[in ] lev the refined level
1398 * @param[inout] fine its coordinate array
1399 * @param[in ] crse the parent's array at the same staggering
1400 */
1401void
1402REMORA::interp_staggered_coords_from_coarse (int lev, MultiFab& fine, const MultiFab& crse)
1403{
1404 AMREX_ASSERT(lev > 0);
1405 const IntVect stag = fine.ixType().toIntVect();
1406 const IntVect rr(ref_ratio[lev-1][0], ref_ratio[lev-1][1], 1);
1407
1408 Box filled = amrex::convert(makeSlab(geom[lev-1].Domain(), 2, 0), stag);
1409 for (int dir = 0; dir < 2; ++dir) {
1410 if (stag[dir] == 0 && !geom[lev-1].isPeriodic(dir)) { filled.grow(dir, 1); }
1411 }
1412
1413 // The parent on the refined layout, wide enough for the stencils of the fine ghost cells
1414 const IntVect ng_c(fine.nGrow(0) / rr[0] + 2, fine.nGrow(1) / rr[1] + 2, 0);
1415 MultiFab crse_on_fine(amrex::coarsen(fine.boxArray(), rr), fine.DistributionMap(), 1, ng_c);
1416 crse_on_fine.setVal(zero);
1417 crse_on_fine.ParallelCopy(crse, 0, 0, 1, crse.nGrowVect(), ng_c);
1418
1419 const Real sx = Real(0.5) * Real(1 - stag[0]);
1420 const Real sy = Real(0.5) * Real(1 - stag[1]);
1421 const int rx = rr[0], ry = rr[1];
1422
1423 for (MFIter mfi(fine); mfi.isValid(); ++mfi) {
1424 const Box use = crse_on_fine[mfi].box() & filled;
1425 AMREX_ALWAYS_ASSERT(use.length(0) >= 2 && use.length(1) >= 2);
1426 const int ilo = use.smallEnd(0), ihi = use.bigEnd(0) - 1;
1427 const int jlo = use.smallEnd(1), jhi = use.bigEnd(1) - 1;
1428 Array4<Real> const& f = fine.array(mfi);
1429 Array4<Real const> const& c = crse_on_fine.const_array(mfi);
1430
1431 ParallelFor(mfi.fabbox(), [=] AMREX_GPU_DEVICE (int i, int j, int k)
1432 {
1433 // Position in the parent's index space, measured from its first point
1434 const Real tx = (Real(i) + sx) / Real(rx) - sx;
1435 const Real ty = (Real(j) + sy) / Real(ry) - sy;
1436 const int ic = amrex::min(amrex::max(static_cast<int>(std::floor(tx)), ilo), ihi);
1437 const int jc = amrex::min(amrex::max(static_cast<int>(std::floor(ty)), jlo), jhi);
1438 const Real wx = tx - Real(ic);
1439 const Real wy = ty - Real(jc);
1440 f(i,j,k) = (one - wx) * (one - wy) * c(ic ,jc ,k) + wx * (one - wy) * c(ic+1,jc ,k)
1441 + (one - wx) * wy * c(ic ,jc+1,k) + wx * wy * c(ic+1,jc+1,k);
1442 });
1443 }
