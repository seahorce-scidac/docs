35/**
36 * \brief NDArray is the datatype designed to hold any data, including scalars, multidimensional
37 * arrays, that read from the NetCDF file.
38 *
39 * The data read from NetCDF file are stored in a continuous memory, and the data layout is described
40 * by using a vector (shape). AMRex Box can be constructed using the data shape information, and MultiFab
41 * data array can be setup using the data that stored in the NDArray.
42 */
43template <typename DataType>
44struct NDArray
45{
46 using DType = typename std::remove_const<DataType>::type;
47
48 /** \brief constructor */
49 explicit NDArray (const std::string vname, const std::vector<MPI_Offset>& vshape)
50 : name{vname}, shape{vshape},
51 data{std::shared_ptr<DType[]>(new DType[this->ndim()],
52 std::default_delete<DType[]>())} {}
53
54 /** \brief default constructor */
55 NDArray () : name{"null"}, data{nullptr} {}
56
57 /** \brief get the data pointer */
58 decltype(auto) get_data () {
59 return data.get();
60 }
61
62 /** \brief get the variable name */
63 std::string get_vname () {
64 return name;
65 }
66
67 /** \brief get the variable data shape */
68 std::vector<MPI_Offset> get_vshape () {
69 return shape;
70 }
71
72 /** \brief return the total number of data */
73 size_t ndim () {
74 size_t num = 1;
75 int isize = static_cast<int>(shape.size());
76 for (auto i=0; i<isize; ++i) num *= shape[i];
77 return num;
78 }
79
80 /** \brief set the data shape information */
81 void set_vshape (std::vector<MPI_Offset> vshape) {
82 shape = vshape;
83 }
84
85 private:
86 std::string name;
87 std::vector<MPI_Offset> shape;
88 std::shared_ptr<DType[]> data;
89};

91/** \brief Read in data from netcdf file and save to data arrays
92 *
93 * Only the I/O processor opens the file and fills \p arrays; the other ranks
94 * return at once, and the caller broadcasts what it needs.
95 *
96 * @param[in ] fname file name to read from
97 * @param[in ] names field names to read
98 * @param[inout] arrays vector of data arrays to fill
99 * @param[in ] one_time whether to read in a single time step of data
100 * @param[in ] fill_time if reading a single time, what time step to read from the file?
101 */
102template<typename DType>
103void ReadNetCDFFile (const std::string& fname, amrex::Vector<std::string> names,
104 amrex::Vector<NDArray<DType> >& arrays, bool one_time=false, int fill_time=0)
105{
106 BL_PROFILE("ReadNetCDFFile()");
107 AMREX_ASSERT(arrays.size() == names.size());

109 if (amrex::ParallelDescriptor::IOProcessor())
110 {
111 auto ncf = ncutils::NCFile::open_local(fname);
112
113 /*
114 // get the dimension information
115 int Time = static_cast<int>(ncf.dim("Time").len());
116 int DateStrLen = static_cast<int>(ncf.dim("DateStrLen").len());
117 int west_east = static_cast<int>(ncf.dim("west_east").len());
118 int south_north = static_cast<int>(ncf.dim("south_north").len());
119 int bottom_top = static_cast<int>(ncf.dim("bottom_top").len());
120 int bottom_top_stag = static_cast<int>(ncf.dim("bottom_top_stag").len());
121 int west_east_stag = static_cast<int>(ncf.dim("west_east_stag").len());

125// amrex::Print() << "Reading the dimensions from the netcdf file " << "\n";

127 for (auto n=0; n<arrays.size(); ++n) {
128 // read the data from NetCDF file
129 std::string vname_to_write = names[n];
130 std::string vname_to_read = names[n];
131
132 if (!ncf.has_var(vname_to_read)) {
133 amrex::Print() << "Variable name " << vname_to_read << " not found!" << std::endl;

149 arrays[n] = NDArray<DType>(vname_to_read,shape);

157 // amrex::Print() << dim << ", " ;
158 // amrex::Print() << ")" << std::endl;
159
160 ncf.var(vname_to_read).get_all(dataPtr, start, shape);
161 }
162 ncf.close();
163 }

218/**
219 * \brief Where fill_fab_from_arrays puts a variable's data, before the shift by
220 * the domain's lower corner.
221 *
222 * @param[in ] var_name variable name, which sets the staggering
223 * @param[in ] NC_dim_type dimension type of the variable in the file
224 * @param[in ] shape the variable's shape in the file (any time count)
225 * @param[in ] custom_ngrow grow cells of the FAB, as in BuildFABsFromNetCDFFile
226 * @param[out] nsx, nsy, nsz extents of the file data in x, y and z
227 * @returns the FAB's box. The data fill the nsx x nsy x nsz corner of it that
228 * starts at its lower corner.
229 */
230inline amrex::Box
231nc_fab_box (const std::string& var_name, NC_Data_Dims_Type NC_dim_type,
232 const std::vector<MPI_Offset>& shape, amrex::IntVect custom_ngrow,
233 int& nsx, int& nsy, int& nsz)
234{
235 nsx = nsy = nsz = 0;
236 if (NC_dim_type == NC_Data_Dims_Type::SN_WE) {
237 nsz = 1;
238 nsy = shape[0];
239 nsx = shape[1];
240 } else if (NC_dim_type == NC_Data_Dims_Type::BT_SN_WE) {
241 nsz = shape[0];
242 nsy = shape[1];
243 nsx = shape[2];
244 } else if (NC_dim_type == NC_Data_Dims_Type::Time_SN_WE) {
245 nsz = 1;
246 nsy = shape[1];
247 nsx = shape[2];
248 } else if (NC_dim_type == NC_Data_Dims_Type::Time_BT_SN_WE) {
249 nsz = shape[1];
250 nsy = shape[2];
251 nsx = shape[3];
252 } else if (NC_dim_type == NC_Data_Dims_Type::Time_BT_Riv) {
253 nsz = shape[1]; // number of z-coords
254 nsy = 1; // dummy
255 nsx = shape[2]; // number of rivers
256 } else if (NC_dim_type == NC_Data_Dims_Type::Time_Riv) {
257 nsz = 1; // number of z-coords
258 nsy = 1; // dummy
259 nsx = shape[1]; // number of rivers
260 } else if (NC_dim_type == NC_Data_Dims_Type::BT_Riv) {
261 nsz = shape[0]; // number of z-coords
262 nsy = 1; // dummy
263 nsx = shape[1]; // number of rivers
264 } else {
265 amrex::Abort("Dont know this NC_Data_Dims_Type");
266 }
267
268 amrex::Box my_box;
269
270 if (var_name == "u" || var_name == "ubar" || var_name == "mask_u" || var_name == "x_u" ||
271 var_name == "y_u" || var_name == "sustr")
272 {
273 my_box.setSmall(amrex::IntVect(0,-1,0));
274 my_box.setBig(amrex::IntVect(nsx-1,nsy-2,nsz-1));
275 my_box.setType(amrex::IndexType(amrex::IntVect(1,0,0)));
276 }
277 else if (var_name == "v" || var_name == "vbar" || var_name == "mask_v" || var_name == "x_v" ||
278 var_name == "y_v" || var_name == "svstr")
279 {
280 my_box.setSmall(amrex::IntVect(-1,0,0));
281 my_box.setBig(amrex::IntVect(nsx-2,nsy-1,nsz-1));
282 my_box.setType(amrex::IndexType(amrex::IntVect(0,1,0)));
283 }
284 else if (var_name == "mask_psi" || var_name == "x_psi" || var_name == "y_psi" ||
285 var_name == "lon_psi" || var_name == "lat_psi")
286 {
287 my_box.setSmall(amrex::IntVect(0,0,0));
288 my_box.setBig(amrex::IntVect(nsx-1,nsy-1,nsz-1));
289 my_box.setType(amrex::IndexType(amrex::IntVect(1,1,0)));
290 // River data has no ghost cells
291 } else if (NC_dim_type == NC_Data_Dims_Type::Time_BT_Riv ||
292 NC_dim_type == NC_Data_Dims_Type::Time_Riv ||
293 NC_dim_type == NC_Data_Dims_Type::BT_Riv) {
294 my_box.setSmall(amrex::IntVect(0,0,0));
295 my_box.setBig(amrex::IntVect(nsx-1,nsy-1,nsz-1));
296 }
297 else // everything cell-centered -- these have one ghost cell
298 {
299 my_box.setSmall(amrex::IntVect(-1,-1,0));
300 my_box.setBig(amrex::IntVect(nsx-2,nsy-2,nsz-1));
301 }
302 // Adjust box size if a non-standard number of grow cells
303 my_box.grow(custom_ngrow - amrex::IntVect(1,1,0));
304 return my_box;
305}

307/**
308 * \brief Helper function for reading data from NetCDF file into a
309 * provided FAB.
310 *
311 * @param iv Index for which variable we are going to fill
312 * @param nc_arrays Arrays of data from NetCDF file
313 * @param var_name Variable name
314 * @param NC_dim_type Dimension type for the variable as stored in the NetCDF file
315 * @param temp FAB where we store the variable data from the NetCDF Arrays
316 * @param custom_ngrow Custom number of grow cells for the FAB
317 */
318template<class FAB,typename DType>
319void
320fill_fab_from_arrays (int iv,
321 amrex::Vector<NDArray<amrex::Real>>& nc_arrays,
322 const std::string& var_name,
323 NC_Data_Dims_Type& NC_dim_type,
324 FAB& temp, amrex::IntVect custom_ngrow)
325{
326 int nsx = 0, nsy = 0, nsz = 0;
327 const amrex::Box my_box = nc_fab_box(var_name, NC_dim_type, nc_arrays[iv].get_vshape(),
328 custom_ngrow, nsx, nsy, nsz);
329
330#ifdef AMREX_USE_GPU
331 // Make sure temp lives on CPU since nc_arrays lives on CPU only
332 temp.resize(my_box,1,amrex::The_Pinned_Arena());
333#else
334 temp.resize(my_box,1);
335#endif
336 amrex::Array4<DType> fab_arr = temp.array();
337
338 int ioff = temp.box().smallEnd()[0];
339 int joff = temp.box().smallEnd()[1];
340
341 auto num_pts_in_box = my_box.numPts();
342 auto num_pts_in_src = nsx*nsy*nsz;
343 // amrex::Print() <<" NUM PTS IN BOX " << num_pts_in_box << std::endl;
344 // amrex::Print() <<" NUM PTS IN SRC " << num_pts_in_src << std::endl;
345 AMREX_ALWAYS_ASSERT(num_pts_in_box >= num_pts_in_src);
346
347 for (int n(0); n < num_pts_in_src; ++n) {
348 int nplane = nsx*nsy;
349 int k = n / nplane;
350 int j = (n - k*nplane) / nsx + joff;
351 int i = n - k*nplane - (j-joff) * nsx + ioff;
352 fab_arr(i,j,k,0) = static_cast<DType>(*(nc_arrays[iv].get_data()+n));
353 }
354}

356/**
357 * \brief Function to read NetCDF variables and fill the corresponding Array4's
358 *
359 * @param fname Name of the NetCDF file to be read
360 * @param nc_var_names Variable names in the NetCDF file
361 * @param NC_dim_types NetCDF data dimension types
362 * @param fab_vars Fab data we are to fill
363 * @param one_time Whether to read in a single time step from file
364 * @param fill_time What time step to read from file, if only reading one step
365 * @param custom_ngrow Number of grow cells to read in the fab; default is based off of one cell in each of x and y directions
366 */
367template<class FAB,typename DType>
368void
369BuildFABsFromNetCDFFile (const amrex::Box& domain,
370 const std::string &fname,
371 amrex::Vector<std::string> nc_var_names,
372 amrex::Vector<enum NC_Data_Dims_Type> NC_dim_types,
373 amrex::Vector<FAB*> fab_vars,
374 bool one_time=false, int fill_time=0,
375 amrex::IntVect custom_ngrow=amrex::IntVect(1,1,0))
376{
377 BL_PROFILE("BuildFABsFromNetCDFFile()");
378 int ioproc = amrex::ParallelDescriptor::IOProcessorNumber(); // I/O rank
379
380 amrex::Vector<NDArray<amrex::Real>> nc_arrays(nc_var_names.size());
381
382
383 ReadNetCDFFile(fname, nc_var_names, nc_arrays, one_time, fill_time); // this is filled only on proc 0
384
385 for (int iv = 0; iv < nc_var_names.size(); iv++)
386 {
387 FAB tmp;
388 if (amrex::ParallelDescriptor::IOProcessor()) {
389 fill_fab_from_arrays<FAB,DType>(iv, nc_arrays, nc_var_names[iv], NC_dim_types[iv], tmp, custom_ngrow);
390 }
391
392 int ncomp = tmp.nComp();
393 amrex::Box box = tmp.box();
394
395 amrex::ParallelDescriptor::Bcast(&box, 1, ioproc);
396 amrex::ParallelDescriptor::Bcast(&ncomp, 1, ioproc);
397
398 if (!amrex::ParallelDescriptor::IOProcessor()) {
399#ifdef AMREX_USE_GPU
400 tmp.resize(box,ncomp,amrex::The_Pinned_Arena());
401#else
402 tmp.resize(box,ncomp);
403#endif
404 }
405
406 amrex::ParallelDescriptor::Bcast(tmp.dataPtr(), tmp.size(), ioproc);
407
408 // Shift box by the domain lower corner
409 amrex::Box fab_bx = tmp.box();
410 amrex::Dim3 dom_lb = lbound(domain);
411 fab_bx += amrex::IntVect(dom_lb.x,dom_lb.y,dom_lb.z);
412 // fab_vars points to data on device
413 fab_vars[iv]->resize(fab_bx,1);
414#ifdef AMREX_USE_GPU
415 amrex::Gpu::copy(amrex::Gpu::hostToDevice,
416 tmp.dataPtr(), tmp.dataPtr() + tmp.size(),
417 fab_vars[iv]->dataPtr());
418#else
419 // Provided by BaseFab inheritance through FArrayBox
420 fab_vars[iv]->copy(tmp,tmp.box(),0,fab_bx,0,1);
421#endif
422 }
423}

425/**
426 * \brief Keeps the files BuildLocalFABsFromNetCDFFile opens open until the
427 * outermost session ends, so a run of readers opens each file once.
428 *
429 * Opening a NetCDF-4 file in parallel costs about a second on a thousand ranks,
430 * and the initialization routines read the grid file alone half a dozen times.
431 * Create one of these, on every rank, around such a run of readers; sessions
432 * nest, and the files close, collectively and in the order they were opened,
433 * when the outermost one is destroyed. A file is kept per communicator, so a
434 * reader over another communicator gets its own handle. Only for files that are
435 * read, not written, while it lasts.
436 */
437class NCReadSession
438{
439public:
440 NCReadSession () { ++state().depth; }
441 ~NCReadSession ()
442 {
443 if (--state().depth == 0) {
444 for (auto& f : state().files) { f.file->close(); }
445 state().files.clear();
446 }
447 }
448 NCReadSession (const NCReadSession&) = delete;
449 NCReadSession& operator= (const NCReadSession&) = delete;
450
451 //! Whether a session is open on this rank
452 static bool active () { return state().depth > 0; }
453
454 //! The file, opened read-only over \p comm the first time it is asked for
455 static ncutils::NCFile& file (const std::string& fname, MPI_Comm comm)
456 {
457 auto& files = state().files;
458 for (auto& f : files) {
459 if (f.name == fname && f.comm == comm) { return *f.file; }
460 }
461 files.push_back(Entry{fname, comm, open_file(fname, comm)});
462 return *files.back().file;
463 }
464
465 //! Open \p fname read-only over \p comm, outside any session
466 static std::unique_ptr<ncutils::NCFile> open_file (const std::string& fname, MPI_Comm comm)
467 {
468 BL_PROFILE("NCReadSession::open_file()");
469 return std::make_unique<ncutils::NCFile>(ncutils::NCFile::open(fname, NC_NOWRITE, comm));
470 }
471
472private:
473 struct Entry {
474 std::string name;
475 MPI_Comm comm;
476 std::unique_ptr<ncutils::NCFile> file;
477 };
478 struct State {
479 int depth = 0;
480 std::vector<Entry> files; // in the order they were opened, the same on every rank
481 };
482 static State& state () { static State s; return s; }
483};

505/**
506 * \brief The boxes, cell-centered and with ghost cells, of every FAB this rank owns
507 * in each of \p mfs: the region BuildLocalFABsFromNetCDFFile has to read for a copy
508 * into them. Empty on a rank that owns none. Null entries are skipped.
509 *
510 * Boxes are merged while their bounding box holds no more cells than the two do
511 * together, so the same box in several MultiFabs, or a FAB and the neighbor whose
512 * ghost cells it overlaps, become one read; FABs far apart stay separate reads, so
513 * a rank never reads the space between them. With remora.v >= 1 the result is
514 * printed: the largest ratio over the ranks of cells to be read to cells owned.
515 *
516 * Collective over amrex::ParallelContext::CommunicatorSub() when remora.v >= 1.
517 */
518inline NCRegion
519nc_local_region (const amrex::Vector<const amrex::MultiFab*>& mfs)
520{
521 NCRegion boxes;
522 amrex::BoxList owned_boxes;
523 for (const amrex::MultiFab* mf : mfs) {
524 if (mf == nullptr) { continue; }
525 for (amrex::MFIter mfi(*mf, false); mfi.isValid(); ++mfi) {
526 const amrex::Box b = amrex::convert(mfi.fabbox(), amrex::IndexType::TheCellType());
527 boxes.push_back(b);
528 owned_boxes.push_back(b);
529 }
530 }
531 // The same FAB in several MultiFabs gives the same box: keep one of each first.
532 auto corners = [] (const amrex::Box& b) {
533 return std::array<int,6>{AMREX_D_DECL(b.smallEnd(0), b.smallEnd(1), b.smallEnd(2)),
534 AMREX_D_DECL(b.bigEnd(0), b.bigEnd(1), b.bigEnd(2))};
535 };
536 std::sort(boxes.begin(), boxes.end(),
537 [&] (const amrex::Box& a, const amrex::Box& b) { return corners(a) < corners(b); });
538 boxes.erase(std::unique(boxes.begin(), boxes.end()), boxes.end());
539 // Then merge pairs whose bounding box adds no cells, until none is left.
540 for (bool merged = true; merged; ) {
541 merged = false;
542 for (int i = 0; i < static_cast<int>(boxes.size()); ++i) {
543 for (int j = i + 1; j < static_cast<int>(boxes.size()); ) {
544 const amrex::Box bb = amrex::Box(boxes[i]).minBox(boxes[j]);
545 if (bb.numPts() <= boxes[i].numPts() + boxes[j].numPts()) {
546 boxes[i] = bb;
547 boxes.erase(boxes.begin() + j);
548 merged = true;
549 } else {
550 ++j;
551 }
552 }
553 }
554 }
555
556 int verbose = 0;
557 amrex::ParmParse("remora").query("v", verbose);
558 if (verbose >= 1) {
559 // Cells this rank's FABs cover, each counted once although several
560 // MultiFabs, or a FAB's ghost cells and its neighbor's, may overlap.
561 long long owned = 0;
562 if (!owned_boxes.isEmpty()) {
563 amrex::BoxArray ba(owned_boxes);
564 ba.removeOverlap();
565 owned = ba.numPts();
566 }
567 long long read = 0;
568 for (const auto& b : boxes) { read += b.numPts(); }
569 struct { double ratio; int rank; } mine, worst;
570 mine.ratio = (owned > 0) ? double(read) / double(owned) : 1.0;
571 mine.rank = amrex::ParallelContext::MyProcSub();
572 MPI_Comm comm = amrex::ParallelContext::CommunicatorSub();
573 MPI_Allreduce(&mine, &worst, 1, MPI_DOUBLE_INT, MPI_MAXLOC, comm);
574 double mean = mine.ratio;
575 amrex::ParallelAllReduce::Sum(mean, comm);
576 mean /= amrex::ParallelContext::NProcsSub();
577 long long worst_cells[3] = {read, owned, static_cast<long long>(boxes.size())};
578 MPI_Bcast(worst_cells, 3, MPI_LONG_LONG, worst.rank, comm);
579 amrex::Print() << "NetCDF read region: up to " << worst.ratio
580 << " times the cells owned, on rank " << worst.rank << " ("
581 << worst_cells[0] << " read in " << worst_cells[2] << " boxes, "
582 << worst_cells[1] << " owned); mean " << mean << " over ranks"
583 << std::endl;
584 }
585 return boxes;
586}

602/**
603 * \brief Read the part of each variable that this rank needs, collectively.
604 *
605 * Reads the same data as BuildFABsFromNetCDFFile, in the same index space, but
606 * each rank reads, and holds, only the file data inside the boxes of \p region
607 * (converted to each variable's staggering): one FAB per box, in an NCLocalFab.
608 * NCLocalFab::copy_to into a MultiFab FAB this rank owns gives the same values as
609 * a copy from BuildFABsFromNetCDFFile's FAB, when \p region came from
610 * nc_local_region for that MultiFab.
611 *
612 * Collective over amrex::ParallelContext::CommunicatorSub(): every rank calls it,
613 * including ranks with an empty \p region. Each variable is one collective read per
614 * box; ranks with fewer boxes than the most any rank has make empty reads to match,
615 * as a collective read needs. That works for every format PnetCDF opens, compressed
616 * NetCDF-4 included. River data, which is not laid out on the model grid, is not
617 * supported here.
618 *
619 * @param domain domain whose lower corner the file data starts at
620 * @param region cell-centered boxes of the domain this rank needs
621 * @param fname file to read
622 * @param nc_var_names variables to read
623 * @param NC_dim_types dimension type of each variable in the file
624 * @param fab_vars where to put each variable
625 * @param one_time whether to read a single time step
626 * @param fill_time time index to read when \p one_time
627 * @param custom_ngrow grow cells, as in BuildFABsFromNetCDFFile
628 * @param open_ncf \p fname, already open read-only over the communicator, for a
629 * caller that keeps it open between calls; opened here if null
630 */
631template<class FAB,typename DType>
632void
633BuildLocalFABsFromNetCDFFile (const amrex::Box& domain,
634 const NCRegion& region,
635 const std::string &fname,
636 amrex::Vector<std::string> nc_var_names,
637 amrex::Vector<enum NC_Data_Dims_Type> NC_dim_types,
638 amrex::Vector<NCLocalFab*> fab_vars,
639 bool one_time=false, int fill_time=0,
640 amrex::IntVect custom_ngrow=amrex::IntVect(1,1,0),
641 ncutils::NCFile* open_ncf=nullptr)
642{
643 BL_PROFILE("BuildLocalFABsFromNetCDFFile()");
644 MPI_Comm comm = amrex::ParallelContext::CommunicatorSub();
645 int nreads = static_cast<int>(region.size());
646 amrex::ParallelAllReduce::Max(nreads, comm);
647
648 // Inside an NCReadSession the file stays open for the next reader.
649 BL_PROFILE_VAR("BuildLocalFABsFromNetCDFFile::open", nc_open);
650 std::unique_ptr<ncutils::NCFile> own;
651 if (open_ncf == nullptr && !NCReadSession::active()) { own = NCReadSession::open_file(fname, comm); }
652 ncutils::NCFile& ncf = open_ncf ? *open_ncf : (own ? *own : NCReadSession::file(fname, comm));
653 BL_PROFILE_VAR_STOP(nc_open);
654
655 BL_PROFILE_VAR("BuildLocalFABsFromNetCDFFile::read", nc_read);
656 for (int iv = 0; iv < nc_var_names.size(); iv++)
657 {
658 const NC_Data_Dims_Type dim_type = NC_dim_types[iv];
659 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(dim_type == NC_Data_Dims_Type::Time_BT_SN_WE ||
660 dim_type == NC_Data_Dims_Type::Time_SN_WE ||
661 dim_type == NC_Data_Dims_Type::BT_SN_WE ||
662 dim_type == NC_Data_Dims_Type::SN_WE,
663 "BuildLocalFABsFromNetCDFFile reads gridded data only");
664 const bool has_time = (dim_type == NC_Data_Dims_Type::Time_BT_SN_WE ||
665 dim_type == NC_Data_Dims_Type::Time_SN_WE);
666 const bool has_z = (dim_type == NC_Data_Dims_Type::Time_BT_SN_WE ||
667 dim_type == NC_Data_Dims_Type::BT_SN_WE);
668
669 const auto var = ncf.var(nc_var_names[iv]);
670 const std::vector<MPI_Offset> shape = var.shape();
671 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
672 static_cast<int>(shape.size()) == int(has_time) + int(has_z) + 2,
673 "NetCDF variable " + nc_var_names[iv] + " in " + fname +
674 " does not have the dimensions expected of it");
675
676 // The same layout as BuildFABsFromNetCDFFile: the file data fill the
677 // corner of the FAB box that starts at its lower corner.
678 int nsx = 0, nsy = 0, nsz = 0;
679 amrex::Box fab_box = nc_fab_box(nc_var_names[iv], dim_type, shape, custom_ngrow,
680 nsx, nsy, nsz);
681 amrex::Box data_box(fab_box.smallEnd(),
682 fab_box.smallEnd() + amrex::IntVect(nsx-1, nsy-1, nsz-1),
683 fab_box.ixType());
684 AMREX_ALWAYS_ASSERT(fab_box.contains(data_box));
685 data_box.shift(domain.smallEnd());
686
687 NCLocalFab& out = *fab_vars[iv];
688 out.pieces.clear();
689 for (int ir = 0; ir < nreads; ++ir) {
690 const amrex::Box local = (ir < static_cast<int>(region.size()))
691 ? (data_box & amrex::convert(region[ir], data_box.ixType())) : amrex::Box();
692
693 // Hyperslab of the file holding `local`; empty when there is none.
694 std::vector<MPI_Offset> start(shape.size(), 0), count(shape.size(), 0);
695 int d = 0;
696 if (has_time) {
697 start[d] = one_time ? fill_time : 0;
698 count[d] = local.ok() ? 1 : 0;
699 ++d;
700 }
701 if (local.ok()) {
702 if (has_z) {
703 start[d] = local.smallEnd(2) - data_box.smallEnd(2);
704 count[d] = local.length(2);
705 }
706 start[d + int(has_z)] = local.smallEnd(1) - data_box.smallEnd(1);
707 count[d + int(has_z)] = local.length(1);
708 start[d + int(has_z) + 1] = local.smallEnd(0) - data_box.smallEnd(0);
709 count[d + int(has_z) + 1] = local.length(0);
710 }
711
712 // The file stores x fastest, then y, then z: the FAB's own order, so
713 // the hyperslab reads straight into it.
714 if (!local.ok()) {
715 amrex::Real dummy = 0;
716 var.get_all(&dummy, start, count);
717 continue;
718 }
719 out.pieces.emplace_back(local, 1);
720#ifdef AMREX_USE_GPU
721 // The pieces live on the device; read on the host and copy over.
722 FAB tmp(local, 1, amrex::The_Pinned_Arena());
723 var.get_all(tmp.dataPtr(), start, count);
724 amrex::Gpu::copy(amrex::Gpu::hostToDevice,
725 tmp.dataPtr(), tmp.dataPtr() + tmp.size(),
726 out.pieces.back().dataPtr());
727#else
728 var.get_all(out.pieces.back().dataPtr(), start, count);
729#endif
730 }
731 }
732 BL_PROFILE_VAR_STOP(nc_read);
733
734 BL_PROFILE_VAR("BuildLocalFABsFromNetCDFFile::close", nc_close);
735 if (own) { own->close(); }
736 BL_PROFILE_VAR_STOP(nc_close);
737}
