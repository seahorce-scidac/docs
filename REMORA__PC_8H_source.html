74// Particles are tracked vertically by a fractional w-level index s in [0,nz]: s = k + f puts
75// a particle the fraction f of the way from w-level k to w-level k+1 of its column, so s = 0
76// is the bed and s = nz the free surface. Moving at fixed s follows the terrain-following
77// levels, which is what keeps a particle in the water column over sloping bathymetry.
78
79/**
80 * Bilinear interpolation of a_arr at level a_k to horizontal position (a_x, a_y). a_nodal
81 * says whether the data are nodal in both x and y (like z_phys_nd, whose (i,j) is the low
82 * corner of cell (i,j)) or cell-centered in both (like pm and pn, stored at k = 0).
83 */
84AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
85amrex::Real interpolate_horizontal ( amrex::ParticleReal a_x, amrex::ParticleReal a_y,
86 int a_k, bool a_nodal,
87 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_plo,
88 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_dxi,
89 amrex::Array4<amrex::Real const> const& a_arr )
90{
91 const amrex::Real offset = a_nodal ? zero : myhalf;
92 const amrex::Real lx = static_cast<amrex::Real>((a_x-a_plo[0])*a_dxi[0]) - offset;
93 const amrex::Real ly = static_cast<amrex::Real>((a_y-a_plo[1])*a_dxi[1]) - offset;
94 const int i = static_cast<int>(amrex::Math::floor(lx));
95 const int j = static_cast<int>(amrex::Math::floor(ly));
96 const amrex::Real sx = lx - static_cast<amrex::Real>(i);
97 const amrex::Real sy = ly - static_cast<amrex::Real>(j);
98 return a_arr(i ,j ,a_k) * (one-sx) * (one-sy) +
99 a_arr(i+1,j ,a_k) * sx * (one-sy) +
100 a_arr(i ,j+1,a_k) * (one-sx) * sy +
101 a_arr(i+1,j+1,a_k) * sx * sy;
102}

104/** Height of w-level a_k at horizontal position (a_x, a_y), from the nodal heights a_z_nd (z_phys_nd) */
105AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
106amrex::Real level_height ( amrex::ParticleReal a_x, amrex::ParticleReal a_y, int a_k,
107 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_plo,
108 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_dxi,
109 amrex::Array4<amrex::Real const> const& a_z_nd )
110{
111 return interpolate_horizontal(a_x, a_y, a_k, true, a_plo, a_dxi, a_z_nd);
112}

114/**
115 * Fractional w-level index of height a_z in the column at (a_x, a_y). A height below the
116 * bed or above the free surface is clamped onto it. a_k, the cell the particle was last
117 * in, is where the search starts. If a_kcell is given, the cell the search ended in and the
118 * thickness of that cell at (a_x, a_y) are returned through a_kcell and a_dz, so a caller
119 * need not recompute the level heights.
120 */
121AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
122amrex::Real height_to_level_index ( amrex::ParticleReal a_x, amrex::ParticleReal a_y,
123 amrex::ParticleReal a_z, int a_k, int a_nz,
124 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_plo,
125 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_dxi,
126 amrex::Array4<amrex::Real const> const& a_z_nd,
127 int* a_kcell = nullptr, amrex::Real* a_dz = nullptr )
128{
129 const auto z = static_cast<amrex::Real>(a_z);
130 int k = amrex::max(0, amrex::min(a_k, a_nz-1));
131 amrex::Real zlo = level_height(a_x, a_y, k , a_plo, a_dxi, a_z_nd);
132 amrex::Real zhi = level_height(a_x, a_y, k+1, a_plo, a_dxi, a_z_nd);
133 while (z > zhi && k < a_nz-1) {
134 ++k;
135 zlo = zhi;
136 zhi = level_height(a_x, a_y, k+1, a_plo, a_dxi, a_z_nd);
137 }
138 while (z < zlo && k > 0) {
139 --k;
140 zhi = zlo;
141 zlo = level_height(a_x, a_y, k, a_plo, a_dxi, a_z_nd);
142 }
143 const amrex::Real f = amrex::max(zero, amrex::min(one, (z - zlo) / (zhi - zlo)));
144 if (a_kcell) {
145 *a_kcell = k;
146 *a_dz = zhi - zlo;
147 }
148 return static_cast<amrex::Real>(k) + f;
149}

151/** Cell holding fractional w-level index a_s; the free surface s = nz belongs to the top cell */
152AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
153int level_index_cell (amrex::Real a_s, int a_nz)
154{
155 return amrex::max(0, amrex::min(static_cast<int>(amrex::Math::floor(a_s)), a_nz-1));
156}

158/** Height of fractional w-level index a_s at (a_x, a_y); the inverse of height_to_level_index */
159AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
160amrex::Real level_index_to_height ( amrex::ParticleReal a_x, amrex::ParticleReal a_y,
161 amrex::Real a_s, int a_nz,
162 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_plo,
163 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_dxi,
164 amrex::Array4<amrex::Real const> const& a_z_nd )
165{
166 const int k = level_index_cell(a_s, a_nz);
167 const amrex::Real f = a_s - static_cast<amrex::Real>(k);
168 const amrex::Real zlo = level_height(a_x, a_y, k , a_plo, a_dxi, a_z_nd);
169 const amrex::Real zhi = level_height(a_x, a_y, k+1, a_plo, a_dxi, a_z_nd);
170 return zlo + f * (zhi - zlo);
171}

173/**
174 * Trilinear interpolation of a_arr to horizontal position (a_x, a_y) and fractional w-level
175 * index a_s. a_nodal gives a_arr's staggering. Data that is cell-centered in z is held
176 * constant over the half cells at the bed and the surface, so the vertical stencil stays
177 * within the valid cells [0,nz) (or w-levels [0,nz] for z-nodal data).
178 */
179AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
180amrex::Real interpolate_to_level_index ( amrex::ParticleReal a_x, amrex::ParticleReal a_y,
181 amrex::Real a_s, int a_nz, amrex::IntVect const& a_nodal,
182 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_plo,
183 amrex::GpuArray<amrex::Real,AMREX_SPACEDIM> const& a_dxi,
184 amrex::Array4<amrex::Real const> const& a_arr )
185{
186 const amrex::Real lx = static_cast<amrex::Real>((a_x-a_plo[0])*a_dxi[0])
187 - static_cast<amrex::Real>(!a_nodal[0]) * myhalf;
188 const amrex::Real ly = static_cast<amrex::Real>((a_y-a_plo[1])*a_dxi[1])
189 - static_cast<amrex::Real>(!a_nodal[1]) * myhalf;
190 const int i0 = static_cast<int>(amrex::Math::floor(lx));
191 const int j0 = static_cast<int>(amrex::Math::floor(ly));
192 const amrex::Real sx[] = {one - (lx - static_cast<amrex::Real>(i0)), lx - static_cast<amrex::Real>(i0)};
193 const amrex::Real sy[] = {one - (ly - static_cast<amrex::Real>(j0)), ly - static_cast<amrex::Real>(j0)};
194
195 const int kmax = a_nodal[2] ? a_nz : a_nz-1;
196 const amrex::Real lz = amrex::max(zero, amrex::min(static_cast<amrex::Real>(kmax),
197 a_s - static_cast<amrex::Real>(!a_nodal[2]) * myhalf));
198 const int k0 = amrex::min(static_cast<int>(amrex::Math::floor(lz)), amrex::max(kmax-1, 0));
199 const int k1 = amrex::min(k0+1, kmax);
200 const amrex::Real sz[] = {one - (lz - static_cast<amrex::Real>(k0)), lz - static_cast<amrex::Real>(k0)};
201
202 amrex::Real val = zero;
203 for (int jj = 0; jj < 2; ++jj) {
204 for (int ii = 0; ii < 2; ++ii) {
205 val += sx[ii] * sy[jj] * ( sz[0] * a_arr(i0+ii, j0+jj, k0)
206 + sz[1] * a_arr(i0+ii, j0+jj, k1) );
207 }
208 }
209 return val;
210}
