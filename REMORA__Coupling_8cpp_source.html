39// One pass over the wet cells (mask > 0.5) of one component of a 2D field, on
40// the valid region at the field's lowest k. Land cells hold zeros or whatever
41// the atmosphere sent over land, so an unmasked min/max says nothing about what
42// the ocean actually receives.
43//
44// min/max skip NaN, which nan_count reports: NaN in a min/max reduction is
45// kept or dropped depending on the reduction order, so the result would depend
46// on the decomposition. With no finite wet cell, has_finite() is false.
47//
48// bad_count counts the wet cells outside [lo, hi], NaN included; the test is
49// !(lo <= v <= hi) because every comparison with NaN is false. One of them is
50// named: the smallest flat (i,j) index, the same cell under any decomposition.
51// Collective.
52struct WetRangeCheck
53{
54 amrex::Long wet_cells;
55 amrex::Real min_value;
56 amrex::Real max_value;
57 amrex::Long bad_count;
58 amrex::Long nan_count;
59 amrex::IntVect example;
60 amrex::Real example_value;
61
62 bool has_finite () const { return min_value <= max_value; }
63};
64
65WetRangeCheck
66CheckWetRange (const amrex::MultiFab& mf, int comp, const amrex::MultiFab& mask,
67 amrex::Real lo_bound = -std::numeric_limits<amrex::Real>::max(),
68 amrex::Real hi_bound = std::numeric_limits<amrex::Real>::max())
69{
70 using namespace amrex;
71 const Box domain = mf.boxArray().minimalBox();
72 const Long nx = domain.length(0);
73 const int k0 = domain.smallEnd(2);
74 constexpr Long no_cell = std::numeric_limits<Long>::max();
75 constexpr Real big = std::numeric_limits<Real>::max();
76
77 ReduceOps<ReduceOpSum, ReduceOpSum, ReduceOpSum,
78 ReduceOpMin, ReduceOpMax, ReduceOpMin> ops;
79 ReduceData<Long, Long, Long, Real, Real, Long> data(ops);
80 using Tuple = typename decltype(data)::Type;
81 for (MFIter mfi(mf); mfi.isValid(); ++mfi) {
82 Box bx = mfi.validbox();
83 bx.makeSlab(2, k0);
84 const auto f = mf.const_array(mfi, comp);
85 const auto m = mask.const_array(mfi);
86 const auto dlo = lbound(domain);
87 ops.eval(bx, data,
88 [=] AMREX_GPU_DEVICE (int i, int j, int k) -> Tuple
89 {
90 const Real v = f(i,j,k);
91 const bool wet = m(i,j,k) > Real(0.5);
92 const bool bad = wet && !(v >= lo_bound && v <= hi_bound);
93 const bool nan = wet && (v != v);
94 const bool finite = wet && !nan;
95 const Long flat = Long(j - dlo.y) * nx + Long(i - dlo.x);
96 return { wet ? Long(1) : Long(0),
97 bad ? Long(1) : Long(0),
98 nan ? Long(1) : Long(0),
99 finite ? v : big,
100 finite ? v : -big,
101 bad ? flat : no_cell };
102 });
103 }
104 auto r = data.value(ops);
105 Long counts[3] = {get<0>(r), get<1>(r), get<2>(r)};
106 Real extrema[2] = {get<3>(r), -get<4>(r)}; // max(v) = -min(-v)
107 Long first = get<5>(r);
108 ParallelDescriptor::ReduceLongSum(counts, 3);
109 ParallelDescriptor::ReduceRealMin(extrema, 2);
110 ParallelDescriptor::ReduceLongMin(first);
111
112 WetRangeCheck out{counts[0], extrema[0], -extrema[1], counts[1],
113 counts[2], IntVect(0), big};
114 if (out.bad_count == 0) { return out; }
115
116 // Only when something is out of range: fetch the named cell's value.
117 out.example = IntVect(domain.smallEnd(0) + static_cast<int>(first % nx),
118 domain.smallEnd(1) + static_cast<int>(first / nx),
119 k0);
120 // Empty off the owning rank. Exactly one rank owns the cell, so a sum with
121 // zero elsewhere carries its value, NaN included, which a min would not.
122 const auto here = get_cell_data(mf, out.example);
123 out.example_value = here.empty() ? Real(0) : here[comp];
124 ParallelDescriptor::ReduceRealSum(out.example_value);
125 return out;
126}

132// Source index region each destination box's stencils actually reference.
133//
134// The entries in index_mf are *source* indices, and for non-conformal grids they
135// routinely fall far outside the destination box's own index range, so staging the
136// source on `dst.boxArray()` plus a fixed ghost halo leaves those reads outside
137// the valid region - zero at best, out of bounds at worst. Mirror of the same
138// helper on the ERF side of the coupling.
139//
140// The result becomes a BoxArray, which must be globally consistent, hence the
141// reduction.
142amrex::BoxArray
143StagedSourceBoxArray (const amrex::MultiFab& src,
144 const amrex::MultiFab& dst,
145 const amrex::iMultiFab& index_mf,
146 int max_stencil_size)
147{
148 using namespace amrex;
149
150 const Box src_domain = src.boxArray().minimalBox();
151 const int nboxes = static_cast<int>(dst.boxArray().size());
152 constexpr int int_big = std::numeric_limits<int>::max();
153
154 Vector<int> lo(2 * nboxes, int_big);
155 Vector<int> hi(2 * nboxes, -int_big);
156
157 for (MFIter mfi(dst); mfi.isValid(); ++mfi) {
158 const int b = mfi.index();
159 const Box bx = mfi.validbox();
160 auto const& idx = index_mf.const_array(mfi);
161
162 ReduceOps<ReduceOpMin, ReduceOpMin, ReduceOpMax, ReduceOpMax> reduce_op;
163 ReduceData<int, int, int, int> reduce_data(reduce_op);
164 using ReduceTuple = typename decltype(reduce_data)::Type;
165
166 reduce_op.eval(bx, reduce_data,
167 [=] AMREX_GPU_DEVICE (int i, int j, int k) -> ReduceTuple
168 {
169 int i_min = int_big, j_min = int_big, i_max = -int_big, j_max = -int_big;
170 for (int m = 0; m < max_stencil_size; ++m) {
171 const int si = idx(i, j, k, m * 3);
172 const int sj = idx(i, j, k, m * 3 + 1);
173 if (si < 0 || sj < 0) { continue; } // -1 marks an unused slot
174 i_min = amrex::min(i_min, si); i_max = amrex::max(i_max, si);
175 j_min = amrex::min(j_min, sj); j_max = amrex::max(j_max, sj);
176 }
177 return {i_min, j_min, i_max, j_max};
178 });
179
180 auto const& hv = reduce_data.value(reduce_op);
181 lo[2*b ] = amrex::get<0>(hv); lo[2*b+1] = amrex::get<1>(hv);
182 hi[2*b ] = amrex::get<2>(hv); hi[2*b+1] = amrex::get<3>(hv);
183 }
184
185 ParallelDescriptor::ReduceIntMin(lo.dataPtr(), static_cast<int>(lo.size()));
186 ParallelDescriptor::ReduceIntMax(hi.dataPtr(), static_cast<int>(hi.size()));
187
188 // The staged boxes must carry the source's index type from the start:
189 // intersecting a cell-centered box with a staggered one trips
190 // Box::operator&='s sameType assertion.
191 const IndexType src_ixtype = src.boxArray().ixType();
192
193 BoxList bl(src_ixtype);
194 for (int b = 0; b < nboxes; ++b) {
195 Box need;
196 if (lo[2*b] > hi[2*b] || lo[2*b+1] > hi[2*b+1]) {
197 // No stencil anywhere in this destination box. A BoxArray cannot hold
198 // an empty box, so stage a single cell; nothing reads it.
199 need = Box(src_domain.smallEnd(), src_domain.smallEnd(), src_ixtype);
200 } else {
201 need = Box(IntVect(lo[2*b], lo[2*b+1], src_domain.smallEnd(2)),
202 IntVect(hi[2*b], hi[2*b+1], src_domain.bigEnd(2)),
203 src_ixtype);
204 need &= src_domain;
205 }
206 bl.push_back(need);
207 }
208 return BoxArray(std::move(bl));
209}

211// fallback_val fills destination cells that no source cell overlaps.
212//
213// Zero is the wrong default for a physical field: the receiving model cannot tell
214// "no data here" from "the ocean says zero", and for an intensive quantity zero is
215// not merely inaccurate, it drives the receiver's flux formulas outside their valid
216// domain. This mirrors the rationale on the atmosphere->ocean twin at
217// ERF/Source/Coupling/ERF_to_REMORA.cpp:122.
218//
219// The ocean->atmosphere SST lane deliberately leaves this at zero and relies on the
220// per-cell coverage flag instead: where REMORA does not cover an ERF cell, ERF keeps
221// its own wrflowinp SST rather than consuming a value invented here. That is a
222// withholding contract, not a value, so no climatological constant belongs at that
223// call site. The parameter exists because the mask blend below needs it and because
224// other lanes have a meaningful value to pass.
225void
226ApplyConservativeRemap (const amrex::MultiFab& src,
227 amrex::MultiFab& dst,
228 const amrex::MultiFab& weight_mf,
229 const amrex::iMultiFab& index_mf,
230 int max_stencil_size,
231 const amrex::MultiFab* dst_mask = nullptr,
232 const amrex::iMultiFab* dst_land_mask = nullptr,
233 amrex::Real fallback_val = amrex::Real(0.0))
234{
235 using namespace amrex;
236
237 // 1. Data Routing: Route REMORA SST data onto the ERF atmospheric layout,
238 // staging exactly the source region the stencils reference.
239 MultiFab src_on_dst(StagedSourceBoxArray(src, dst, index_mf, max_stencil_size),
240 dst.DistributionMap(), src.nComp(), 0);
241 src_on_dst.setVal(0.0);
242 src_on_dst.ParallelCopy(src);
243
244 dst.setVal(0.0);
245
246 // 2. Stencil Application: Execute the local sparse dot product
247 for (MFIter mfi(dst, TilingIfNotGPU()); mfi.isValid(); ++mfi) {
248 Box bx = mfi.tilebox();
249
250 auto const& w_arr = weight_mf.const_array(mfi);
251 auto const& idx_arr = index_mf.const_array(mfi);
252 auto const& src_arr = src_on_dst.const_array(mfi);
253 auto dst_arr = dst.array(mfi);
254 const bool has_mask = (dst_mask != nullptr);
255 auto const& mask_arr = has_mask ? dst_mask->const_array(mfi) : Array4<const Real>{};
256 // ERF land mask: 0 = water, anything non-zero is land. Destination cells
257 // over land are zeroed out (wet/dry masking only, no vector rotation).
258 const bool has_land_mask = (dst_land_mask != nullptr);
259 auto const& land_arr = has_land_mask ? dst_land_mask->const_array(mfi) : Array4<const int>{};
260
261 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k) {
262 // No stencil entry at all means no source cell overlapped this
263 // destination cell. Hand back the fallback rather than an accumulated
264 // zero, and do not apply the masks to it: a masked-out cell still gets
265 // evaluated by the receiving model's flux formulas before the mask is
266 // applied, so it too must hold an admissible value. Mirrors
267 // ERF/Source/Coupling/ERF_to_REMORA.cpp:174.
268 if (idx_arr(i, j, k, 0) < 0) {
269 dst_arr(i, j, k) = fallback_val;
270 return;
271 }
272
273 Real sum = 0.0;
274 for (int m = 0; m < max_stencil_size; ++m) {
275 Real w = w_arr(i, j, k, m);
276 if (w > 0.0) {
277 int src_i = idx_arr(i, j, k, m * 3);
278 int src_j = idx_arr(i, j, k, m * 3 + 1);
279 int src_k = idx_arr(i, j, k, m * 3 + 2);
280
281 sum += w * src_arr(src_i, src_j, src_k);
282 }
283 }
284 // Blend toward the fallback rather than multiplying by the mask.
285 // Multiplying drives partially masked cells toward exactly zero, which
286 // is right for a flux lane (fallback_val = 0, so this reduces to
287 // sum *= mask and is bit-identical) but destructive for an intensive
288 // state lane such as SST, where it scales a temperature toward 0 K.
289 // Mirrors ERF/Source/Coupling/ERF_to_REMORA.cpp:202.
290 if (has_mask) {
291 const Real mask = mask_arr(i, j, k);
292 sum = mask * sum + (Real(1.0) - mask) * fallback_val;
293 }
294 // Test against zero, not against 1: ERF stamps lmask = 2 for cells
295 // under ImmersedForcing buildings (ERF/Source/ERF_MakeNewArrays.cpp:890),
296 // so an == 1 test reads every building as water and hands it a remapped
297 // SST. The driver's atmosphere->ocean side already uses the tolerant
298 // form (ERFRemoraMultiBlockContainer.cpp:1767), so == 1 also made the
299 // two directions disagree about the same cell within one timestep.
300 if (has_land_mask && land_arr(i, j, k) != 0) { sum = Real(0.0); }
301 dst_arr(i, j, k) = sum;
302 });
303 }
304}

307amrex::Real
308REMORA::EvolveOneStep (amrex::Real /*time*/, amrex::Real /*dt_request*/)
309{
310 Real cur_time = t_new[0];
311 const int step = istep[0];
312
313 if (cur_time >= elapsed_time(stop_time)) {
314 return zero;
315 }
316
317 ComputeDt();
318 update_wvel_before_checkpoint_step(step, cur_time, elapsed_time(stop_time));
319
320 int lev = 0;
321 int iteration = 1;
322 // Must match Evolve's choice: the subcycle-only paths are gated on do_substep, not on
323 // which driver ran, and timeStepML does not register the mass flux they read.
324 if (max_level == 0 || do_substep) {
325 timeStep(lev, cur_time, iteration);
326 } else {
327 timeStepML(cur_time, iteration);
328 }
329
330 cur_time += dt[0];
331
332 WriteAtIntermediateTime(step, cur_time);
333
334 post_timestep(step, cur_time, dt[0]);
335
336 return dt[0];
337}

494/*
495 * \brief Extracts SST from the 3D conservative state for the atmospheric driver.
496 *
497 * Reads Temp_comp at the top water-column cell (k_sfc), converts from
498 * Celsius to Kelvin, and conservatively remaps the result into state[SSTIndex].
499 */
500void
501REMORA::PackSurfaceState (Vector<MultiFab*>& state,
502 Real /*time*/,
503 const amrex::MultiFab* weight_o2a_mf,
504 const amrex::iMultiFab* index_o2a_mf,
505 int max_stencil_size,
506 const amrex::iMultiFab* dst_land_mask)
507{
508 if (state.empty() || state[SSTIndex] == nullptr) { return; }
509 const int lev = 0;
510
511 // REMORA stores temperature in Celsius. Surface is at k=N (top of water column).
512 const int k_sfc = cons_new[lev]->boxArray().minimalBox().bigEnd(2);
513
514 // Build a temp MultiFab on REMORA's ba2d (k=0) derived from cons_new's BoxArray.
515 BoxList bl2d = cons_new[lev]->boxArray().boxList();
516 for (auto& b : bl2d) { b.setRange(2, 0); }
517 BoxArray ba2d(std::move(bl2d));
518 MultiFab tmp(ba2d, cons_new[lev]->DistributionMap(), 1, 0);
519
520 for (MFIter mfi(*cons_new[lev]); mfi.isValid(); ++mfi) {
521 auto const& c = cons_new[lev]->const_array(mfi);
522 auto t = tmp.array(mfi);
523 Box bx = makeSlab(mfi.validbox(), 2, k_sfc);
524 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int) {
525 // Write to k=0 in tmp (ba2d range); convert Celsius → Kelvin.
526 t(i, j, 0) = c(i, j, k_sfc, Temp_comp) + Real(273.15);
527 });
528 }
529
530 // Surface temperature sanity check on REMORA's own wet cells, in the SST
531 // that is about to be sent to the atmosphere: one pass, once per exchange
532 // (this runs at pack time, not every ERF substep). Warned the first time
533 // and again only when it gets worse; remora.v >= 1 also prints the wet
534 // min/max.
535 // min/max exclude NaN, so the NaN count is tracked separately. verbose and
536 // the reduced values are rank-uniform, so every rank takes the same
537 // branches.
538 if (!vec_mskr.empty() && vec_mskr[lev] != nullptr) {
539 MultiFab wet(ba2d, cons_new[lev]->DistributionMap(), 1, 0);
540 wet.setVal(zero);
541 wet.ParallelCopy(*vec_mskr[lev], 0, 0, 1);
542
543 constexpr Real t_lo = Real(273.15 - 2.5); // below seawater freezing
544 constexpr Real t_hi = Real(273.15 + 40.0);
545 const WetRangeCheck c = CheckWetRange(tmp, 0, wet, t_lo, t_hi);
546 if (verbose >= 1) {
547 amrex::Print os;
548 os << "REMORA surface temperature over " << c.wet_cells
549 << " wet cells: ";
550 if (c.has_finite()) {
551 os << "min/max = " << c.min_value << " / " << c.max_value
552 << " K";
553 } else {
554 os << "no finite values";
555 }
556 os << "\n";
557 }
558 // With no finite wet cell, min/max are the +/-max sentinels, which
559 // never compare as worse; the NaN count still does.
560 const bool worse = c.min_value < m_warned_surface_temp_min ||
561 c.max_value > m_warned_surface_temp_max ||
562 c.nan_count > m_warned_surface_temp_nan;
563 if (c.bad_count > 0 && worse) {
564 m_warned_surface_temp_min =
565 std::min(m_warned_surface_temp_min, c.min_value);
566 m_warned_surface_temp_max =
567 std::max(m_warned_surface_temp_max, c.max_value);
568 m_warned_surface_temp_nan =
569 std::max(m_warned_surface_temp_nan, c.nan_count);
570 amrex::Print os;
571 os << "WARNING: REMORA surface temperature outside [" << t_lo
572 << ", " << t_hi << "] K on " << c.bad_count << " of "
573 << c.wet_cells << " wet cells (" << c.nan_count << " NaN";
574 if (c.has_finite()) {
575 os << "; min/max " << c.min_value << " / " << c.max_value
576 << " K";
577 }
578 os << "; e.g. REMORA (" << c.example[0] << "," << c.example[1]
579 << ") = " << c.example_value
580 << " K). Repeated only if it gets worse.\n";
581 }
582 }
583
584 MultiFab& dst = *state[SSTIndex];
585
586 if (weight_o2a_mf != nullptr && index_o2a_mf != nullptr) {
587 // Execute sparse conservative remap from REMORA SST to ERF layout
588 ApplyConservativeRemap(tmp, dst, *weight_o2a_mf, *index_o2a_mf, max_stencil_size,
589 nullptr, dst_land_mask);
590 } else {
591 // Fallback for un-stenciled or synthetic runs
592 dst.setVal(zero);
593 dst.ParallelCopy(tmp, 0, 0, 1);
594 }

602{
603 running_with_coupling_driver = true;
604 driver_atmos_forcing_mode = DriverAtmosForcingMode::State;
605 driver_atmos_state_from_driver.fill(false);
606 if (finest_level < 0) { return; }
607
608 // Wind (m/s) — no unit conversion
609 if (vec_uwind[0] != nullptr) {
610 if (states.size() > AtmosState::Uwind && states[AtmosState::Uwind] != nullptr) {
611 vec_uwind[0]->ParallelCopy(*states[AtmosState::Uwind], 0, 0, 1);
612 vec_uwind[0]->FillBoundary(geom[0].periodicity());
613 driver_atmos_state_from_driver[AtmosState::Uwind] = true;

615 }
616 if (vec_vwind[0] != nullptr) {
617 if (states.size() > AtmosState::Vwind && states[AtmosState::Vwind] != nullptr) {
618 vec_vwind[0]->ParallelCopy(*states[AtmosState::Vwind], 0, 0, 1);
619 vec_vwind[0]->FillBoundary(geom[0].periodicity());
620 driver_atmos_state_from_driver[AtmosState::Vwind] = true;
621 }
622 }
623
624 // Atmospheric pressure: Pa → mb (REMORA bulk flux expects mb)
625 if (vec_Pair[0] != nullptr) {
626 if (states.size() > AtmosState::Pair && states[AtmosState::Pair] != nullptr) {
627 vec_Pair[0]->ParallelCopy(*states[AtmosState::Pair], 0, 0, 1);

629 vec_Pair[0]->FillBoundary(geom[0].periodicity());
630 driver_atmos_state_from_driver[AtmosState::Pair] = true;

634 // Specific humidity (kg/kg) — no conversion

637 vec_qair[0]->ParallelCopy(*states[AtmosState::Qair], 0, 0, 1);
638 vec_qair[0]->FillBoundary(geom[0].periodicity());
639 driver_atmos_state_from_driver[AtmosState::Qair] = true;

641 }
642
643 // Air temperature: K → °C (REMORA stores/uses Celsius internally)
644 if (vec_Tair[0] != nullptr) {
645 if (states.size() > AtmosState::Tair && states[AtmosState::Tair] != nullptr) {
646 vec_Tair[0]->ParallelCopy(*states[AtmosState::Tair], 0, 0, 1);
647 vec_Tair[0]->plus(Real(-273.15), 0, 1);
648 vec_Tair[0]->FillBoundary(geom[0].periodicity());
649 driver_atmos_state_from_driver[AtmosState::Tair] = true;
650 }
651 }
652
653 // Cloud fraction [0-1], rain, SW/LW radiation — no unit conversion
654 if (vec_cloud[0] != nullptr) {
655 if (states.size() > AtmosState::Cloud && states[AtmosState::Cloud] != nullptr) {
656 vec_cloud[0]->ParallelCopy(*states[AtmosState::Cloud], 0, 0, 1);
657 vec_cloud[0]->FillBoundary(geom[0].periodicity());
658 driver_atmos_state_from_driver[AtmosState::Cloud] = true;
659 }
660 }
661 if (vec_rain[0] != nullptr) {
662 if (states.size() > AtmosState::Rain && states[AtmosState::Rain] != nullptr) {
663 vec_rain[0]->ParallelCopy(*states[AtmosState::Rain], 0, 0, 1);
664 vec_rain[0]->FillBoundary(geom[0].periodicity());
665 driver_atmos_state_from_driver[AtmosState::Rain] = true;
666 }
667 }
668 if (vec_srflx[0] != nullptr) {
669 if (states.size() > AtmosState::SWrad && states[AtmosState::SWrad] != nullptr) {
670 vec_srflx[0]->ParallelCopy(*states[AtmosState::SWrad], 0, 0, 1);
671 vec_srflx[0]->FillBoundary(geom[0].periodicity());
672 driver_atmos_state_from_driver[AtmosState::SWrad] = true;
673 }
674 }
675 if (vec_longwave_down[0] != nullptr) {
676 if (states.size() > AtmosState::LWrad && states[AtmosState::LWrad] != nullptr) {
677 vec_longwave_down[0]->ParallelCopy(*states[AtmosState::LWrad], 0, 0, 1);
678 vec_longwave_down[0]->FillBoundary(geom[0].periodicity());
679 driver_atmos_state_from_driver[AtmosState::LWrad] = true;
680 }
681 }

685void
686REMORA::ApplyAtmosphericFluxes (const Vector<MultiFab*>& states, Real /*time*/)
687{
688 running_with_coupling_driver = true;
689 driver_atmos_forcing_mode = DriverAtmosForcingMode::Flux;
690 driver_atmos_state_from_driver.fill(false);
691 if (finest_level < 0) { return; }
692
693 if (states.size() <= AtmosFluxes::Evap ||
694 states[AtmosFluxes::TauX] == nullptr ||
695 states[AtmosFluxes::TauY] == nullptr ||
696 states[AtmosFluxes::SHflux] == nullptr ||
697 states[AtmosFluxes::LHflux] == nullptr ||
698 states[AtmosFluxes::SWrad] == nullptr ||
699 states[AtmosFluxes::LWrad] == nullptr ||
700 states[AtmosFluxes::Rain] == nullptr ||
701 states[AtmosFluxes::Evap] == nullptr ||
702 vec_sustr[0] == nullptr || vec_svstr[0] == nullptr ||
703 vec_stflux[0] == nullptr || vec_mskr[0] == nullptr ||
704 vec_msku[0] == nullptr || vec_mskv[0] == nullptr ||
705 vec_srflx[0] == nullptr || vec_lrflx[0] == nullptr ||
706 vec_lhflx[0] == nullptr || vec_shflx[0] == nullptr ||
707 vec_rain[0] == nullptr || vec_evap[0] == nullptr) {
708 return;
709 }
710
711 const Real Hscale2 = one / (solverChoice.rho0 * Cp);
712 const Real rho0 = solverChoice.rho0;
713
714 MultiFab tau_x_tmp(vec_sustr[0]->boxArray(), vec_sustr[0]->DistributionMap(), 1,
715 vec_sustr[0]->nGrowVect());
716 MultiFab tau_y_tmp(vec_svstr[0]->boxArray(), vec_svstr[0]->DistributionMap(), 1,
717 vec_svstr[0]->nGrowVect());
718 MultiFab shflux_tmp(vec_shflx[0]->boxArray(), vec_shflx[0]->DistributionMap(), 1,
719 vec_shflx[0]->nGrowVect());
720 MultiFab lhflux_tmp(vec_lhflx[0]->boxArray(), vec_lhflx[0]->DistributionMap(), 1,
721 vec_lhflx[0]->nGrowVect());
722 MultiFab lwflux_tmp(vec_lrflx[0]->boxArray(), vec_lrflx[0]->DistributionMap(), 1,
723 vec_lrflx[0]->nGrowVect());
724
725 tau_x_tmp.setVal(zero);
726 tau_x_tmp.ParallelCopy(*states[AtmosFluxes::TauX], 0, 0, 1);
727 tau_x_tmp.FillBoundary(geom[0].periodicity());
728
729 tau_y_tmp.setVal(zero);
730 tau_y_tmp.ParallelCopy(*states[AtmosFluxes::TauY], 0, 0, 1);
731 tau_y_tmp.FillBoundary(geom[0].periodicity());
732
733 shflux_tmp.setVal(zero);
734 shflux_tmp.ParallelCopy(*states[AtmosFluxes::SHflux], 0, 0, 1);
735 shflux_tmp.FillBoundary(geom[0].periodicity());
736
737 lhflux_tmp.setVal(zero);
738 lhflux_tmp.ParallelCopy(*states[AtmosFluxes::LHflux], 0, 0, 1);
739 lhflux_tmp.FillBoundary(geom[0].periodicity());
740
741 lwflux_tmp.setVal(zero);
742 lwflux_tmp.ParallelCopy(*states[AtmosFluxes::LWrad], 0, 0, 1);
743 lwflux_tmp.FillBoundary(geom[0].periodicity());
744
745 vec_srflx[0]->setVal(zero);
746 vec_srflx[0]->ParallelCopy(*states[AtmosFluxes::SWrad], 0, 0, 1);
747 vec_srflx[0]->FillBoundary(geom[0].periodicity());
748
749 vec_rain[0]->setVal(zero);
750 vec_rain[0]->ParallelCopy(*states[AtmosFluxes::Rain], 0, 0, 1);
751 vec_rain[0]->FillBoundary(geom[0].periodicity());
752
753 vec_evap[0]->setVal(zero);
754 vec_evap[0]->ParallelCopy(*states[AtmosFluxes::Evap], 0, 0, 1);
755 vec_evap[0]->FillBoundary(geom[0].periodicity());
756
757 vec_lrflx[0]->setVal(zero);
758 vec_lhflx[0]->setVal(zero);
759 vec_shflx[0]->setVal(zero);
760 vec_stflux[0]->setVal(zero);
761
762 for (MFIter mfi(*vec_sustr[0], TilingIfNotGPU()); mfi.isValid(); ++mfi) {
763 Array4<Real> const& sustr = vec_sustr[0]->array(mfi);
764 Array4<const Real> const& msku = vec_msku[0]->const_array(mfi);
765 Array4<const Real> const& tau_x = tau_x_tmp.const_array(mfi);
766 Box ubx = mfi.grownnodaltilebox(0, IntVect(NGROW,NGROW,0));
767 Box ubxD = ubx;
768 ubxD.makeSlab(2,0);
769 ParallelFor(ubxD, [=] AMREX_GPU_DEVICE (int i, int j, int ) {
770 sustr(i,j,0) = -tau_x(i,j,0) / rho0 * msku(i,j,0);
771 });
772 }
773
774 for (MFIter mfi(*vec_svstr[0], TilingIfNotGPU()); mfi.isValid(); ++mfi) {
775 Array4<Real> const& svstr = vec_svstr[0]->array(mfi);
776 Array4<const Real> const& mskv = vec_mskv[0]->const_array(mfi);
777 Array4<const Real> const& tau_y = tau_y_tmp.const_array(mfi);
778 Box vbx = mfi.grownnodaltilebox(1, IntVect(NGROW,NGROW,0));
779 Box vbxD = vbx;
780 vbxD.makeSlab(2,0);
781
782 ParallelFor(vbxD, [=] AMREX_GPU_DEVICE (int i, int j, int ) {
783 svstr(i,j,0) = -tau_y(i,j,0) / rho0 * mskv(i,j,0);
784 });
785 }
786
787 for (MFIter mfi(*vec_stflux[0], TilingIfNotGPU()); mfi.isValid(); ++mfi) {
788 Array4<Real> const& stflux = vec_stflux[0]->array(mfi);
789 Array4<Real> const& lrflx = vec_lrflx[0]->array(mfi);
790 Array4<Real> const& lhflx = vec_lhflx[0]->array(mfi);
791 Array4<Real> const& shflx = vec_shflx[0]->array(mfi);
792 Array4<const Real> const& mskr = vec_mskr[0]->const_array(mfi);
793 Array4<const Real> const& srflx = vec_srflx[0]->const_array(mfi);
794 Array4<const Real> const& rain = vec_rain[0]->const_array(mfi);
795 Array4<const Real> const& evap = vec_evap[0]->const_array(mfi);
796 Array4<const Real> const& shflux = shflux_tmp.const_array(mfi);
797 Array4<const Real> const& lhflux = lhflux_tmp.const_array(mfi);
798 Array4<const Real> const& lwflux = lwflux_tmp.const_array(mfi);
799
800 Box gbx2 = mfi.growntilebox(IntVect(NGROW,NGROW,0));
801 Box gbx2D = gbx2;
802 gbx2D.makeSlab(2,0);
803
804 ParallelFor(gbx2D, [=] AMREX_GPU_DEVICE (int i, int j, int ) {
805 lrflx(i,j,0) = lwflux(i,j,0) * Hscale2;
806 lhflx(i,j,0) = -lhflux(i,j,0) * Hscale2;
807 shflx(i,j,0) = -shflux(i,j,0) * Hscale2;
808 stflux(i,j,0,Temp_comp) =
809 (srflx(i,j,0) * Hscale2 + lrflx(i,j,0) + lhflx(i,j,0) + shflx(i,j,0))
810 * mskr(i,j,0);
811 stflux(i,j,0,Salt_comp) =
812 mskr(i,j,0) * (evap(i,j,0) - rain(i,j,0)) / rhow;
813 });
814 }
815
816 vec_sustr[0]->FillBoundary(geom[0].periodicity());
817 vec_svstr[0]->FillBoundary(geom[0].periodicity());
818 vec_srflx[0]->FillBoundary(geom[0].periodicity());
819 vec_lrflx[0]->FillBoundary(geom[0].periodicity());
820 vec_lhflx[0]->FillBoundary(geom[0].periodicity());
821 vec_shflx[0]->FillBoundary(geom[0].periodicity());
822 vec_stflux[0]->FillBoundary(geom[0].periodicity());
823 vec_rain[0]->FillBoundary(geom[0].periodicity());
824 vec_evap[0]->FillBoundary(geom[0].periodicity());
825 vec_stflux[0]->FillBoundary(geom[0].periodicity());
826
827 // Once per run, or every exchange at remora.v >= 1. verbose is a ParmParse
828 // value, so every rank takes the same branch around the collectives.
829 if (m_reported_atm_flux_validation && verbose < 1) { return; }
830 m_reported_atm_flux_validation = true;

833 const Row rows[] = {
834 {"sustr", vec_sustr[0].get(), 0, vec_msku[0].get()},
835 {"svstr", vec_svstr[0].get(), 0, vec_mskv[0].get()},

837 {"stflux(Salt)", vec_stflux[0].get(), Salt_comp, vec_mskr[0].get()},
838 {"srflx", vec_srflx[0].get(), 0, vec_mskr[0].get()},
839 {"lrflx", vec_lrflx[0].get(), 0, vec_mskr[0].get()},
840 {"lhflx", vec_lhflx[0].get(), 0, vec_mskr[0].get()},
841 {"shflx", vec_shflx[0].get(), 0, vec_mskr[0].get()},
842 };
843 amrex::Print() << "REMORA ApplyAtmosphericFluxes validation (wet cells only; srflx W/m2, "
844 << "others kinematic):\n";
845 for (const auto& r : rows) {
846 const WetRangeCheck e = CheckWetRange(*r.mf, r.comp, *r.mask);
847 amrex::Print os;
848 os << " " << r.name << ": ";
849 if (e.wet_cells == 0) {
850 os << "no wet cells";
851 } else if (!e.has_finite()) {
852 os << "all " << e.wet_cells << " wet cells NaN";
853 } else {
854 os << "min=" << e.min_value << " max=" << e.max_value;

856 }
857 os << "\n";
858 }
