232REMORA::append3DPlotVariables (const std::string& pp_plot_var_names_3d)
233{
234 ParmParse pp(pp_prefix);
235
236 if (pp.contains("plot_vars")) {
237 amrex::Abort("You must use plot_vars_3d rather than plot_vars");
238 }
239
240 // This runs after the particle containers are set up, so the only thing it may add is
241 // a particle mesh variable, which set3DPlotVariables could not know about yet. Adding
242 // anything else back would resurrect a name set3DPlotVariables deliberately dropped as
243 // unavailable, and nothing downstream would ever fill it: the plotfile would carry the
244 // uninitialized sentinel from WritePlotFile under a real-looking variable name. The
245 // Biology group aliases were expanded by set3DPlotVariables.
246#ifdef REMORA_USE_PARTICLES
247 Vector<std::string> particle_mesh_plot_names;
248 particleData.GetMeshPlotVarNames( particle_mesh_plot_names );
249
250 if (pp.contains(pp_plot_var_names_3d.c_str())) {
251 std::string nm;
252 int nPltVars = pp.countval(pp_plot_var_names_3d.c_str());
253 for (int i = 0; i < nPltVars; i++) {
254 pp.get(pp_plot_var_names_3d.c_str(), nm, i);
255 if (containerHasElement(particle_mesh_plot_names, nm) &&
256 !containerHasElement(plot_var_names_3d, nm)) {
257 plot_var_names_3d.push_back(nm);
258 }
259 }
260 }
261#else
262 amrex::ignore_unused(pp_plot_var_names_3d);
263#endif
264}
