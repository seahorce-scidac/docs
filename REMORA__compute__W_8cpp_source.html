5/**
6 * Vertical volume flux across the w-levels (ROMS omega times the cell area), from the
7 * vertical integral of the horizontal mass flux divergence. W is zero at the bed and at
8 * the free surface.
9 *
10 * @param[in ] bx2d k = 0 slab of the columns to compute
11 * @param[ out] W vertical volume flux
12 * @param[in ] Huon u-volume flux
13 * @param[in ] Hvom v-volume flux
14 * @param[in ] z_w z coordinates at w points
15 * @param[in ] h bathymetry
16 * @param[in ] N number of vertical levels minus one
17 */
18void
19REMORA::compute_W (const Box& bx2d,
20 const Array4<Real >& W,
21 const Array4<Real const>& Huon,
22 const Array4<Real const>& Hvom,
23 const Array4<Real const>& z_w,
24 const Array4<Real const>& h,
25 const int N)
26{
27 ParallelFor(bx2d, [=] AMREX_GPU_DEVICE (int i, int j, int )
28 {
29 W(i,j,0) = zero;
30 });
31 ParallelFor(bx2d, N+1,
32 [=] AMREX_GPU_DEVICE (int i, int j, int , int kk)
33 {
34 // Starting with zero vertical velocity at the bottom, integrate
35 // from the bottom (k=0) to the free-surface (k=N). The w(:,:,N(ng))
36 // contains the vertical velocity at the free-surface, d(zeta)/d(t).
37 // Notice that barotropic mass flux divergence is not used directly.
38 //
39 int k = kk + 1;
40 W(i,j,k) = W(i,j,k-1) - (Huon(i+1,j,k-1)-Huon(i,j,k-1)) - (Hvom(i,j+1,k-1)-Hvom(i,j,k-1));
41 });
42 ParallelFor(bx2d, [=] AMREX_GPU_DEVICE (int i, int j, int )
43 {
44 W(i,j,N+1)=W(i,j,N+1)/(z_w(i,j,N+1)+h(i,j,0,0)); // wrk_i
45 });
46 ParallelFor(bx2d, N,
47 [=] AMREX_GPU_DEVICE (int i, int j, int , int kk)
48 {
49 int k = kk + 1;
50 W(i,j,k) = W(i,j,k)- W(i,j,N+1)*(z_w(i,j,k)+h(i,j,0,0));
51 });
52 ParallelFor(bx2d, [=] AMREX_GPU_DEVICE (int i, int j, int )
53 {
54 W(i,j,N+1) = zero;
55 });
56}

58/**
59 * W converted from a volume flux to a velocity, as ROMS scale_omega. It is zero at the bed
60 * and the free surface.
61 *
62 * If both mf_omega and the fluxes have ghost cells, the first ghost column is computed
63 * directly from the fluxes there, as advance_3d computes W, so it is right at domain and
64 * coarse/fine boundaries too. The rest of the ghost cells are then filled across grids and
65 * periodic boundaries, and are zero elsewhere outside the domain.
66 *
67 * @param[in ] lev level
68 * @param[out] mf_omega z-nodal omega (m/s)
69 * @param[in ] mf_Huon u-volume flux on the valid faces, and on the first ghost faces for
70 * the directly computed ghost column
71 * @param[in ] mf_Hvom v-volume flux, likewise
72 */
73void
74REMORA::compute_omega (int lev, MultiFab& mf_omega,
75 const MultiFab& mf_Huon, const MultiFab& mf_Hvom)
76{
77 const int N = Geom(lev).Domain().size()[2]-1;
78
79 // The first ghost column is computed only where both the output and the fluxes have it:
80 // the particles' omega has it from the fluxes update_massflux_3d keeps on grown boxes,
81 // while the diagnostic omega builds its fluxes on the valid faces only.
82 IntVect ng(0);
83 for (int d = 0; d < 2; ++d) {
84 ng[d] = amrex::min(amrex::min(mf_omega.nGrowVect()[d], 1),
85 amrex::min(mf_Huon.nGrowVect()[d], mf_Hvom.nGrowVect()[d]));
86 }
87
88 mf_omega.setVal(zero);
89
90 // Not tiled: compute_W integrates whole columns
91#ifdef _OPENMP
92#pragma omp parallel if (Gpu::notInLaunchRegion())
93#endif
94 for (MFIter mfi(mf_omega); mfi.isValid(); ++mfi)
95 {
96 Array4<Real> const& W = mf_omega.array(mfi);
97
98 Array4<Real const> const& pm = vec_pm[lev]->const_array(mfi);
99 Array4<Real const> const& pn = vec_pn[lev]->const_array(mfi);
100
101 Box bx2d = makeSlab(grow(mfi.validbox(), ng), 2, 0);
102 compute_W(bx2d, W, mf_Huon.const_array(mfi), mf_Hvom.const_array(mfi),
103 vec_z_w[lev]->const_array(mfi), vec_h[lev]->const_array(mfi), N);
104
105 ParallelFor(bx2d, N, [=] AMREX_GPU_DEVICE (int i, int j, int, int kk)
106 {
107 W(i,j,kk+1) *= pm(i,j,0) * pn(i,j,0);
108 });
109 }
110
111 mf_omega.FillBoundary(geom[lev].periodicity());
112}

132/**
133 * The "true" vertical velocity w (m/s) at w-points, a port of ROMS wvelocity. In
134 * terrain-following coordinates
135 *
136 * Hz * omega = w - d(z)/d(t) - div(z), div(z) = pm * u * d(z)/d(xi) + pn * v * d(z)/d(eta)
137 *
138 * so w adds back to omega the stretching of the levels by the moving free surface
139 * (from the barotropic flux divergence) and the motion along the sloping levels, which
140 * is computed at rho-points and shifted to w-points with ROMS's cubic interpolation.
141 * Like ROMS, this is for output only: nothing in the dynamics reads it, and omega comes
142 * from the mass fluxes of the current velocities (compute_diagnostic_omega).
143 *
144 * @param[in ] lev level
145 * @param[out] mf_w z-nodal w on the valid region
146 */
147void
148REMORA::compute_wvel (int lev, MultiFab& mf_w)
149{
150 MultiFab mf_omega(convert(grids[lev],IntVect(0,0,1)), dmap[lev], 1, IntVect(0));
151 compute_diagnostic_omega(lev, mf_omega);
152 compute_wvel(lev, mf_w, mf_omega);
153}

155/**
156 * @param[in ] lev level
157 * @param[out] mf_w z-nodal w on the valid region
158 * @param[in ] mf_omega z-nodal omega from compute_diagnostic_omega, on the valid region
159 */
160void
161REMORA::compute_wvel (int lev, MultiFab& mf_w, const MultiFab& mf_omega)
162{
163 const int nz = Geom(lev).Domain().length(2);
164 // The cubic stencil reaches three rho-levels in from each end
165 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(nz >= 3, "compute_wvel needs at least 3 vertical levels");
166
167 const Real cff1 = Real(3.0)/Real(8.0);
168 const Real cff2 = Real(3.0)/Real(4.0);
169 const Real cff3 = Real(1.0)/Real(8.0);
170 const Real cff4 = Real(9.0)/Real(16.0);
171 const Real cff5 = Real(1.0)/Real(16.0);
172
173 // Not tiled: each column of w reads the whole column of vert
174#ifdef _OPENMP
175#pragma omp parallel if (Gpu::notInLaunchRegion())
176#endif
177 for (MFIter mfi(mf_omega); mfi.isValid(); ++mfi)
178 {
179 Array4<Real > const& wvel = mf_w.array(mfi);
180 Array4<Real const> const& omega = mf_omega.const_array(mfi);
181
182 Array4<Real const> const& u = xvel_new[lev]->const_array(mfi);
183 Array4<Real const> const& v = yvel_new[lev]->const_array(mfi);
184 Array4<Real const> const& z_r = vec_z_r[lev]->const_array(mfi);
185 Array4<Real const> const& z_w = vec_z_w[lev]->const_array(mfi);
186 Array4<Real const> const& pm = vec_pm[lev]->const_array(mfi);
187 Array4<Real const> const& pn = vec_pn[lev]->const_array(mfi);
188 Array4<Real const> const& DU_avg1 = vec_DU_avg1[lev]->const_array(mfi);
189 Array4<Real const> const& DV_avg1 = vec_DV_avg1[lev]->const_array(mfi);
190
191 const Box bx = mfi.validbox().enclosedCells(2);
192 FArrayBox fab_vert(bx,1,amrex::The_Async_Arena());
193 auto vert = fab_vert.array();
194
195 // Motion along the s-levels, (Ui + Vj) * GRADs(z), at rho-points
196 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k)
197 {
198 const Real wrk_ulo = u(i ,j,k) * (z_r(i ,j,k)-z_r(i-1,j,k)) * (pm(i-1,j,0)+pm(i ,j,0));
199 const Real wrk_uhi = u(i+1,j,k) * (z_r(i+1,j,k)-z_r(i ,j,k)) * (pm(i ,j,0)+pm(i+1,j,0));
200 const Real wrk_vlo = v(i,j ,k) * (z_r(i,j ,k)-z_r(i,j-1,k)) * (pn(i,j-1,0)+pn(i,j ,0));
201 const Real wrk_vhi = v(i,j+1,k) * (z_r(i,j+1,k)-z_r(i,j ,k)) * (pn(i,j ,0)+pn(i,j+1,0));
202 vert(i,j,k) = Real(0.25)*(wrk_ulo+wrk_uhi) + Real(0.25)*(wrk_vlo+wrk_vhi);
203 });
204
205 // ROMS k (w-levels 0..N, rho-levels 1..N) maps to REMORA k (w-levels 0..nz,
206 // rho-levels 0..nz-1), so ROMS vert(k) is vert(k-1) here
207 ParallelFor(makeSlab(bx,2,0), [=] AMREX_GPU_DEVICE (int i, int j, int)
208 {
209 // Free-surface stretching: d(zeta)/d(t) spread linearly through the column
210 const Real wrk = (DU_avg1(i,j,0)-DU_avg1(i+1,j,0) + DV_avg1(i,j,0)-DV_avg1(i,j+1,0))
211 / (z_w(i,j,nz)-z_w(i,j,0));
212 const Real pmpn = pm(i,j,0)*pn(i,j,0);
213 auto interior = [&] (int k) {
214 return omega(i,j,k) + pmpn*wrk*(z_w(i,j,k)-z_w(i,j,0));
215 };
216
217 Real slope = (z_r(i,j,0)-z_w(i,j,0)) / (z_r(i,j,1)-z_r(i,j,0)); // extrapolation slope
218 wvel(i,j,0) = cff1*(vert(i,j,0) - slope*(vert(i,j,1)-vert(i,j,0)))
219 + cff2*vert(i,j,0) - cff3*vert(i,j,1);
220 wvel(i,j,1) = interior(1) + cff1*vert(i,j,0) + cff2*vert(i,j,1) - cff3*vert(i,j,2);
221 for (int k = 2; k <= nz-2; ++k) {
222 wvel(i,j,k) = interior(k) + cff4*(vert(i,j,k-1)+vert(i,j,k ))
223 - cff5*(vert(i,j,k-2)+vert(i,j,k+1));
224 }
225 slope = (z_w(i,j,nz)-z_r(i,j,nz-1)) / (z_r(i,j,nz-1)-z_r(i,j,nz-2)); // extrapolation slope
226 wvel(i,j,nz) = pmpn*wrk*(z_w(i,j,nz)-z_w(i,j,0))
227 + cff1*(vert(i,j,nz-1) + slope*(vert(i,j,nz-1)-vert(i,j,nz-2)))
228 + cff2*vert(i,j,nz-1) - cff3*vert(i,j,nz-2);
229 wvel(i,j,nz-1) = interior(nz-1) + cff1*vert(i,j,nz-1) + cff2*vert(i,j,nz-2)
230 - cff3*vert(i,j,nz-3);
231 });
232 }
233}

243void
244REMORA::set_massflux_3d (int lev, MultiFab& mf_Huon, MultiFab& mf_Hvom)
245{
246#ifdef _OPENMP
247#pragma omp parallel if (Gpu::notInLaunchRegion())
248#endif

250 {
251 Array4<Real > const& Huon = mf_Huon.array(mfi);
252 Array4<Real const> const& u = xvel_new[lev]->const_array(mfi);

256 ParallelFor(mfi.tilebox(), [=] AMREX_GPU_DEVICE (int i, int j, int k)
257 {
258 Real on_u = two / (pn(i-1,j,0)+pn(i,j,0));
259 Huon(i,j,k)=Real(0.5)*(Hz(i,j,k)+Hz(i-1,j,k))*u(i,j,k)* on_u;
260 });
261 }
262#ifdef _OPENMP
263#pragma omp parallel if (Gpu::notInLaunchRegion())
264#endif
265 for (MFIter mfi(mf_Hvom, TilingIfNotGPU()); mfi.isValid(); ++mfi)
266 {
267 Array4<Real > const& Hvom = mf_Hvom.array(mfi);
268 Array4<Real const> const& v = yvel_new[lev]->const_array(mfi);
269 Array4<Real const> const& Hz = vec_Hz[lev]->const_array(mfi);
270 Array4<Real const> const& pm = vec_pm[lev]->const_array(mfi);
271
272 ParallelFor(mfi.tilebox(), [=] AMREX_GPU_DEVICE (int i, int j, int k)
273 {
274 Real om_v= two / (pm(i,j-1,0)+pm(i,j,0));

276 });
277 }
