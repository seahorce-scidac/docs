1// Analytic biology profile for BioToy and Upwelling, from ROMS ana_biology.h.
2//
3// A code fragment included from Problem::init_analytic_biology in
4// Exec/REMORA_Prob.cpp (and from the biology unit test). It needs only
5// `biology` (REMORABiology::Biology const&) and `context`
6// (REMORABiology::BiologyInitContext&) in scope, and locates its fields by
7// tracer name, so it never sees FennelParameters. Optional Fennel pools that
8// the run does not carry have index -1 and are skipped; every write is
9// guarded by its own index.
10//
11// The values are Fennel's, in Fennel's units, so the profile refuses any
12// other model even if it reuses the same tracer names.
13//
14// The profile is a function of temperature alone, so it carries over unchanged
15// to any problem that stratifies temperature.
16{
17 if (biology.name() != "fennel") {
18 amrex::Error("The BioToy analytic biology profile is Fennel's;"
19 " remora.biology_model = " + biology.name() +
20 " needs its own profile or"
21 " remora.biology_ic_type = netcdf.");
22 }
23 auto& mf_cons = context.state;
24 auto const& bio_declaration = biology.declaration();
25 auto comp = [&] (const std::string& name) {
26 return bio_declaration.component_index(name, context.component_start);
27 };
28 const int no3 = comp("NO3");
29 const int nh4 = comp("NH4");

45 AMREX_ALWAYS_ASSERT(no3 >= 0 && nh4 >= 0 && chlo >= 0 && phyt >= 0 &&
46 zoop >= 0 && lden >= 0 && sden >= 0);

49 Real cff2=Real( 2.0)/Real(3.0);
50 for (MFIter mfi(mf_cons, TilingIfNotGPU()); mfi.isValid(); ++mfi) {
51 Array4<const Real> const& temp = mf_cons.array(mfi,Temp_comp);
52 Array4< Real> const& cons = mf_cons.array(mfi);

54 Box bx = mfi.growntilebox(mf_cons.nGrowVect());

56 ParallelFor(bx, [=] AMREX_GPU_DEVICE (int i, int j, int k) {
57 Real SiO4;
58 if (temp(i,j,k) < Real(8.0)) {
59 SiO4 = Real(30.0);
60 } else if ((temp(i,j,k) >= Real(8.0)) && (temp(i,j,k) <= Real(11.0))) {
61 SiO4 = Real(30.0) - ((temp(i,j,k) - Real(8.0))*cff1);
62 } else if ((temp(i,j,k) > Real(11.0)) && (temp(i,j,k) <= Real(13.0))) {
63 SiO4 = Real(10.0) - ((temp(i,j,k) - Real(11.0))*Real(4.0));
64 } else if ((temp(i,j,k) > Real(13.0)) && (temp(i,j,k) <= Real(16.0))) {
65 SiO4 = Real(2.0) - ((temp(i,j,k) - Real(13.0))*cff2);
66 } else { //else if (temp(i,j,k) > Real(16.0))
67 SiO4 = Real(0.0);
68 }

70 Real(0.0144)*SiO4*SiO4+
71 Real(0.0003099)*SiO4*SiO4*SiO4;
72 cons(i,j,k,phyt)=Real(0.08);
73 cons(i,j,k,zoop)=Real(0.06);
74 cons(i,j,k,nh4)=Real(0.1);
75 cons(i,j,k,lden)=Real(0.02);
76 cons(i,j,k,sden)=Real(0.04);
77 cons(i,j,k,chlo)=Real(0.02);
78 if (oxyg >= 0) {

80 }
81 if (tic >= 0) {
82 cons(i,j,k,tic)=Real(2100.0);
83 }
84 if (talk >= 0) {
85 cons(i,j,k,talk)=Real(2350.0);
86 }
87 if (ldec >= 0) {
88 cons(i,j,k,ldec)=Real(0.002);
89 }
90 if (sdec >= 0) {
91 cons(i,j,k,sdec)=Real(0.06);
92 }
93 if (rden >= 0) {
94 // ROMS ana_biology.h has no river DON profile, so pick values
95 // that make the pool visible: 0.05 mmol_N/m3 is the same order
96 // as SdetritusN above, so its remineralization to NH4 is a
97 // real contribution rather than a rounding artifact, and a
98 // zero start would leave the whole option a no-op no matter
99 // whether the terms are right.
100 cons(i,j,k,rden)=Real(0.05);
101 }
102 if (rdec >= 0) {
103 cons(i,j,k,rdec)=Real(0.3);
104 }
105 if (po4 >= 0) {
106 // 0.5 mmol_P/m3: a plausible coastal subsurface value, far
107 // above the 1e-6 MinVal floor that would otherwise mask the
108 // uptake terms. With the default K_PO4=32 this gives
109 // L_PO4 ~ 0.94, deliberately the same order as LTOT so that
110 // MIN(LTOT, L_PO4) is a real comparison rather than a
111 // foregone conclusion.
112 //
113 // Note the PO4 branch of biology_tile is structurally
114 // different from the default one (mu-based cff4/cff5, and
115 // LMIN rather than LTOT in the chlorophyll denominator), and
116 // it is taken for any non-zero PO4 regardless of which side
117 // the MIN selects. Which side it selects does depend on the
118 // NO3/NH4 field, so a run that changes those should re-check
119 // that both sides still get exercised.
120 cons(i,j,k,po4)=Real(0.5);
121 }
122 if (odu >= 0) {
123 // ODU is inert on both paths: ROMS fennel.h never references
124 // iODU_ (zero occurrences), and the native kernel only clamps
125 // it on entry and writes it back on exit. This value exists so
126 // the tracer-count, index-mapping and pack/unpack comparison
127 // is not vacuous -- initialising it to zero would compare
128 // 0 against 0 and prove nothing about the plumbing.
129 cons(i,j,k,odu)=Real(1.0);
130 }
131 });
